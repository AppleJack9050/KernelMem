import torch
import torch.nn as nn
import triton
import triton.language as tl


@triton.jit
def flash_swa_kernel(
    Q, K, V, O,
    scale,
    N: tl.constexpr,
    D: tl.constexpr,
    WINDOW: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
):
    start_m = tl.program_id(0)
    off_bh = tl.program_id(1)

    base = off_bh * N * D

    offs_m = start_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_d = tl.arange(0, D)

    q_ptrs = Q + base + offs_m[:, None] * D + offs_d[None, :]
    q = tl.load(q_ptrs, mask=offs_m[:, None] < N, other=0.0)

    m_i = tl.full((BLOCK_M,), float("-inf"), dtype=tl.float32)
    l_i = tl.zeros((BLOCK_M,), dtype=tl.float32)
    acc = tl.zeros((BLOCK_M, D), dtype=tl.float32)

    # key range restricted to the causal sliding window
    lo = start_m * BLOCK_M - WINDOW + 1
    lo = tl.maximum(lo, 0)
    lo = (lo // BLOCK_N) * BLOCK_N
    hi = start_m * BLOCK_M + BLOCK_M

    for start_n in range(lo, hi, BLOCK_N):
        offs_n = start_n + tl.arange(0, BLOCK_N)
        n_valid = offs_n < N

        k_ptrs = K + base + offs_n[:, None] * D + offs_d[None, :]
        k = tl.load(k_ptrs, mask=n_valid[:, None], other=0.0)  # (BLOCK_N, D)

        # TF32 tensor-core MMA with fp32 accumulation
        qk = tl.dot(q, tl.trans(k), input_precision="tf32") * scale  # (BLOCK_M, BLOCK_N)

        rel = offs_m[:, None] - offs_n[None, :]
        mask = (rel >= 0) & (rel < WINDOW) & n_valid[None, :]
        qk = tl.where(mask, qk, float("-inf"))

        m_ij = tl.max(qk, 1)
        m_new = tl.maximum(m_i, m_ij)

        # Guard against -inf - (-inf) = NaN for fully-masked rows/blocks.
        m_safe = tl.where(m_new == float("-inf"), 0.0, m_new)

        p = tl.exp(qk - m_safe[:, None])
        alpha = tl.exp(m_i - m_safe)

        l_i = l_i * alpha + tl.sum(p, 1)

        v_ptrs = V + base + offs_n[:, None] * D + offs_d[None, :]
        v = tl.load(v_ptrs, mask=n_valid[:, None], other=0.0)  # (BLOCK_N, D)

        # TF32 tensor-core MMA; v is already fp32, keep fp32 accumulator
        acc = acc * alpha[:, None] + tl.dot(p, v, input_precision="tf32")
        m_i = m_new

    l_i = tl.where(l_i == 0.0, 1.0, l_i)
    out = acc / l_i[:, None]

    o_ptrs = O + base + offs_m[:, None] * D + offs_d[None, :]
    tl.store(o_ptrs, out.to(O.dtype.element_ty), mask=offs_m[:, None] < N)


class ModelNew(nn.Module):
    def __init__(self, num_heads: int, head_dim: int, window_size: int):
        super().__init__()
        self.num_heads = num_heads
        self.head_dim = head_dim
        self.window_size = window_size
        self.scale = head_dim ** -0.5

    def forward(self, q: torch.Tensor, k: torch.Tensor, v: torch.Tensor) -> torch.Tensor:
        B, H, N, D = q.shape

        q = q if q.is_contiguous() else q.contiguous()
        k = k if k.is_contiguous() else k.contiguous()
        v = v if v.is_contiguous() else v.contiguous()

        qf = q.view(B * H, N, D)
        kf = k.view(B * H, N, D)
        vf = v.view(B * H, N, D)
        out = torch.empty_like(qf)

        BLOCK_M = 64
        BLOCK_N = 64

        grid = (triton.cdiv(N, BLOCK_M), B * H)
        # num_stages 2->1 halves per-block shared-mem footprint to lift occupancy;
        # num_warps=8 retained. Anti-regression: keep TF32 + fp32 accumulator.
        flash_swa_kernel[grid](
            qf, kf, vf, out,
            self.scale,
            N=N,
            D=D,
            WINDOW=self.window_size,
            BLOCK_M=BLOCK_M,
            BLOCK_N=BLOCK_N,
            num_warps=8,
            num_stages=1,
        )
        return out.view(B, H, N, D)
