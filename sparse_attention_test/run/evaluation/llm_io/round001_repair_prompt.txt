import torch
import torch.nn as nn

# Sliding-window (local) causal self-attention.
#
# Each query position i attends only to key positions j with
# (i - window_size) < j <= i  -- i.e. a causal band of width `window_size`.
# This is the sparse-attention pattern used by e.g. Mistral / Longformer.
#
# This reference is the "obvious" dense PyTorch implementation: it materializes
# the full (B, H, N, N) score matrix, masks out everything outside the causal
# window, softmaxes, and multiplies by V. It is deliberately the naive baseline
# -- it does O(N^2) work and memory even though only an O(N * window) band is
# ever used. A good custom kernel (FlashAttention-style, iterating only over the
# key blocks inside each query's window) can skip the masked region entirely.


class Model(nn.Module):
    def __init__(self, num_heads: int, head_dim: int, window_size: int):
        super().__init__()
        self.num_heads = num_heads
        self.head_dim = head_dim
        self.window_size = window_size
        self.scale = head_dim ** -0.5

    def forward(self, q: torch.Tensor, k: torch.Tensor, v: torch.Tensor) -> torch.Tensor:
        # q, k, v: (B, H, N, D)
        B, H, N, D = q.shape

        scores = torch.matmul(q, k.transpose(-2, -1)) * self.scale  # (B, H, N, N)

        idx = torch.arange(N, device=q.device)
        rel = idx[:, None] - idx[None, :]            # i - j
        mask = (rel >= 0) & (rel < self.window_size)  # causal AND inside the window
        scores = scores.masked_fill(~mask[None, None, :, :], float("-inf"))

        attn = torch.softmax(scores, dim=-1)          # (B, H, N, N)
        out = torch.matmul(attn, v)                   # (B, H, N, D)
        return out

import torch
import torch.nn as nn
import triton
import triton.language as tl


# Granularity: (D) fully rewrite forward.
# Replaced ops: the entire QK^T scale + windowed-causal mask + softmax + AV
#   dense attention is replaced by a single FlashAttention-style Triton kernel
#   (sliding_window_attn_kernel) that iterates ONLY over the key blocks inside
#   each query's causal window, skipping the masked O(N^2) region entirely.
# Fusion: matmul(QK^T) + scale + windowed-causal masking + online softmax +
#   matmul(attn,V) are all fused into sliding_window_attn_kernel.
# Remaining PyTorch ops: none in the attention math (only shape/alloc glue),
#   because a fused flash kernel avoids materializing the (B,H,N,N) scores.


@triton.jit
def sliding_window_attn_kernel(
    Q, K, V, Out,
    N, WINDOW, scale,
    stride_bh, stride_n, stride_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, D: tl.constexpr,
):
    pid_m = tl.program_id(0)
    pid_bh = tl.program_id(1)

    base = pid_bh * stride_bh
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_d = tl.arange(0, D)

    q_ptrs = base + offs_m[:, None] * stride_n + offs_d[None, :] * stride_d
    q_mask = offs_m[:, None] < N
    q = tl.load(Q + q_ptrs, mask=q_mask, other=0.0)

    m_i = tl.full((BLOCK_M,), float("-inf"), dtype=tl.float32)
    l_i = tl.zeros((BLOCK_M,), dtype=tl.float32)
    acc = tl.zeros((BLOCK_M, D), dtype=tl.float32)

    start_m = pid_m * BLOCK_M
    # key positions relevant: j > i-WINDOW and j <= i, over the query block
    lo = start_m - WINDOW + 1
    lo = tl.maximum(lo, 0)
    lo = (lo // BLOCK_N) * BLOCK_N
    hi = tl.minimum(start_m + BLOCK_M, N)

    for start_n in range(lo, hi, BLOCK_N):
        offs_n = start_n + tl.arange(0, BLOCK_N)
        k_ptrs = base + offs_n[:, None] * stride_n + offs_d[None, :] * stride_d
        n_mask = offs_n[:, None] < N
        k = tl.load(K + k_ptrs, mask=n_mask, other=0.0)

        qk = tl.dot(q, tl.trans(k)) * scale  # (BLOCK_M, BLOCK_N)

        rel = offs_m[:, None] - offs_n[None, :]
        valid = (rel >= 0) & (rel < WINDOW) & (offs_n[None, :] < N)
        qk = tl.where(valid, qk, float("-inf"))

        m_new = tl.maximum(m_i, tl.max(qk, 1))
        alpha = tl.exp(m_i - m_new)
        p = tl.exp(qk - m_new[:, None])

        l_i = l_i * alpha + tl.sum(p, 1)
        acc = acc * alpha[:, None]

        v = tl.load(V + k_ptrs, mask=n_mask, other=0.0)
        acc += tl.dot(p.to(v.dtype), v)

        m_i = m_new

    l_safe = tl.where(l_i == 0.0, 1.0, l_i)
    out = acc / l_safe[:, None]

    o_ptrs = base + offs_m[:, None] * stride_n + offs_d[None, :] * stride_d
    tl.store(Out + o_ptrs, out.to(Out.dtype.element_ty), mask=q_mask)


def sliding_window_attention(q, k, v, window, scale):
    q = q if q.is_contiguous() else q.contiguous()
    k = k if k.is_contiguous() else k.contiguous()
    v = v if v.is_contiguous() else v.contiguous()

    B, H, N, D = q.shape
    BH = B * H
    out = torch.empty_like(q)

    qf = q.view(BH, N, D)
    kf = k.view(BH, N, D)
    vf = v.view(BH, N, D)
    of = out.view(BH, N, D)

    BLOCK_M = 64
    BLOCK_N = 64

    grid = (triton.cdiv(N, BLOCK_M), BH)
    sliding_window_attn_kernel[grid](
        qf, kf, vf, of,
        N, window, scale,
        qf.stride(0), qf.stride(1), qf.stride(2),
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, D=D,
        num_warps=4, num_stages=2,
    )
    return out
