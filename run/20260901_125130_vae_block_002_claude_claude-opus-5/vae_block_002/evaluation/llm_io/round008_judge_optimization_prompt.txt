_SRC = r"""
#include <torch/extension.h>
#include <cuda_runtime.h>
#include <ATen/cuda/CUDAContext.h>
#include <c10/cuda/CUDAException.h>

#define TP 32

__device__ __forceinline__ float silu_f(float v) {
    return v / (1.0f + __expf(-v));
}

// ---------------------------------------------------------------- statistics
// x: NHWC contiguous. one block handles (n,g) pixel-segment s.
__global__ void gn_stats_kernel(const float* __restrict__ x,
                                float* __restrict__ partial,   // [2, BG, S]
                                int HW, int C, int G, int CPG, int S, int BG) {
    const int s  = blockIdx.x;
    const int bg = blockIdx.y;
    const int n  = bg / G;
    const int g  = bg - n * G;

    const long long base = (long long)n * HW * (long long)C + (long long)g * CPG;
    const int p_start = (int)(((long long)HW * s) / S);
    const int p_end   = (int)(((long long)HW * (s + 1)) / S);
    const int cpg4    = CPG >> 2;

    float sum = 0.f, sq = 0.f;
    const long long cnt = (long long)(p_end - p_start) * cpg4;
    for (long long i = threadIdx.x; i < cnt; i += blockDim.x) {
        int p = p_start + (int)(i / cpg4);
        int j = (int)(i - (long long)(p - p_start) * cpg4);
        const float4 v = *reinterpret_cast<const float4*>(x + base + (long long)p * C + (j << 2));
        sum += v.x + v.y + v.z + v.w;
        sq  += v.x * v.x + v.y * v.y + v.z * v.z + v.w * v.w;
    }

    // block reduce
    __shared__ float ws[2][32];
    unsigned mask = 0xffffffffu;
    #pragma unroll
    for (int off = 16; off > 0; off >>= 1) {
        sum += __shfl_down_sync(mask, sum, off);
        sq  += __shfl_down_sync(mask, sq,  off);
    }
    int lane = threadIdx.x & 31, warp = threadIdx.x >> 5;
    if (lane == 0) { ws[0][warp] = sum; ws[1][warp] = sq; }
    __syncthreads();
    int nw = blockDim.x >> 5;
    if (warp == 0) {
        sum = (lane < nw) ? ws[0][lane] : 0.f;
        sq  = (lane < nw) ? ws[1][lane] : 0.f;
        #pragma unroll
        for (int off = 16; off > 0; off >>= 1) {
            sum += __shfl_down_sync(mask, sum, off);
            sq  += __shfl_down_sync(mask, sq,  off);
        }
        if (lane == 0) {
            partial[(long long)bg * S + s]                 = sum;
            partial[(long long)BG * S + (long long)bg * S + s] = sq;
        }
    }
}

__global__ void gn_finalize_kernel(const float* __restrict__ partial,
                                   float* __restrict__ mean,
                                   float* __restrict__ rstd,
                                   int S, int BG, float inv_count, float eps) {
    const int bg = blockIdx.x;
    float sum = 0.f, sq = 0.f;
    for (int i = threadIdx.x; i < S; i += blockDim.x) {
        sum += partial[(long long)bg * S + i];
        sq  += partial[(long long)BG * S + (long long)bg * S + i];
    }
    __shared__ float ws[2][32];
    unsigned mask = 0xffffffffu;
    #pragma unroll
    for (int off = 16; off > 0; off >>= 1) {
        sum += __shfl_down_sync(mask, sum, off);
        sq  += __shfl_down_sync(mask, sq,  off);
    }
    int lane = threadIdx.x & 31, warp = threadIdx.x >> 5;
    if (lane == 0) { ws[0][warp] = sum; ws[1][warp] = sq; }
    __syncthreads();
    int nw = blockDim.x >> 5;
    if (warp == 0) {
        sum = (lane < nw) ? ws[0][lane] : 0.f;
        sq  = (lane < nw) ? ws[1][lane] : 0.f;
        #pragma unroll
        for (int off = 16; off > 0; off >>= 1) {
            sum += __shfl_down_sync(mask, sum, off);
            sq  += __shfl_down_sync(mask, sq,  off);
        }
        if (lane == 0) {
            float m = sum * inv_count;
            float v = sq * inv_count - m * m;
            v = v < 0.f ? 0.f : v;
            mean[bg] = m;
            rstd[bg] = rsqrtf(v + eps);
        }
    }
}

// ------------------------------------------- GN(affine) + SiLU, NHWC -> NHWC
__global__ void gn_silu_nhwc_kernel(const float* __restrict__ x,
                                    const float* __restrict__ gamma,
                                    const float* __restrict__ beta,
                                    const float* __restrict__ mean,
                                    const float* __restrict__ rstd,
                                    float* __restrict__ y,
                                    int HWC4, int C4, int CPG, int G) {
    const int n = blockIdx.y;
    int idx = blockIdx.x * blockDim.x + threadIdx.x;
    if (idx >= HWC4) return;
    const int c0 = (idx % C4) << 2;
    const int g  = c0 / CPG;
    const int bg = n * G + g;
    const float mu = mean[bg], rs = rstd[bg];

    const long long off = (long long)n * HWC4 + idx;
    const float4 v  = *reinterpret_cast<const float4*>(x + (off << 2));
    const float4 gm = *reinterpret_cast<const float4*>(gamma + c0);
    const float4 bt = *reinterpret_cast<const float4*>(beta + c0);
    float4 o;
    o.x = silu_f((v.x - mu) * rs * gm.x + bt.x);
    o.y = silu_f((v.y - mu) * rs * gm.y + bt.y);
    o.z = silu_f((v.z - mu) * rs * gm.z + bt.z);
    o.w = silu_f((v.w - mu) * rs * gm.w + bt.w);
    *reinterpret_cast<float4*>(y + (off << 2)) = o;
}

// ---------------- GN(affine) + SiLU + residual add, NHWC in -> NHWC out
__global__ void gn_silu_res_nhwc_kernel(const float* __restrict__ x,
                                        const float* __restrict__ res,
                                        const float* __restrict__ gamma,
                                        const float* __restrict__ beta,
                                        const float* __restrict__ mean,
                                        const float* __restrict__ rstd,
                                        float* __restrict__ out,
                                        int HWC4, int C4, int CPG, int G) {
    const int n = blockIdx.y;
    int idx = blockIdx.x * blockDim.x + threadIdx.x;
    if (idx >= HWC4) return;
    const int c0 = (idx % C4) << 2;
    const int g  = c0 / CPG;
    const int bg = n * G + g;
    const float mu = mean[bg], rs = rstd[bg];

    const long long off = (long long)n * HWC4 + idx;
    const float4 v  = *reinterpret_cast<const float4*>(x + (off << 2));
    const float4 r  = *reinterpret_cast<const float4*>(res + (off << 2));
    const float4 gm = *reinterpret_cast<const float4*>(gamma + c0);
    const float4 bt = *reinterpret_cast<const float4*>(beta + c0);
    float4 o;
    o.x = silu_f((v.x - mu) * rs * gm.x + bt.x) + r.x;
    o.y = silu_f((v.y - mu) * rs * gm.y + bt.y) + r.y;
    o.z = silu_f((v.z - mu) * rs * gm.z + bt.z) + r.z;
    o.w = silu_f((v.w - mu) * rs * gm.w + bt.w) + r.w;
    *reinterpret_cast<float4*>(out + (off << 2)) = o;
}

// (kept as unused fallback / dead code per plan item 3; not called by ModelNew)
__global__ void gn_silu_res_nchw_kernel(const float* __restrict__ x,
                                        const float* __restrict__ res,
                                        const float* __restrict__ gamma,
                                        const float* __restrict__ beta,
                                        const float* __restrict__ mean,
                                        const float* __restrict__ rstd,
                                        float* __restrict__ out,
                                        int HW, int C, int G, int CPG) {
    extern __shared__ float sm[];
    const int bg = blockIdx.y;
    const int n  = bg / G;
    const int g  = bg - n * G;
    const int p0 = blockIdx.x * TP;
    const float mu = mean[bg], rs = rstd[bg];

    const long long base = (long long)n * HW * (long long)C + (long long)g * CPG;
    const int total = CPG * TP;
    for (int i = threadIdx.x; i < total; i += blockDim.x) {
        int c = i % CPG;
        int p = i / CPG;
        int pg = p0 + p;
        if (pg < HW) {
            float v = x[base + (long long)pg * C + c];
            v = (v - mu) * rs * gamma[g * CPG + c] + beta[g * CPG + c];
            sm[c * (TP + 1) + p] = silu_f(v);
        }
    }
    __syncthreads();
    const long long obase = (long long)n * C * (long long)HW + (long long)g * CPG * HW;
    for (int i = threadIdx.x; i < total; i += blockDim.x) {
        int p = i % TP;
        int c = i / TP;
        int pg = p0 + p;
        if (pg < HW) {
            long long oi = obase + (long long)c * HW + pg;
            out[oi] = sm[c * (TP + 1) + p] + res[oi];
        }
    }
}

static void compute_stats(const at::Tensor& x, int B, int HW, int C, int G,
                          float eps, at::Tensor& mean, at::Tensor& rstd) {
    const int CPG = C / G;
    const int BG = B * G;
    int S = (456 + BG - 1) / BG;
    int maxS = (HW + 255) / 256;
    if (S > maxS) S = maxS;
    if (S < 1) S = 1;
    auto opts = x.options();
    auto partial = at::empty({2, BG, S}, opts);
    auto stream = at::cuda::getCurrentCUDAStream();

    dim3 grid(S, BG);
    gn_stats_kernel<<<grid, 256, 0, stream>>>(
        x.data_ptr<float>(), partial.data_ptr<float>(), HW, C, G, CPG, S, BG);
    C10_CUDA_KERNEL_LAUNCH_CHECK();

    float inv_count = 1.0f / (float)((long long)HW * CPG);
    gn_finalize_kernel<<<BG, 256, 0, stream>>>(
        partial.data_ptr<float>(), mean.data_ptr<float>(), rstd.data_ptr<float>(),
        S, BG, inv_count, eps);
    C10_CUDA_KERNEL_LAUNCH_CHECK();
}

        super().__init__()

    def forward(self, x, conv1_weight, norm1_weight, norm1_bias,

        B, C, H, W = x.shape
        HW = H * W

            # generic fallback (keeps semantics for unsupported configs)
            out = F.conv2d(x, conv1_weight, None, 1, 1)
            out = F.group_norm(out, G, norm1_weight, norm1_bias, eps)
            out = F.silu(out)
            out = F.conv2d(out, conv2_weight, None, 1, 1)
            out = F.group_norm(out, G, norm2_weight, norm2_bias, eps)
            out = F.silu(out)
            return out + x

        w1 = conv1_weight.contiguous(memory_format=torch.channels_last)
        w2 = conv2_weight.contiguous(memory_format=torch.channels_last)
        g1 = norm1_weight if norm1_weight.is_contiguous() else norm1_weight.contiguous()
        b1 = norm1_bias if norm1_bias.is_contiguous() else norm1_bias.contiguous()
        g2 = norm2_weight if norm2_weight.is_contiguous() else norm2_weight.contiguous()
        b2 = norm2_bias if norm2_bias.is_contiguous() else norm2_bias.contiguous()

        y = _ext.gn_silu_nhwc(y, g1, b1, B, HW, C, G, float(eps))

      (b) fuse all post-ops into ONE custom kernel consuming the mainloop output to minimize write-backs.
  - Only propose rewriting mainloops with custom kernels if metrics + fixed shapes justify outperforming vendor libs.
  - Exception (GEMM/Matmul/Linear): you MAY propose switching to cuBLASLt Matmul (or changing computeType/algo/layout/epilogue) when doing so is necessary to enable Tensor Cores / TF32 and is consistent with the fp32-tolerance contract.

- Do NOT modify or redefine get_inputs() / get_init_inputs().
- Inputs are already on the correct device; do NOT move tensors across devices in forward.
- Outputs must be allocated on the SAME device as inputs.

- **Bottleneck identified**: [memory/layout-transform bound] The conv mainloop (58.8%) is already at SOL (~296us/conv = 261 TFLOPS TF32 vs 223 measured peak), but its satellite `elementwise_kernel` layout copies hold 15.4% steady (1858us/36 launches = 155us per forward) doing zero math: x->channels_last plus two OIHW->OHWI weight permutes, whose strided TensorIterator gathers waste read sectors.
- **Optimization method**: Replace the three PyTorch TensorIterator permute-copies with ONE custom shared-memory tiled NCHW->NHWC transpose kernel (32x32 tile, +1 padding) so both the read (32 contiguous S elements) and the write (32 contiguous C elements) are 128B-coalesced. Justification E2: identical semantics, same bytes, but removes the sector-utilization loss of the stride-S/stride-9 gather that makes these no-math kernels cost 2-4x their DRAM-ideal 78us. Conv, GN and residual mechanisms untouched.
- **Method name**: smem_tiling_transpose
- **Modification plan**: SMEM BUDGET CHECK (mandatory): tile is 32x33 floats = 4224 B per block, buffers=1, vs 227 KB/block limit on H100 -> passes by 50x; occupancy limit from smem is >32 blocks/SM, so no gate violation.

1. In _SRC add the generic kernel (exact index math):
```
__global__ void transpose_cs_kernel(const float* __restrict__ src,
                                    float* __restrict__ dst,
                                    int C, int S) {
  __shared__ float tile[32][33];
  const int n = blockIdx.z;
  const int p0 = blockIdx.x * 32, c0 = blockIdx.y * 32;
  const int tx = threadIdx.x, ty = threadIdx.y;   // blockDim=(32,8)
  const long long sbase = (long long)n * C * S;
  #pragma unroll
  for (int k = 0; k < 32; k += 8) {
    int c = c0 + ty + k, p = p0 + tx;
    if (c < C && p < S) tile[ty + k][tx] = src[sbase + (long long)c * S + p];
  }
  __syncthreads();
  const long long dbase = (long long)n * S * C;
  #pragma unroll
  for (int k = 0; k < 32; k += 8) {
    int c = c0 + tx, p = p0 + ty + k;
    if (c < C && p < S) dst[dbase + (long long)p * C + c] = tile[tx][ty + k];
  }
}
```
Reads: a warp fetches 32 consecutive p (128B). Writes: a warp writes 32 consecutive c (128B). smem read stride 33 -> bank-conflict free.
