```python
# ============================================================================
# ModelNew  --  granularity (C): "fuse many ops into one/few kernels"
#
# 1) Chosen granularity: (C) fuse many ops into a few custom CUDA kernels.
# 2) Ops replaced by custom CUDA: both GroupNorm's (mean/var reduction +
#    affine), both SiLU's, the residual add, and the final NHWC->NCHW layout
#    restoration (which in the reference is a separate cuDNN nhwcToNchw pass).
# 3) Fusion map:
#      - gn_stats_kernel + gn_finalize_kernel : per-(n,group) sum/sumsq ->
#        mean/rstd  (replaces at::RowwiseMomentsCUDAKernel + ComputeFusedParams)
#      - gn_silu_nhwc_kernel  : normalize * gamma + beta, then SiLU, in one
#        pass over NHWC data (replaces 2 elementwise kernels of GroupNorm #1)
#      - gn_silu_res_nchw_kernel : normalize + affine + SiLU + residual add
#        AND the NHWC->NCHW transpose, all in one shared-memory-tiled kernel
#        (replaces 3 elementwise kernels + cudnn nhwcToNchw of GroupNorm #2)
#      - the whole pipeline runs in channels_last, so cuDNN's 4x nchwToNhwc /
#        2x nhwcToNchw layout kernels (21% of reference GPU time) disappear;
#        only one cheap NCHW->NHWC conversion of the input remains.
# 4) Left in PyTorch:
#      - the two 3x3 convolutions: they are the tuned vendor tensor-core
#        implicit-GEMM (sm90 TF32) already running near roofline; feeding them
#        channels_last hits the exact same kernel the reference reaches after
#        its layout conversions, so numerics are unchanged and 21% of the
#        runtime is removed for free.
#      - the one-off input NCHW->channels_last copy: a single library copy is
#        already bandwidth-bound.
# ============================================================================

_SRC = r"""
#include <torch/extension.h>
#include <cuda_runtime.h>
#include <ATen/cuda/CUDAContext.h>
#include <c10/cuda/CUDAException.h>

#define TP 32

__device__ __forceinline__ float silu_f(float v) {
    return v / (1.0f + __expf(-v));
}

// ---------------------------------------------------------------- statistics
// x: NHWC contiguous. one block handles (n,g) pixel-segment s.
__global__ void gn_stats_kernel(const float* __restrict__ x,
                                float* __restrict__ partial,   // [2, BG, S]
                                int HW, int C, int G, int CPG, int S, int BG) {
    const int s  = blockIdx.x;
    const int bg = blockIdx.y;
    const int n  = bg / G;
    const int g  = bg - n * G;

    const long long base = (long long)n * HW * (long long)C + (long long)g * CPG;
    const int p_start = (int)(((long long)HW * s) / S);
    const int p_end   = (int)(((long long)HW * (s + 1)) / S);
    const int cpg4    = CPG >> 2;

    float sum = 0.f, sq = 0.f;
    const long long cnt = (long long)(p_end - p_start) * cpg4;
    for (long long i = threadIdx.x; i < cnt; i += blockDim.x) {
        int p = p_start + (int)(i / cpg4);
        int j = (int)(i - (long long)(p - p_start) * cpg4);
        const float4 v = *reinterpret_cast<const float4*>(x + base + (long long)p * C + (j << 2));
        sum += v.x + v.y + v.z + v.w;
        sq  += v.x * v.x + v.y * v.y + v.z * v.z + v.w * v.w;
    }

    // block reduce
    __shared__ float ws[2][32];
    unsigned mask = 0xffffffffu;
    #pragma unroll
    for (int off = 16; off > 0; off >>= 1) {
        sum += __shfl_down_sync(mask, sum, off);
        sq  += __shfl_down_sync(mask, sq,  off);
    }
    int lane = threadIdx.x & 31, warp = threadIdx.x >> 5;
    if (lane == 0) { ws[0][warp] = sum; ws[1][warp] = sq; }
    __syncthreads();
    int nw = blockDim.x >> 5;
    if (warp == 0) {
        sum = (lane < nw) ? ws[0][lane] : 0.f;
        sq  = (lane < nw) ? ws[1][lane] : 0.f;
        #pragma unroll
        for (int off = 16; off > 0; off >>= 1) {
            sum += __shfl_down_sync(mask, sum, off);
            sq  += __shfl_down_sync(mask, sq,  off);
        }
        if (lane == 0) {
            partial[(long long)bg * S + s]                 = sum;
            partial[(long long)BG * S + (long long)bg * S + s] = sq;
        }
    }
}

__global__ void gn_finalize_kernel(const float* __restrict__ partial,
                                   float* __restrict__ mean,
                                   float* __restrict__ rstd,
                                   int S, int BG, float inv_count, float eps) {
    const int bg = blockIdx.x;
    float sum = 0.f, sq = 0.f;
    for (int i = threadIdx.x; i < S; i += blockDim.x) {
        sum += partial[(long long)bg * S + i];
        sq  += partial[(long long)BG * S + (long long)bg * S + i];
    }
    __shared__ float ws[2][32];
    unsigned mask = 0xffffffffu;
    #pragma unroll
    for (int off = 16; off > 0; off >>= 1) {
        sum += __shfl_down_sync(mask, sum, off);
        sq  += __shfl_down_sync(mask, sq,  off);
    }
    int lane = threadIdx.x & 31, warp = threadIdx.x >> 5;
    if (lane == 0) { ws[0][warp] = sum; ws[1][warp] = sq; }
    __syncthreads();
    int nw = blockDim.x >> 5;
    if (warp == 0) {
        sum = (lane < nw) ? ws[0][lane] : 0.f;
        sq  = (lane < nw) ? ws[1][lane] : 0.f;
        #pragma unroll
        for (int off = 16; off > 0; off >>= 1) {
            sum += __shfl_down_sync(mask, sum, off);
            sq  += __shfl_down_sync(mask, sq,  off);
        }
        if (lane == 0) {
            float m = sum * inv_count;
            float v = sq * inv_count - m * m;
            v = v < 0.f ? 0.f : v;
            mean[bg] = m;
            rstd[bg] = rsqrtf(v + eps);
        }
    }
}

// ------------------------------------------- GN(affine) + SiLU, NHWC -> NHWC
__global__ void gn_silu_nhwc_kernel(const float* __restrict__ x,
                                    const float* __restrict__ gamma,
                                    const float* __restrict__ beta,
                                    const float* __restrict__ mean,
                                    const float* __restrict__ rstd,
                                    float* __restrict__ y,
                                    int HWC4, int C4, int CPG, int G) {
    const int n = blockIdx.y;
    int idx = blockIdx.x * blockDim.x + threadIdx.x;
    if (idx >= HWC4) return;
    const int c0 = (idx % C4) << 2;
    const int g  = c0 / CPG;
    const int bg = n * G + g;
    const float mu = mean[bg], rs = rstd[bg];

    const long long off = (long long)n * HWC4 + idx;
    const float4 v  = *reinterpret_cast<const float4*>(x + (off << 2));
    const float4 gm = *reinterpret_cast<const float4*>(gamma + c0);
    const float4 bt = *reinterpret_cast<const float4*>(beta + c0);
    float4 o;
    o.x = silu_f((v.x - mu) * rs * gm.x + bt.x);
    o.y = silu_f((v.y - mu) * rs * gm.y + bt.y);
    o.z = silu_f((v.z - mu) * rs * gm.z + bt.z);
    o.w = silu_f((v.w - mu) * rs * gm.w + bt.w);
    *reinterpret_cast<float4*>(y + (off << 2)) = o;
}

// ---------------- GN(affine) + SiLU + residual add, NHWC in -> NCHW out
__global__ void gn_silu_res_nchw_kernel(const float* __restrict__ x,
                                        const float* __restrict__ res,
                                        const float* __restrict__ gamma,
                                        const float* __restrict__ beta,
                                        const float* __restrict__ mean,
                                        const float* __restrict__ rstd,
                                        float* __restrict__ out,
                                        int HW, int C, int G, int CPG) {
    extern __shared__ float sm[];
    const int bg = blockIdx.y;
    const int n  = bg / G;
    const int g  = bg - n * G;
    const int p0 = blockIdx.x * TP;
    const float mu = mean[bg], rs = rstd[bg];

    const long long base = (long long)n * HW * (long long)C + (long long)g * CPG;
    const int total = CPG * TP;
    for (int i = threadIdx.x; i < total; i += blockDim.x) {
        int c = i % CPG;
        int p = i / CPG;
        int pg = p0 + p;
        if (pg < HW) {
            float v = x[base + (long long)pg * C + c];
            v = (v - mu) * rs * gamma[g * CPG + c] + beta[g * CPG + c];
            sm[c * (TP + 1) + p] = silu_f(v);
        }
    }
    __syncthreads();
    const long long obase = (long long)n * C * (long long)HW + (long long)g * CPG * HW;
    for (int i = threadIdx.x; i < total; i += blockDim.x) {
        int p = i % TP;
        int c = i / TP;
        int pg = p0 + p;
        if (pg < HW) {
            long long oi = obase + (long long)c * HW + pg;
            out[oi] = sm[c * (TP + 1) + p] + res[oi];
        }
    }
}

// ------------------------------------------------------------------- launchers
static void compute_stats(const at::Tensor& x, int B, int HW, int C, int G,
                          float eps, at::Tensor& mean, at::Tensor& rstd) {
    const int CPG = C / G;
    const int BG = B * G;
    int S = (456 + BG - 1) / BG;
    int maxS = (HW + 255) / 256;
    if (S > maxS) S = maxS;
    if (S < 1) S = 1;
    auto opts = x.options();
    auto partial = at::empty({2, BG, S}, opts);
    auto stream = at::cuda::getCurrentCUDAStream();

    dim3 grid(S, BG);
    gn_stats_kernel<<<grid, 256, 0, stream>>>(
        x.data_ptr<float>(), partial.data_ptr<float>(), HW, C, G, CPG, S, BG);
    C10_CUDA_KERNEL_LAUNCH_CHECK();

    float inv_count = 1.0f / (float)((long long)HW * CPG);
    gn_finalize_kernel<<<BG, 256, 0, stream>>>(
        partial.data_ptr<float>(), mean.data_ptr<float>(), rstd.data_ptr<float>(),
        S, BG, inv_count, eps);
    C10_CUDA_KERNEL_LAUNCH_CHECK();
}

torch::Tensor gn_silu_nhwc(torch::Tensor x, torch::Tensor gamma, torch::Tensor beta,
                           int64_t B, int64_t HW, int64_t C, int64_t G, double eps) {
    TORCH_CHECK(x.scalar_type() == at::kFloat, "float32 only");
    auto opts = x.options();
    auto mean = at::empty({B * G}, opts);
    auto rstd = at::empty({B * G}, opts);
    compute_stats(x, (int)B, (int)HW, (int)C, (int)G, (float)eps, mean, rstd);

    auto y = at::empty_like(x);
    const int C4 = (int)C / 4;
    const int HWC4 = (int)HW * C4;
    auto stream = at::cuda::getCurrentCUDAStream();
    dim3 grid((HWC4 + 255) / 256, (unsigned)B);
    gn_silu_nhwc_kernel<<<grid, 256, 0, stream>>>(
        x.data_ptr<float>(), gamma.data_ptr<float>(), beta.data_ptr<float>(),
        mean.data_ptr<float>(), rstd.data_ptr<float>(), y.data_ptr<float>(),
        HWC4, C4, (int)(C / G), (int)G);
    C10_CUDA_KERNEL_LAUNCH_CHECK();
    return y;
}
```

Includes library kernels (cuDNN/cuBLAS/PyTorch) that NCU does not profile.

so this table is the sole view of how the forward pass actually divides its
time. Read the percentages before choosing a target: optimizing a kernel
that holds 4% of the time caps the whole round at 4%, however well it goes.
Look also for kernels that do no math — layout transforms, copies, casts.

READ 'steady %', NOT 'raw %'. Rows marked (one-time) are cuDNN autotune
probes and similar setup work: they run a handful of times regardless of
the workload and can swamp the raw column. 'steady %' re-bases on the
kernels that actually run every forward.

ATTRIBUTE SATELLITE KERNELS TO THE OP THEY SERVE. A vendor conv/GEMM is
not just its mainloop: cuDNN's TF32 path also launches a separate
convertTensor (fp32->TF32) pass, and layout transforms / im2col / casts
belong to the op they surround. Judging 'the conv is at 91% of TF32 peak,
so it has no headroom' from the MAINLOOP ALONE is a mistake — add the
satellite kernels in first. Removing a satellite needs no faster math:
owning the op (e.g. via CUTLASS) can delete it outright.

  Intent: Cache forward execution sequence with torch.cuda.CUDAGraph and pin memory addresses using static CUDA tensors (static inputs/outputs) to accelerate replay.

  Requirements: multi_kernel_forward, static_shapes_assumed_in_benchmark, allocate_static_buffers, warmup_before_capture, capture_once_replay_many
  Reference implementation / instance knowledge:
    ```
    class ModelNew(nn.Module):
        def __init__(self, input_size, hidden_size, num_layers=3, bias=True, batch_first=False):
            super().__init__()
            self.gru = nn.GRU(...)  # parameter holder / reference computation
            # CUDA Graph state
            self._graph_captured = False
            self._graph = None
            self._graph_stream = torch.cuda.Stream()
            # Static buffers for replay (addresses fixed)
            self._static_x = None
            self._static_h0 = None
            self._static_out = None
    
        def _capture_graph(self, x, h0):
            # Static inputs with fixed addresses
            self._static_x = torch.empty_like(x).contiguous()
            self._static_h0 = torch.empty_like(h0).contiguous()
            # Warmup to trigger internal algo selection / workspace allocs
            out0, _ = self.gru(self._static_x, self._static_h0)
            self._static_out = torch.empty_like(out0).contiguous()
            torch.cuda.synchronize()
            with torch.cuda.stream(self._graph_stream):
                self._graph = torch.cuda.CUDAGraph()
                self._static_x.copy_(x)
                self._static_h0.copy_(h0)
                with torch.cuda.graph(self._graph):
                    out, _ = self.gru(self._static_x, self._static_h0)
                    self._static_out.copy_(out)
            torch.cuda.synchronize()
            self._graph_captured = True
    
        def forward(self, x, h0):
            # Keep exact reference behavior in training
            if self.training or (not x.is_cuda):
                out, _ = self.gru(x, h0)
                return out
            x = x.contiguous()
            h0 = h0.contiguous()
            # Re-capture if shape/dtype/device changes
            if (not self._graph_captured) or (self._static_x is None) or \
               (x.shape != self._static_x.shape) or (h0.shape != self._static_h0.shape) or \
               (x.dtype != self._static_x.dtype) or (h0.dtype != self._static_h0.dtype) or \
               (x.device != self._static_x.device) or (h0.device != self._static_h0.device):
                self._graph_captured = False
                self._capture_graph(x, h0)
            self._static_x.copy_(x)
            self._static_h0.copy_(h0)
            self._graph.replay()
            return self._static_out
    ```
