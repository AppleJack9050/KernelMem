#include <torch/extension.h>
#include <ATen/ATen.h>
#include <ATen/cuda/CUDAContext.h>
#include <c10/cuda/CUDAException.h>
#include <cuda_runtime.h>

#define NGROUPS 32

__device__ __forceinline__ void warp_red2(float &a, float &b) {
#pragma unroll
    for (int off = 16; off > 0; off >>= 1) {
        a += __shfl_down_sync(0xffffffff, a, off);
        b += __shfl_down_sync(0xffffffff, b, off);
    }
}

// ---------------------------------------------------------------------
// Original NCHW path (kept verbatim as fallback: non-float, dim!=4,
// C % NGROUPS != 0, or channels_last-but-C!=256 where the specialised
// NHWC kernels below do not apply -- see TODO near nhwc_ok).
// ---------------------------------------------------------------------
template <int BS>
__global__ void gn_reduce_kernel(const float4 *__restrict__ x,
                                 float2 *__restrict__ part,
                                 long Lv, long chunk) {
    const int seg = blockIdx.x;
    const int row = blockIdx.y;
    const long base = (long)row * Lv;
    long start = (long)seg * chunk;
    long end = start + chunk;
    if (end > Lv) end = Lv;

    float s = 0.f, q = 0.f;
    for (long i = start + threadIdx.x; i < end; i += BS) {
        float4 v = x[base + i];
        s += v.x + v.y + v.z + v.w;
        q += v.x * v.x + v.y * v.y + v.z * v.z + v.w * v.w;
    }
    __shared__ float ss[BS / 32], sq[BS / 32];
    const int lane = threadIdx.x & 31;
    const int wid = threadIdx.x >> 5;
    warp_red2(s, q);
    if (lane == 0) { ss[wid] = s; sq[wid] = q; }
    __syncthreads();
    if (wid == 0) {
        s = (lane < (BS / 32)) ? ss[lane] : 0.f;
        q = (lane < (BS / 32)) ? sq[lane] : 0.f;
        warp_red2(s, q);
        if (lane == 0) part[(long)row * gridDim.x + seg] = make_float2(s, q);
    }
}

__device__ __forceinline__ float2 block_stats(const float2 *__restrict__ part,
                                              int grow, int segs,
                                              double invN, double eps) {
    __shared__ float2 sh;
    if (threadIdx.x < 32) {
        double s = 0.0, q = 0.0;
        for (int i = threadIdx.x; i < segs; i += 32) {
            float2 p = part[(long)grow * segs + i];
            s += (double)p.x;
            q += (double)p.y;
        }
#pragma unroll
        for (int off = 16; off > 0; off >>= 1) {
            s += __shfl_down_sync(0xffffffff, s, off);
            q += __shfl_down_sync(0xffffffff, q, off);
        }
        if (threadIdx.x == 0) {
            double mean = s * invN;
            double var = q * invN - mean * mean;
            if (var < 0.0) var = 0.0;
            sh = make_float2((float)mean, (float)(1.0 / sqrt(var + eps)));
        }
    }
    __syncthreads();
    return sh;
}

__device__ __forceinline__ float silu_f(float v) {
    return v / (1.f + __expf(-v));
}

template <int BS, bool RESID>
__global__ void gn_apply_vec_kernel(const float4 *__restrict__ x,
                                    const float4 *__restrict__ res,
                                    float4 *__restrict__ y,
                                    const float2 *__restrict__ part,
                                    const float *__restrict__ gamma,
                                    const float *__restrict__ beta,
                                    long HWv, long chunk, int C, int cpg,
                                    int segs, double invN, double eps) {
    const int seg = blockIdx.x;
    const int row = blockIdx.y;
    const int c = row % C;
    const int n = row / C;
    const float2 st = block_stats(part, n * NGROUPS + c / cpg, segs, invN, eps);
    const float scale = st.y * gamma[c];
    const float shift = beta[c] - st.x * scale;

    const long base = (long)row * HWv;
    long start = (long)seg * chunk;
    long end = start + chunk;
    if (end > HWv) end = HWv;

    for (long i = start + threadIdx.x; i < end; i += BS) {
        float4 v = x[base + i];
        float4 o;
        o.x = silu_f(v.x * scale + shift);
        o.y = silu_f(v.y * scale + shift);
        o.z = silu_f(v.z * scale + shift);
        o.w = silu_f(v.w * scale + shift);
        if (RESID) {
            float4 r = res[base + i];
            o.x += r.x; o.y += r.y; o.z += r.z; o.w += r.w;
        }
        y[base + i] = o;
    }
}

template <int BS, bool RESID>
__global__ void gn_apply_sca_kernel(const float *__restrict__ x,
                                    const float *__restrict__ res,
                                    float *__restrict__ y,
                                    const float2 *__restrict__ part,
                                    const float *__restrict__ gamma,
                                    const float *__restrict__ beta,
                                    long HW, long chunk, int C, int cpg,
                                    int segs, double invN, double eps) {
    const int seg = blockIdx.x;
    const int row = blockIdx.y;
    const int c = row % C;
    const int n = row / C;
    const float2 st = block_stats(part, n * NGROUPS + c / cpg, segs, invN, eps);
    const float scale = st.y * gamma[c];
    const float shift = beta[c] - st.x * scale;

    const long base = (long)row * HW;
    long start = (long)seg * chunk;
    long end = start + chunk;
    if (end > HW) end = HW;

    for (long i = start + threadIdx.x; i < end; i += BS) {
        float o = silu_f(x[base + i] * scale + shift);
        if (RESID) o += res[base + i];
        y[base + i] = o;
    }
}

static void gn_silu(at::Tensor &t, const at::Tensor &gamma, const at::Tensor &beta,
                    const at::Tensor *residual, double eps) {
    const int B = (int)t.size(0), C = (int)t.size(1);
    const long HW = (long)t.size(2) * (long)t.size(3);
    const int cpg = C / NGROUPS;
    const long L = (long)cpg * HW;
    const long Lv = L / 4;
    auto stream = at::cuda::getCurrentCUDAStream();

    const int rows_g = B * NGROUPS;
    long chunk = 4096;
    int segs = (int)((Lv + chunk - 1) / chunk);
    if (segs < 1) segs = 1;
    while ((long)segs * rows_g < 512 && chunk > 256) {
        chunk >>= 1;
        segs = (int)((Lv + chunk - 1) / chunk);
    }

    auto part = at::empty({(long)rows_g * segs, 2}, t.options());

    constexpr int BS = 256;
    dim3 grid1(segs, rows_g);
    gn_reduce_kernel<BS><<<grid1, BS, 0, stream>>>(
        (const float4 *)t.data_ptr<float>(), (float2 *)part.data_ptr<float>(), Lv, chunk);
    C10_CUDA_KERNEL_LAUNCH_CHECK();

    const double invN = 1.0 / (double)L;
    const int rows_c = B * C;
    if ((HW & 3L) == 0) {
        const long HWv = HW / 4;
        long chunk2 = 2048;
        int segs2 = (int)((HWv + chunk2 - 1) / chunk2);
        if (segs2 < 1) segs2 = 1;

            chunk2 >>= 1;
            segs2 = (int)((HWv + chunk2 - 1) / chunk2);

        dim3 grid2(segs2, rows_c);
        if (residual) {
            gn_apply_vec_kernel<BS, true><<<grid2, BS, 0, stream>>>(
                (const float4 *)t.data_ptr<float>(),
                (const float4 *)residual->data_ptr<float>(),

                gamma.data_ptr<float>(), beta.data_ptr<float>(),
                HWv, chunk2, C, cpg, segs, invN, eps);
        } else {
            gn_apply_vec_kernel<BS, false><<<grid2, BS, 0, stream>>>(
                (const float4 *)t.data_ptr<float>(), nullptr,
                (float4 *)t.data_ptr<float>(), (const float2 *)part.data_ptr<float>(),
                gamma.data_ptr<float>(), beta.data_ptr<float>(),
                HWv, chunk2, C, cpg, segs, invN, eps);
        }
        C10_CUDA_KERNEL_LAUNCH_CHECK();
    } else {
        long chunk2 = 8192;
        int segs2 = (int)((HW + chunk2 - 1) / chunk2);
        if (segs2 < 1) segs2 = 1;
        while ((long)segs2 * rows_c < 512 && chunk2 > 512) {
            chunk2 >>= 1;
            segs2 = (int)((HW + chunk2 - 1) / chunk2);
        }
        dim3 grid2(segs2, rows_c);
        if (residual) {
            gn_apply_sca_kernel<BS, true><<<grid2, BS, 0, stream>>>(
                t.data_ptr<float>(), residual->data_ptr<float>(), t.data_ptr<float>(),
                (const float2 *)part.data_ptr<float>(), gamma.data_ptr<float>(),
                beta.data_ptr<float>(), HW, chunk2, C, cpg, segs, invN, eps);

                beta.data_ptr<float>(), HW, chunk2, C, cpg, segs, invN, eps);
        }
        C10_CUDA_KERNEL_LAUNCH_CHECK();
    }
}

// ---------------------------------------------------------------------
// NHWC (channels_last) path -- specialised for C == 256, NGROUPS == 32,
// cpg == 8 (matching the modification plan literally: QPP = C/4 = 64
// quads per pixel, group = quad_in_pixel >> 1, 2 quads per group so a
// single __shfl_down_sync(...,1) merges the pair before the atomicAdd).
// Block size is fixed at 256 so that 256 % QPP == 0, which is what makes
// the per-thread group id constant across the whole strided loop.
// TODO: generalise to other channel counts (currently falls back to the
// NCHW path above when C != 256).
// ---------------------------------------------------------------------
__global__ void gn_reduce_nhwc_kernel(const float4 *__restrict__ x,
                                      float2 *__restrict__ part,
                                      long HW, long chunk) {
    const int seg = blockIdx.x;
    const int n = blockIdx.y;
    const long QPP = 64;                 // C/4
    const long Lq = HW * QPP;
    const long base = (long)n * Lq;
    long start = (long)seg * chunk;
    long end = start + chunk;
    if (end > Lq) end = Lq;

    // constant per thread: 256 % QPP(=64) == 0, and chunk is always even
    // (power of two), so start is even and (start+tid) % QPP is fixed.
    const int g = (int)(((start + threadIdx.x) % QPP) >> 1);

    float s = 0.f, q = 0.f;
    for (long i = start + threadIdx.x; i < end; i += blockDim.x) {
        float4 v = x[base + i];
        s += v.x + v.y + v.z + v.w;
        q += v.x * v.x + v.y * v.y + v.z * v.z + v.w * v.w;
    }

    __shared__ float ssum[32], ssq[32];
    if (threadIdx.x < 32) { ssum[threadIdx.x] = 0.f; ssq[threadIdx.x] = 0.f; }
    __syncthreads();

    float sp = __shfl_down_sync(0xffffffff, s, 1);
    float qp = __shfl_down_sync(0xffffffff, q, 1);
    if ((threadIdx.x & 1) == 0) {
        s += sp; q += qp;
        atomicAdd(&ssum[g], s);
        atomicAdd(&ssq[g], q);
    }
    __syncthreads();

    if (threadIdx.x < 32) {
        part[(long)(n * 32 + threadIdx.x) * gridDim.x + seg] = make_float2(ssum[threadIdx.x], ssq[threadIdx.x]);
    }
}

template <bool RESID>
__global__ void gn_apply_nhwc_kernel(const float4 *__restrict__ x,
                                     const float4 *__restrict__ res,
                                     float4 *__restrict__ y,
                                     const float2 *__restrict__ part,
                                     const float *__restrict__ gamma,
                                     const float *__restrict__ beta,
                                     long HW, long chunk, int segsR,
                                     double invN, double eps) {
    const int seg = blockIdx.x;
    const int n = blockIdx.y;

    __shared__ float mean_s[32], rstd_s[32];
    if (threadIdx.x < 32) {
        int g = threadIdx.x;
        double s = 0.0, q = 0.0;
        for (int i = 0; i < segsR; i++) {
            float2 p = part[(long)(n * 32 + g) * segsR + i];
            s += (double)p.x;
            q += (double)p.y;
        }
        double mean = s * invN;
        double var = q * invN - mean * mean;
        if (var < 0.0) var = 0.0;
        mean_s[g] = (float)mean;
        rstd_s[g] = (float)(1.0 / sqrt(var + eps));
    }
    __syncthreads();

    __shared__ float sc[256], sh[256];
    for (int c = threadIdx.x; c < 256; c += blockDim.x) {
        int g = c >> 3;               // cpg = 8
        float scale = rstd_s[g] * gamma[c];
        sc[c] = scale;
        sh[c] = beta[c] - mean_s[g] * scale;
    }
    __syncthreads();

    const long QPP = 64;
    const long Lq = HW * QPP;
    const long base = (long)n * Lq;
    long start = (long)seg * chunk;
    long end = start + chunk;
    if (end > Lq) end = Lq;

    const float4 *sc4 = (const float4 *)sc;
    const float4 *sh4 = (const float4 *)sh;

    for (long i = start + threadIdx.x; i < end; i += blockDim.x) {
        float4 v = x[base + i];
        int q = (int)(i % QPP);
        float4 s4 = sc4[q];
        float4 h4 = sh4[q];
        float4 o;
        o.x = silu_f(v.x * s4.x + h4.x);
        o.y = silu_f(v.y * s4.y + h4.y);
        o.z = silu_f(v.z * s4.z + h4.z);
        o.w = silu_f(v.w * s4.w + h4.w);
        if (RESID) {
            float4 r = res[base + i];
            o.x += r.x; o.y += r.y; o.z += r.z; o.w += r.w;
        }
        y[base + i] = o;
    }
}

static void gn_silu_nhwc(at::Tensor &t, const at::Tensor &gamma, const at::Tensor &beta,
                         const at::Tensor *residual, double eps) {
    const int B = (int)t.size(0);
    const long HW = (long)t.size(2) * (long)t.size(3);
    const long QPP = 64;
    const long Lq = HW * QPP;
    auto stream = at::cuda::getCurrentCUDAStream();
    constexpr int BS = 256;

    long chunk = 4096;
    int segsR = (int)((Lq + chunk - 1) / chunk);
    if (segsR < 1) segsR = 1;
    while ((long)segsR * B < 1024 && chunk > 512) {
        chunk >>= 1;
        segsR = (int)((Lq + chunk - 1) / chunk);
    }

    auto part = at::empty({(long)B * NGROUPS * segsR, 2}, t.options());

    dim3 grid1(segsR, B);
    gn_reduce_nhwc_kernel<<<grid1, BS, 0, stream>>>(
        (const float4 *)t.data_ptr<float>(), (float2 *)part.data_ptr<float>(), HW, chunk);
    C10_CUDA_KERNEL_LAUNCH_CHECK();

    const double invN = 1.0 / (double)(8 * HW);  // cpg = 8

    long chunk2 = 4096;
    int segs2 = (int)((Lq + chunk2 - 1) / chunk2);
    if (segs2 < 1) segs2 = 1;
    while ((long)segs2 * B < 1024 && chunk2 > 512) {
        chunk2 >>= 1;
        segs2 = (int)((Lq + chunk2 - 1) / chunk2);
    }
    dim3 grid2(segs2, B);
    if (residual) {
        gn_apply_nhwc_kernel<true><<<grid2, BS, 0, stream>>>(
            (const float4 *)t.data_ptr<float>(), (const float4 *)residual->data_ptr<float>(),
            (float4 *)t.data_ptr<float>(), (const float2 *)part.data_ptr<float>(),
            gamma.data_ptr<float>(), beta.data_ptr<float>(), HW, chunk2, segsR, invN, eps);
    } else {
        gn_apply_nhwc_kernel<false><<<grid2, BS, 0, stream>>>(
            (const float4 *)t.data_ptr<float>(), nullptr,
            (float4 *)t.data_ptr<float>(), (const float2 *)part.data_ptr<float>(),
            gamma.data_ptr<float>(), beta.data_ptr<float>(), HW, chunk2, segsR, invN, eps);
    }
    C10_CUDA_KERNEL_LAUNCH_CHECK();
}

torch::Tensor residual_block(torch::Tensor x, torch::Tensor w1, torch::Tensor g1,
                             torch::Tensor b1, torch::Tensor w2, torch::Tensor g2,
                             torch::Tensor b2, double eps) {
    at::NoGradGuard ng;
    TORCH_CHECK(x.is_cuda(), "input must be a CUDA tensor");

    // Plan item 1/2: channels_last path, specialised for C == 256 (see
    // TODO above the NHWC kernels). Falls back to the original NCHW
    // kernels otherwise.
    const bool nhwc_ok = (x.scalar_type() == at::kFloat) && (x.dim() == 4) &&
                         (x.size(1) == 256) && (x.size(1) % NGROUPS == 0);

    if (nhwc_ok) {
        auto xc = x.contiguous(at::MemoryFormat::ChannelsLast);
        auto w1c = w1.contiguous(at::MemoryFormat::ChannelsLast);
        auto w2c = w2.contiguous(at::MemoryFormat::ChannelsLast);

        at::Tensor t = at::conv2d(xc, w1c, {}, {1, 1}, {1, 1});
        if (!t.is_contiguous(at::MemoryFormat::ChannelsLast))
            t = t.contiguous(at::MemoryFormat::ChannelsLast);
        gn_silu_nhwc(t, g1, b1, nullptr, eps);

        at::Tensor u = at::conv2d(t, w2c, {}, {1, 1}, {1, 1});
        if (!u.is_contiguous(at::MemoryFormat::ChannelsLast))
            u = u.contiguous(at::MemoryFormat::ChannelsLast);
        gn_silu_nhwc(u, g2, b2, &xc, eps);
        return u;   // plan item 9: no trailing .contiguous()
    }

    auto xc = x.is_contiguous() ? x : x.contiguous();
    auto w1c = w1.is_contiguous() ? w1 : w1.contiguous();
    auto w2c = w2.is_contiguous() ? w2 : w2.contiguous();

    const bool fast = (x.scalar_type() == at::kFloat) && (x.dim() == 4) &&
                      (x.size(1) % NGROUPS == 0);

    at::Tensor t = at::conv2d(xc, w1c, {}, {1, 1}, {1, 1});
    if (!t.is_contiguous()) t = t.contiguous();
    if (fast) {
        gn_silu(t, g1, b1, nullptr, eps);
    } else {
        t = at::silu(at::group_norm(t, NGROUPS, g1, b1, eps, true));
    }

    at::Tensor u = at::conv2d(t, w2c, {}, {1, 1}, {1, 1});
    if (!u.is_contiguous()) u = u.contiguous();
    if (fast) {
        gn_silu(u, g2, b2, &xc, eps);
    } else {
        u = at::silu(at::group_norm(u, NGROUPS, g2, b2, eps, true)) + xc;
    }
    return u;
}
