# ==========================================================================
# ModelNew — fused VAE residual block
#   Conv3x3 -> GN -> SiLU -> Conv3x3 -> GN -> SiLU -> + x
#
# OPTIMISATION: gridsync_fused_groupnorm
#   Each GroupNorm's stats -> finalize -> apply triple is ONE persistent

import torch
import torch.nn as nn
import torch.nn.functional as F
from torch.utils.cpp_extension import load_inline

_CUDA_SRC = r"""
#include <torch/extension.h>
#include <ATen/ATen.h>
#include <ATen/cuda/CUDAContext.h>
#include <c10/cuda/CUDAException.h>
#include <cuda_runtime.h>
#include <cooperative_groups.h>
#include <algorithm>

namespace cg = cooperative_groups;

#define TILE 32
#define TBY  8
#define STAT_BS 256

__device__ __forceinline__ float silu_f(float z) {
    return z / (1.0f + __expf(-z));
}

// ------------------------------------------------------------------ grid barrier
// Under -rdc=true the cooperative_groups grid barrier is used directly; otherwise
// an equivalent arrive/wait counter barrier is used (same residency guarantee,
// since the kernel is always launched with cudaLaunchCooperativeKernel and a grid
// that fits the occupancy limit).
__device__ __forceinline__ void grid_barrier(unsigned int* bar,
                                             unsigned int nblocks,
                                             unsigned int gen) {
#if defined(__CUDACC_RDC__)
    (void)bar; (void)nblocks; (void)gen;
    cg::this_grid().sync();
#else
    __syncthreads();
    if (threadIdx.x == 0) {
        __threadfence();
        atomicAdd(bar, 1u);
        const unsigned int target = nblocks * (gen + 1u);
        while (atomicAdd(bar, 0u) < target) {
#if __CUDA_ARCH__ >= 700
            __nanosleep(48);
#endif
        }
        __threadfence();
    }
    __syncthreads();
#endif
}

// ---------------------------------------------------------------- K0
__global__ void nchw_to_nhwc_kernel(const float* __restrict__ in,
                                    float* __restrict__ out,
                                    int HW, int C) {
    __shared__ float tile[TILE][TILE + 1];
    const int n  = blockIdx.z;
    const int s0 = blockIdx.x * TILE;
    const int c0 = blockIdx.y * TILE;
    const int tx = threadIdx.x, ty = threadIdx.y;

    const float* pin = in + (size_t)n * (size_t)C * (size_t)HW;
    {
        int s = s0 + tx;
        #pragma unroll
        for (int k = 0; k < TILE / TBY; ++k) {
            int c = c0 + ty + k * TBY;
            float v = 0.0f;
            if (s < HW && c < C) v = pin[(size_t)c * (size_t)HW + (size_t)s];
            tile[ty + k * TBY][tx] = v;
        }
    }
    __syncthreads();
    {
        float* pout = out + (size_t)n * (size_t)HW * (size_t)C;
        int c = c0 + tx;
        #pragma unroll
        for (int k = 0; k < TILE / TBY; ++k) {
            int s = s0 + ty + k * TBY;
            if (s < HW && c < C)
                pout[(size_t)s * (size_t)C + (size_t)c] = tile[tx][ty + k * TBY];
        }
    }
}

// ================================================================= FUSED GN
// TRANSPOSE_OUT == false : stats + finalize + (affine+SiLU), NHWC -> NHWC
// TRANSPOSE_OUT == true  : stats + finalize + (affine+SiLU+residual), NHWC->NCHW
template<bool TRANSPOSE_OUT>
__global__ void gn_fused_kernel(const float* __restrict__ in,     // NHWC
                                const float* __restrict__ gamma,
                                const float* __restrict__ beta,
                                const float* __restrict__ res,    // NCHW (may be null)
                                float* __restrict__ out,
                                float* __restrict__ partial,
                                float* __restrict__ Ag,
                                float* __restrict__ Bg,
                                unsigned int* __restrict__ bar,
                                int N, int HW, int C, int G, int D,
                                int nblk, int chunk, float eps, float cnt) {
    extern __shared__ float shab[];          // 2*C floats : A,B staging
    float* As = shab;
    float* Bs = shab + C;

    __shared__ float  ssum[STAT_BS];
    __shared__ float  ssq [STAT_BS];
    __shared__ double dsa [STAT_BS];
    __shared__ double dsb [STAT_BS];
    __shared__ float  tile[TRANSPOSE_OUT ? TILE * (TILE + 1) : 1];

    const int tid = threadIdx.x;
    const int b   = blockIdx.x;
    const int tx  = tid & (TILE - 1);
    const int ty  = tid >> 5;

    int s0 = b * chunk;
    int s1 = s0 + chunk;
    if (s0 > HW) s0 = HW;
    if (s1 > HW) s1 = HW;

    const int ngrp = STAT_BS / D;
    unsigned int gen = 0u;

    for (int n = 0; n < N; ++n) {
        const float* __restrict__ p = in + (size_t)n * (size_t)HW * (size_t)C;

        // ------------------------------------------------ phase A : moments
        for (int cbase = 0; cbase < C; cbase += STAT_BS) {
            int c = cbase + tid;
            float a0 = 0.f, a1 = 0.f, a2 = 0.f, a3 = 0.f;
            float q0 = 0.f, q1 = 0.f, q2 = 0.f, q3 = 0.f;
            if (c < C && s0 < s1) {
                const float* q = p + (size_t)s0 * (size_t)C + (size_t)c;
                int si = s0;
                for (; si + 4 <= s1; si += 4) {
                    float v0 = q[0];
                    float v1 = q[(size_t)C];
                    float v2 = q[(size_t)2 * C];
                    float v3 = q[(size_t)3 * C];
                    a0 += v0; q0 += v0 * v0;
                    a1 += v1; q1 += v1 * v1;
                    a2 += v2; q2 += v2 * v2;
                    a3 += v3; q3 += v3 * v3;
                    q += (size_t)4 * C;
                }
                for (; si < s1; ++si) {
                    float v = *q;
                    a0 += v; q0 += v * v;
                    q += (size_t)C;
                }
            }
            ssum[tid] = (a0 + a1) + (a2 + a3);
            ssq [tid] = (q0 + q1) + (q2 + q3);
            __syncthreads();
            if (tid < ngrp) {
                float sa = 0.f, sq = 0.f;
                for (int d = 0; d < D; ++d) {
                    sa += ssum[tid * D + d];
                    sq += ssq [tid * D + d];
                }
                int g = cbase / D + tid;
                if (g < G) {
                    size_t o = ((size_t)b * (size_t)G + (size_t)g) * 2;
                    partial[o + 0] = sa;      // every slot written (no tail)
                    partial[o + 1] = sq;
                }
            }
            __syncthreads();
        }

        grid_barrier(bar, (unsigned int)nblk, gen++);

        // ------------------------------------------------ finalize : mean/rstd -> A,B
        for (int g = b; g < G; g += gridDim.x) {
            double s = 0.0, ss = 0.0;
            for (int bb = tid; bb < nblk; bb += STAT_BS) {
                const float* pp = partial + ((size_t)bb * (size_t)G + (size_t)g) * 2;
                s  += (double)__ldcg(pp + 0);
                ss += (double)__ldcg(pp + 1);
            }
            dsa[tid] = s;
            dsb[tid] = ss;
            __syncthreads();
            for (int off = STAT_BS >> 1; off > 0; off >>= 1) {
                if (tid < off) {
                    dsa[tid] += dsa[tid + off];
                    dsb[tid] += dsb[tid + off];
                }
                __syncthreads();
            }
            double mean = dsa[0] / (double)cnt;
            double var  = dsb[0] / (double)cnt - mean * mean;
            if (var < 0.0) var = 0.0;
            float rstd = (float)(1.0 / sqrt(var + (double)eps));
            float fm   = (float)mean;
            for (int d = tid; d < D; d += STAT_BS) {
                int c = g * D + d;
                float gg = gamma[c];
                Ag[c] = rstd * gg;
                Bg[c] = beta[c] - fm * rstd * gg;
            }
            __syncthreads();
        }

        grid_barrier(bar, (unsigned int)nblk, gen++);

        // ------------------------------------------------ phase B : apply (L2 re-read)
        for (int c = tid; c < C; c += STAT_BS) {
            As[c] = __ldcg(Ag + c);
            Bs[c] = __ldcg(Bg + c);
        }
        __syncthreads();

        if constexpr (!TRANSPOSE_OUT) {
            const int C4 = C >> 2;
            const float4* pin4  = (const float4*)p;
            float4*       pout4 = (float4*)(out + (size_t)n * (size_t)HW * (size_t)C);
            const int stepc = STAT_BS % C4;
            const int steps = STAT_BS / C4;
            int c4 = tid % C4;
            int s  = s0 + tid / C4;
            while (s < s1) {
                long idx = (long)s * (long)C4 + (long)c4;
                int c = c4 << 2;
                float4 v = pin4[idx];
                float4 o;
                o.x = silu_f(v.x * As[c + 0] + Bs[c + 0]);
                o.y = silu_f(v.y * As[c + 1] + Bs[c + 1]);
                o.z = silu_f(v.z * As[c + 2] + Bs[c + 2]);
                o.w = silu_f(v.w * As[c + 3] + Bs[c + 3]);
                pout4[idx] = o;
                c4 += stepc;
                s  += steps;
                if (c4 >= C4) { c4 -= C4; s += 1; }
            }
        } else {
            const size_t nbase = (size_t)n * (size_t)C * (size_t)HW;
            for (int st = s0; st < s1; st += TILE) {
                for (int c0 = 0; c0 < C; c0 += TILE) {
                    __syncthreads();
                    int c = c0 + tx;
                    float a = 0.f, bb2 = 0.f;
                    if (c < C) { a = As[c]; bb2 = Bs[c]; }
                    #pragma unroll
                    for (int k = 0; k < TILE / TBY; ++k) {
                        int s = st + ty + k * TBY;
                        float y = 0.f;
                        if (s < s1 && c < C) {
                            float v = p[(size_t)s * (size_t)C + (size_t)c];
                            y = silu_f(v * a + bb2);
                        }
                        tile[(ty + k * TBY) * (TILE + 1) + tx] = y;
                    }
                    __syncthreads();
                    int s = st + tx;
                    if (s < s1) {
                        #pragma unroll
                        for (int k = 0; k < TILE / TBY; ++k) {
                            int cc = c0 + ty + k * TBY;
                            if (cc < C) {
                                size_t off = nbase + (size_t)cc * (size_t)HW + (size_t)s;
                                out[off] = tile[tx * (TILE + 1) + ty + k * TBY] + res[off];
                            }
                        }
                    }
                }
            }
        }
        // no extra barrier: the next iteration's phase A only writes partial[],
        // which phase B never reads, and its first grid_barrier orders A/B writes.
    }
}

// ============================================ legacy 3-kernel path (fallback, unmodified)
__global__ void gn_stats_kernel(const float* __restrict__ in,
                                float* __restrict__ partial,
                                int HW, int C, int G, int D,
                                int nblk, int chunk) {
    __shared__ float ssum[STAT_BS];
    __shared__ float ssq [STAT_BS];

    const int n = blockIdx.y;
    const int b = blockIdx.x;
    const int s0 = b * chunk;
    int s1 = s0 + chunk;
    if (s1 > HW) s1 = HW;

    const float* p = in + (size_t)n * (size_t)HW * (size_t)C;
    const int tid = threadIdx.x;

    for (int cbase = 0; cbase < C; cbase += STAT_BS) {
        int c = cbase + tid;
        float a0 = 0.f, a1 = 0.f, a2 = 0.f, a3 = 0.f;
        float q0 = 0.f, q1 = 0.f, q2 = 0.f, q3 = 0.f;
        if (c < C && s0 < s1) {
            const float* q = p + (size_t)s0 * (size_t)C + (size_t)c;
            int si = s0;
            for (; si + 4 <= s1; si += 4) {
                float v0 = q[0];
                float v1 = q[(size_t)C];
                float v2 = q[(size_t)2 * C];
                float v3 = q[(size_t)3 * C];
                a0 += v0; q0 += v0 * v0;
                a1 += v1; q1 += v1 * v1;
                a2 += v2; q2 += v2 * v2;
                a3 += v3; q3 += v3 * v3;
                q += (size_t)4 * C;
            }
            for (; si < s1; ++si) {
                float v = *q;
                a0 += v; q0 += v * v;
                q += (size_t)C;
            }
        }
        ssum[tid] = (a0 + a1) + (a2 + a3);
        ssq [tid] = (q0 + q1) + (q2 + q3);
        __syncthreads();

        int ngrp = STAT_BS / D;
        if (tid < ngrp) {
            float sa = 0.f, sq = 0.f;
            for (int d = 0; d < D; ++d) {
                sa += ssum[tid * D + d];
                sq += ssq [tid * D + d];
            }
            int g = cbase / D + tid;
            if (g < G) {
                size_t o = ((size_t)((size_t)n * nblk + b) * G + g) * 2;
                partial[o + 0] = sa;
                partial[o + 1] = sq;
            }
        }
        __syncthreads();
    }
}

__global__ void gn_finalize_kernel(const float* __restrict__ partial,
                                   const float* __restrict__ gamma,
                                   const float* __restrict__ beta,
                                   float* __restrict__ A,
                                   float* __restrict__ B,
                                   int nblk, int G, int C, int D,
                                   float eps, float cnt) {
    __shared__ double smA[256];
    __shared__ double smB[256];

    const int idx = blockIdx.x;
    const int n = idx / G;
    const int g = idx - n * G;

    const float* p = partial + ((size_t)n * nblk * G + g) * 2;
    double s = 0.0, ss = 0.0;
    for (int b = threadIdx.x; b < nblk; b += blockDim.x) {
        s  += (double)p[(size_t)b * G * 2 + 0];
        ss += (double)p[(size_t)b * G * 2 + 1];
    }
    smA[threadIdx.x] = s;
    smB[threadIdx.x] = ss;
    __syncthreads();
    for (int off = blockDim.x >> 1; off > 0; off >>= 1) {
        if ((int)threadIdx.x < off) {
            smA[threadIdx.x] += smA[threadIdx.x + off];
            smB[threadIdx.x] += smB[threadIdx.x + off];
        }
        __syncthreads();
    }
    double mean = smA[0] / (double)cnt;
    double var  = smB[0] / (double)cnt - mean * mean;
    if (var < 0.0) var = 0.0;
    float rstd = (float)(1.0 / sqrt(var + (double)eps));
    float fm   = (float)mean;

    for (int d = threadIdx.x; d < D; d += blockDim.x) {
        int c = g * D + d;
        float gg = gamma[c];
        A[(size_t)n * C + c] = rstd * gg;
        B[(size_t)n * C + c] = beta[c] - fm * rstd * gg;
    }
}

__global__ void gn_apply_nhwc_kernel(const float* __restrict__ in,
                                     const float* __restrict__ A,
                                     const float* __restrict__ B,
                                     float* __restrict__ out,
                                     long total4, int C4, int C) {
    extern __shared__ float sh[];
    float* As = sh;
    float* Bs = sh + C;

    const int n = blockIdx.y;
    for (int c = threadIdx.x; c < C; c += blockDim.x) {
        As[c] = A[(size_t)n * C + c];
        Bs[c] = B[(size_t)n * C + c];
    }
    __syncthreads();

    long i = (long)blockIdx.x * blockDim.x + threadIdx.x;
    if (i >= total4) return;

    int c4 = (int)(i - (i / (long)C4) * (long)C4);
    int c  = c4 * 4;

    const float4* pin  = (const float4*)(in  + (size_t)n * (size_t)total4 * 4);
    float4*       pout = (float4*)      (out + (size_t)n * (size_t)total4 * 4);

    float4 v = pin[i];
    float4 o;
    o.x = silu_f(v.x * As[c + 0] + Bs[c + 0]);
    o.y = silu_f(v.y * As[c + 1] + Bs[c + 1]);
    o.z = silu_f(v.z * As[c + 2] + Bs[c + 2]);
    o.w = silu_f(v.w * As[c + 3] + Bs[c + 3]);
    pout[i] = o;
}

__global__ void gn_apply_transpose_kernel(const float* __restrict__ in,
                                          const float* __restrict__ A,
                                          const float* __restrict__ B,
                                          const float* __restrict__ res,
                                          float* __restrict__ out,
                                          int HW, int C) {
    __shared__ float tile[TILE][TILE + 1];
    const int n  = blockIdx.z;
    const int s0 = blockIdx.x * TILE;
    const int c0 = blockIdx.y * TILE;
    const int tx = threadIdx.x, ty = threadIdx.y;

    const float* pin = in + (size_t)n * (size_t)HW * (size_t)C;
    int c = c0 + tx;
    float a = 0.f, b = 0.f;
    if (c < C) {
        a = A[(size_t)n * C + c];
        b = B[(size_t)n * C + c];
    }
    #pragma unroll
    for (int k = 0; k < TILE / TBY; ++k) {
        int s = s0 + ty + k * TBY;
        float y = 0.f;
        if (s < HW && c < C) {
            float v = pin[(size_t)s * (size_t)C + (size_t)c];
            y = silu_f(v * a + b);
        }
        tile[ty + k * TBY][tx] = y;
    }
    __syncthreads();

    const size_t nbase = (size_t)n * (size_t)C * (size_t)HW;
    int s = s0 + tx;
    #pragma unroll
    for (int k = 0; k < TILE / TBY; ++k) {
        int cc = c0 + ty + k * TBY;
        if (s < HW && cc < C) {
            size_t off = nbase + (size_t)cc * (size_t)HW + (size_t)s;
            out[off] = tile[tx][ty + k * TBY] + res[off];
        }
    }
}

// ---------------------------------------------------------------- host helpers
static void run_gn_stats(const float* src_nhwc,
                         const float* gamma, const float* beta,
                         float* A, float* B, float* partial,
                         int64_t N, int64_t C, int64_t HW,
                         int G, int D, int nblk, int chunk,
                         float eps, cudaStream_t stream) {
    dim3 gs((unsigned)nblk, (unsigned)N, 1);
    gn_stats_kernel<<<gs, STAT_BS, 0, stream>>>(
        src_nhwc, partial, (int)HW, (int)C, G, D, nblk, chunk);
    C10_CUDA_KERNEL_LAUNCH_CHECK();

    gn_finalize_kernel<<<(unsigned)(N * G), 256, 0, stream>>>(
        partial, gamma, beta, A, B,
        nblk, G, (int)C, D, eps, (float)((double)HW * (double)D));
    C10_CUDA_KERNEL_LAUNCH_CHECK();
}

// legacy (fallback) full GroupNorm + apply
static void legacy_gn(const float* src, const float* gamma, const float* beta,
                      const float* res, float* outp, bool transpose,
                      int64_t N, int64_t C, int64_t HW, int G, int D,
                      float eps, const at::TensorOptions& opts,
                      cudaStream_t stream) {
    int64_t per_img = std::max<int64_t>(1, 680 / std::max<int64_t>(1, N));
    int nblk_init = (int)std::max<int64_t>(1, std::min<int64_t>((HW + 63) / 64, per_img));
    int chunk = (int)((HW + nblk_init - 1) / nblk_init);
    if (chunk < 1) chunk = 1;
    int nblk = (int)((HW + chunk - 1) / chunk);

    auto partial = at::empty({N, (int64_t)nblk, (int64_t)G, 2}, opts);
    auto A = at::empty({N, C}, opts);
    auto B = at::empty({N, C}, opts);

    run_gn_stats(src, gamma, beta, A.data_ptr<float>(), B.data_ptr<float>(),
                 partial.data_ptr<float>(), N, C, HW, G, D, nblk, chunk, eps, stream);

    if (!transpose) {
        const int C4 = (int)(C / 4);
        const long total4 = (long)HW * (long)C4;
        const int bs = 256;
        dim3 grid((unsigned)((total4 + bs - 1) / bs), (unsigned)N, 1);
        size_t shm = (size_t)2 * C * sizeof(float);
        gn_apply_nhwc_kernel<<<grid, bs, shm, stream>>>(
            src, A.data_ptr<float>(), B.data_ptr<float>(), outp, total4, C4, (int)C);
        C10_CUDA_KERNEL_LAUNCH_CHECK();
    } else {
        dim3 grid((unsigned)((HW + TILE - 1) / TILE),
                  (unsigned)((C + TILE - 1) / TILE), (unsigned)N);
        dim3 blk(TILE, TBY, 1);
        gn_apply_transpose_kernel<<<grid, blk, 0, stream>>>(
            src, A.data_ptr<float>(), B.data_ptr<float>(), res, outp, (int)HW, (int)C);
        C10_CUDA_KERNEL_LAUNCH_CHECK();
    }
}

// cooperative fused GroupNorm; returns false if the fast path is unavailable
template<bool TR>
static bool run_gn_fused(const float* in, const float* gamma, const float* beta,
                         const float* res, float* out, unsigned int* bar,
                         int64_t N, int64_t C, int64_t HW, int G, int D,
                         float eps, const at::TensorOptions& opts,
                         cudaStream_t stream) {
    static int coop_supported = -1;
    static int sm_count = 0;
    if (coop_supported < 0) {
        int dev = 0;
        cudaGetDevice(&dev);
        int v = 0;
        if (cudaDeviceGetAttribute(&v, cudaDevAttrCooperativeLaunch, dev) != cudaSuccess) v = 0;
        coop_supported = v;
        const auto* prop = at::cuda::getCurrentDeviceProperties();
        sm_count = prop ? prop->multiProcessorCount : 0;
    }
    if (!coop_supported || sm_count <= 0) return false;
    if (C % 4 != 0 || C % TILE != 0) return false;

    const size_t smem = (size_t)2 * (size_t)C * sizeof(float);
    int mb = 0;
    if (cudaOccupancyMaxActiveBlocksPerMultiprocessor(
            &mb, (const void*)gn_fused_kernel<TR>, STAT_BS, smem) != cudaSuccess) {
        (void)cudaGetLastError();
        return false;
    }
    if (mb <= 0) return false;

    const int cap   = mb * sm_count;
    const int tiles = (int)((HW + TILE - 1) / TILE);
    int nblk = cap < tiles ? cap : tiles;
    if (nblk < 1) nblk = 1;
    int chunk = (int)((HW + nblk - 1) / nblk);
    chunk = ((chunk + TILE - 1) / TILE) * TILE;
    if (chunk < TILE) chunk = TILE;
    nblk = (int)((HW + chunk - 1) / chunk);
    if (nblk < 1) nblk = 1;
    if (nblk > cap) return false;

    auto partial = at::empty({(int64_t)nblk * (int64_t)G * 2}, opts);
    auto At = at::empty({C}, opts);
    auto Bt = at::empty({C}, opts);

    const float* a_in    = in;
    const float* a_gamma = gamma;
    const float* a_beta  = beta;
    const float* a_res   = res;
    float* a_out  = out;
    float* a_part = partial.data_ptr<float>();
    float* a_A    = At.data_ptr<float>();
    float* a_B    = Bt.data_ptr<float>();
    unsigned int* a_bar = bar;
    int a_N = (int)N, a_HW = (int)HW, a_C = (int)C, a_G = G, a_D = D;
    int a_nblk = nblk, a_chunk = chunk;
    float a_eps = eps;
    float a_cnt = (float)((double)HW * (double)D);

    void* args[] = { (void*)&a_in, (void*)&a_gamma, (void*)&a_beta, (void*)&a_res,
                     (void*)&a_out, (void*)&a_part, (void*)&a_A, (void*)&a_B,
                     (void*)&a_bar, (void*)&a_N, (void*)&a_HW, (void*)&a_C,
                     (void*)&a_G, (void*)&a_D, (void*)&a_nblk, (void*)&a_chunk,
                     (void*)&a_eps, (void*)&a_cnt };

    dim3 grid((unsigned)nblk, 1, 1);
    dim3 blk(STAT_BS, 1, 1);
    cudaError_t err = cudaLaunchCooperativeKernel(
        (const void*)gn_fused_kernel<TR>, grid, blk, args, smem, stream);
    if (err != cudaSuccess) {
        (void)cudaGetLastError();
        return false;
    }
    return true;
}

torch::Tensor fused_resblock(torch::Tensor x,
                             torch::Tensor w1, torch::Tensor gamma1, torch::Tensor beta1,
                             torch::Tensor w2, torch::Tensor gamma2, torch::Tensor beta2,
                             double eps) {
    TORCH_CHECK(x.is_cuda(), "x must be CUDA");
    TORCH_CHECK(x.dim() == 4, "x must be 4D");
    TORCH_CHECK(x.scalar_type() == at::kFloat, "x must be float32");

    const int64_t N = x.size(0), C = x.size(1), H = x.size(2), W = x.size(3);
    const int64_t HW = H * W;
    const int G = 32;
    TORCH_CHECK(C % G == 0 && C % 256 == 0, "unsupported channel count");
    const int D = (int)(C / G);
    TORCH_CHECK(256 % D == 0 || D >= 256, "unsupported group width");

    auto stream = at::cuda::getCurrentCUDAStream();

    auto xc = x.is_contiguous() ? x : x.contiguous();
    auto opts    = xc.options();
    auto opts_cl = opts.memory_format(at::MemoryFormat::ChannelsLast);

    // ---- K0 : x -> NHWC ----
    auto xn = at::empty({N, C, H, W}, opts_cl);
    {
        dim3 grid((unsigned)((HW + TILE - 1) / TILE),
                  (unsigned)((C + TILE - 1) / TILE), (unsigned)N);
        dim3 blk(TILE, TBY, 1);
        nchw_to_nhwc_kernel<<<grid, blk, 0, stream>>>(
            xc.data_ptr<float>(), xn.data_ptr<float>(), (int)HW, (int)C);
        C10_CUDA_KERNEL_LAUNCH_CHECK();
    }

    auto w1c = w1.contiguous(at::MemoryFormat::ChannelsLast);
    auto w2c = w2.contiguous(at::MemoryFormat::ChannelsLast);
    auto g1c = gamma1.is_contiguous() ? gamma1 : gamma1.contiguous();
    auto b1c = beta1.is_contiguous()  ? beta1  : beta1.contiguous();
    auto g2c = gamma2.is_contiguous() ? gamma2 : gamma2.contiguous();
    auto b2c = beta2.is_contiguous()  ? beta2  : beta2.contiguous();

    // one zeroed 2-slot barrier buffer for both grid-synchronised GN kernels
    auto barT = at::zeros({2}, opts.dtype(at::kInt));
    unsigned int* bar0 = reinterpret_cast<unsigned int*>(barT.data_ptr<int>());
    unsigned int* bar1 = bar0 + 1;

    // ---- conv1 (vendor, channels_last in/out) ----
    auto c1 = at::conv2d(xn, w1c, {}, {1, 1}, {1, 1}, {1, 1}, 1);
    if (!c1.is_contiguous(at::MemoryFormat::ChannelsLast))
        c1 = c1.contiguous(at::MemoryFormat::ChannelsLast);

    // ---- GN1 : fused stats+finalize+apply (NHWC -> NHWC) ----
    auto y1 = at::empty({N, C, H, W}, opts_cl);
    bool ok1 = run_gn_fused<false>(c1.data_ptr<float>(),
                                   g1c.data_ptr<float>(), b1c.data_ptr<float>(),
                                   nullptr, y1.data_ptr<float>(), bar0,
                                   N, C, HW, G, D, (float)eps, opts, stream);
    if (!ok1) {
        legacy_gn(c1.data_ptr<float>(), g1c.data_ptr<float>(), b1c.data_ptr<float>(),
                  nullptr, y1.data_ptr<float>(), /*transpose=*/false,
                  N, C, HW, G, D, (float)eps, opts, stream);
    }

    // ---- conv2 (vendor, channels_last in/out) ----
    auto c2 = at::conv2d(y1, w2c, {}, {1, 1}, {1, 1}, {1, 1}, 1);
    if (!c2.is_contiguous(at::MemoryFormat::ChannelsLast))
        c2 = c2.contiguous(at::MemoryFormat::ChannelsLast);

    // ---- GN2 : fused stats+finalize+apply+residual+transpose (NHWC -> NCHW) ----
    auto out = at::empty({N, C, H, W}, opts);
    bool ok2 = run_gn_fused<true>(c2.data_ptr<float>(),
                                  g2c.data_ptr<float>(), b2c.data_ptr<float>(),
                                  xc.data_ptr<float>(), out.data_ptr<float>(), bar1,
                                  N, C, HW, G, D, (float)eps, opts, stream);
    if (!ok2) {
        legacy_gn(c2.data_ptr<float>(), g2c.data_ptr<float>(), b2c.data_ptr<float>(),
                  xc.data_ptr<float>(), out.data_ptr<float>(), /*transpose=*/true,
                  N, C, HW, G, D, (float)eps, opts, stream);
    }

    return out;
}
