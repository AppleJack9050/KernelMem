__global__ void gn_stats_kernel(const float* __restrict__ psum,
                                const float* __restrict__ psq,
                                const float* __restrict__ gamma,
                                const float* __restrict__ beta,
                                float* __restrict__ scale,
                                float* __restrict__ shift,
                                int nPtile, double cnt, double eps, int C) {
    __shared__ double ss[256], sq[256];
    const int idx = blockIdx.x;
    const int n = idx >> 5, gg = idx & 31;
    double a = 0.0, b = 0.0;
    for (int i = threadIdx.x; i < nPtile; i += blockDim.x) {
        a += (double)psum[(size_t)idx * nPtile + i];
        b += (double)psq[(size_t)idx * nPtile + i];
    }
    ss[threadIdx.x] = a;
    sq[threadIdx.x] = b;
    __syncthreads();
    for (int s = 128; s > 0; s >>= 1) {
        if ((int)threadIdx.x < s) {
            ss[threadIdx.x] += ss[threadIdx.x + s];
            sq[threadIdx.x] += sq[threadIdx.x + s];
        }
        __syncthreads();
    }
    if (threadIdx.x < 8) {
        double mean = ss[0] / cnt;
        double var  = sq[0] / cnt - mean * mean;
        if (var < 0.0) var = 0.0;
        double rstd = 1.0 / sqrt(var + eps);
        int ch = gg * 8 + threadIdx.x;
        double gm = (double)gamma[ch], bt = (double)beta[ch];
        scale[n * C + ch] = (float)(rstd * gm);
        shift[n * C + ch] = (float)(bt - mean * rstd * gm);
    }
}

        float4 v = reinterpret_cast<float4*>(base)[i];
        int c0 = (int)((i * 4) & (size_t)cmask);
        float4 a = *reinterpret_cast<const float4*>(scn + c0);
        float4 b = *reinterpret_cast<const float4*>(shn + c0);
        float x0 = v.x * a.x + b.x, x1 = v.y * a.y + b.y;
        float x2 = v.z * a.z + b.z, x3 = v.w * a.w + b.w;
        v.x = x0 / (1.f + expf(-x0));
        v.y = x1 / (1.f + expf(-x1));
        v.z = x2 / (1.f + expf(-x2));
        v.w = x3 / (1.f + expf(-x3));
        reinterpret_cast<float4*>(base)[i] = v;
    }
}

__global__ void final_kernel(const float* __restrict__ o2,
                             const float* __restrict__ sc,
                             const float* __restrict__ sh,
                             const float* __restrict__ resid,
                             float* __restrict__ y, int C, int P) {
    __shared__ float s[32][33];
    const int p0 = blockIdx.x * 32, c0 = blockIdx.y * 32, n = blockIdx.z;
    const int tx = threadIdx.x, ty = threadIdx.y;
    const int ch = c0 + tx;
    const float a = sc[n * C + ch], b = sh[n * C + ch];
    #pragma unroll
    for (int i = 0; i < 32; i += 8) {
        int pp = i + ty, p = p0 + pp;
        float v = 0.f;
        if (p < P) {
            float u = o2[((size_t)n * P + p) * C + ch] * a + b;
            v = u / (1.f + expf(-u));
        }
        s[pp][tx] = v;
    }
    __syncthreads();
    #pragma unroll
    for (int i = 0; i < 32; i += 8) {
        int cc = i + ty, p = p0 + tx;
        if (p < P) {
            size_t o = ((size_t)n * C + (c0 + cc)) * P + p;
            y[o] = s[tx][cc] + resid[o];
        }
    }
}

static void launch_conv(const float* A, const float* B, float* out, float* ps,
                        float* pq, int N, int H, int W, int P, int C,
                        int nPtile, cudaStream_t st) {
    dim3 grid(nPtile, C / BN, N);
    conv_kernel<<<grid, THREADS, 0, st>>>(A, B, out, ps, pq, H, W, P, C, nPtile);
    C10_CUDA_KERNEL_LAUNCH_CHECK();
}

torch::Tensor fused_block(torch::Tensor x, torch::Tensor w1, torch::Tensor g1,
                          torch::Tensor b1, torch::Tensor w2, torch::Tensor g2,
                          torch::Tensor b2, double eps) {
    TORCH_CHECK(x.is_cuda() && x.scalar_type() == at::kFloat, "fp32 cuda only");
    TORCH_CHECK(x.dim() == 4, "NCHW expected");
    auto xc = x.is_contiguous() ? x : x.contiguous();
    auto w1c = w1.is_contiguous() ? w1 : w1.contiguous();
    auto w2c = w2.is_contiguous() ? w2 : w2.contiguous();
    auto g1c = g1.is_contiguous() ? g1 : g1.contiguous();
    auto b1c = b1.is_contiguous() ? b1 : b1.contiguous();
    auto g2c = g2.is_contiguous() ? g2 : g2.contiguous();
    auto b2c = b2.is_contiguous() ? b2 : b2.contiguous();

    const int N = (int)xc.size(0), C = (int)xc.size(1);
    const int H = (int)xc.size(2), W = (int)xc.size(3);
    const int P = H * W;
    TORCH_CHECK(C == 256, "kernel specialised for C=256 (32 groups of 8)");

    auto opts = xc.options();
    auto stream = at::cuda::getDefaultCUDAStream();

    auto xn  = torch::empty({N, P, C}, opts);
    auto t1  = torch::empty({N, P, C}, opts);
    auto t2  = torch::empty({N, P, C}, opts);
    auto Bm1 = torch::empty({9 * C, C}, opts);
    auto Bm2 = torch::empty({9 * C, C}, opts);

    const int nPtile = (P + BM - 1) / BM;
    auto ps = torch::empty({N, 32, nPtile}, opts);
    auto pq = torch::empty({N, 32, nPtile}, opts);
    auto sc1 = torch::empty({N, C}, opts);
    auto sh1 = torch::empty({N, C}, opts);
    auto sc2 = torch::empty({N, C}, opts);
    auto sh2 = torch::empty({N, C}, opts);
    auto y = torch::empty({N, C, H, W}, opts);

    wtrans_kernel<<<C, C, 0, stream>>>(w1c.data_ptr<float>(),
                                       Bm1.data_ptr<float>(), C);
    C10_CUDA_KERNEL_LAUNCH_CHECK();
    wtrans_kernel<<<C, C, 0, stream>>>(w2c.data_ptr<float>(),
                                       Bm2.data_ptr<float>(), C);
    C10_CUDA_KERNEL_LAUNCH_CHECK();

    dim3 tgrid((P + 31) / 32, C / 32, N);
    dim3 tblk(32, 8);
    nchw2nhwc_kernel<<<tgrid, tblk, 0, stream>>>(xc.data_ptr<float>(),
                                                 xn.data_ptr<float>(), C, P);
    C10_CUDA_KERNEL_LAUNCH_CHECK();

    const double cnt = (double)8.0 * (double)P;

    launch_conv(xn.data_ptr<float>(), Bm1.data_ptr<float>(),
                t1.data_ptr<float>(), ps.data_ptr<float>(),
                pq.data_ptr<float>(), N, H, W, P, C, nPtile, stream);
    gn_stats_kernel<<<N * 32, 256, 0, stream>>>(
        ps.data_ptr<float>(), pq.data_ptr<float>(), g1c.data_ptr<float>(),
        b1c.data_ptr<float>(), sc1.data_ptr<float>(), sh1.data_ptr<float>(),
        nPtile, cnt, eps, C);
    C10_CUDA_KERNEL_LAUNCH_CHECK();

    const size_t PC4 = (size_t)P * (size_t)C / 4;
    int nblk = (int)((PC4 + 255) / 256);
    if (nblk > 4096) nblk = 4096;
    dim3 ngrid(nblk, N);
    norm_silu_kernel<<<ngrid, 256, 0, stream>>>(
        t1.data_ptr<float>(), sc1.data_ptr<float>(), sh1.data_ptr<float>(),
        C, PC4);
    C10_CUDA_KERNEL_LAUNCH_CHECK();

    launch_conv(t1.data_ptr<float>(), Bm2.data_ptr<float>(),
                t2.data_ptr<float>(), ps.data_ptr<float>(),
                pq.data_ptr<float>(), N, H, W, P, C, nPtile, stream);
    gn_stats_kernel<<<N * 32, 256, 0, stream>>>(
        ps.data_ptr<float>(), pq.data_ptr<float>(), g2c.data_ptr<float>(),
        b2c.data_ptr<float>(), sc2.data_ptr<float>(), sh2.data_ptr<float>(),
        nPtile, cnt, eps, C);
    C10_CUDA_KERNEL_LAUNCH_CHECK();

    final_kernel<<<tgrid, tblk, 0, stream>>>(
        t2.data_ptr<float>(), sc2.data_ptr<float>(), sh2.data_ptr<float>(),
        xc.data_ptr<float>(), y.data_ptr<float>(), C, P);
    C10_CUDA_KERNEL_LAUNCH_CHECK();

    return y;
}
