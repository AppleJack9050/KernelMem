#define CC 256
#define GG 32
#define CPG 8

// grid (nchunk, B), block 256 threads == one per channel.
// MLP-improved variant of round-1's per-channel reduction: the pixel loop is
// manually unrolled by 4 so 4 independent strided loads are issued before
// any of them are consumed, raising memory-level parallelism. Accumulation
// stays scalar fp32 (s, s2); no precision change, no half-precision, no
// vectorized-across-channel load (that would combine two DIFFERENT channels
// per element and is not applied here -- see Section A deviation note).
__global__ void gn_stats_nhwc(const float* __restrict__ x, float* __restrict__ psum,
                              float* __restrict__ psum2, long N, int nchunk) {
  const int chunk = blockIdx.x, b = blockIdx.y;
  const long begin = (long)(chunk) * N / nchunk;
  const long end = (long)(chunk + 1) * N / nchunk;
  const int c = threadIdx.x;
  const float* p = x + (long)b * N * CC + begin * CC + c;
  const long n = end - begin;
  float s = 0.f, s2 = 0.f;
  long i = 0;
  for (; i + 4 <= n; i += 4) {
    float v0 = p[(i + 0) * CC];
    float v1 = p[(i + 1) * CC];
    float v2 = p[(i + 2) * CC];
    float v3 = p[(i + 3) * CC];
    s += v0 + v1 + v2 + v3;
    s2 += v0 * v0 + v1 * v1 + v2 * v2 + v3 * v3;
  }
  for (; i < n; ++i) {
    float v = p[i * CC];
    s += v; s2 += v * v;
  }
  // reduce within groups of CPG=8 consecutive lanes
  #pragma unroll
  for (int off = CPG >> 1; off > 0; off >>= 1) {
    s += __shfl_down_sync(0xffffffff, s, off, CPG);
    s2 += __shfl_down_sync(0xffffffff, s2, off, CPG);
  }
  if ((c & (CPG - 1)) == 0) {
    int g = c / CPG;
    long o = ((long)b * nchunk + chunk) * GG + g;
    psum[o] = s; psum2[o] = s2;
  }
}

// grid B*GG
__global__ void gn_finalize_nhwc(const float* __restrict__ psum, const float* __restrict__ psum2,
                                 float* __restrict__ mean, float* __restrict__ rstd,
                                 int nchunk, float inv_n, float eps) {
  const int bg = blockIdx.x;
  const int b = bg / GG, g = bg - b * GG;
  float s = 0.f, s2 = 0.f;
  for (int i = threadIdx.x; i < nchunk; i += blockDim.x) {
    long o = ((long)b * nchunk + i) * GG + g;
    s += psum[o]; s2 += psum2[o];
  }
  __shared__ float sa[32], sb[32];
  int lane = threadIdx.x & 31, wid = threadIdx.x >> 5;
  warp_red2(s, s2);
  if (lane == 0) { sa[wid] = s; sb[wid] = s2; }
  __syncthreads();
  int nw = blockDim.x >> 5;
  if (wid == 0) {
    s = (lane < nw) ? sa[lane] : 0.f; s2 = (lane < nw) ? sb[lane] : 0.f;
    warp_red2(s, s2);
    if (lane == 0) {
      float m = s * inv_n; float v = s2 * inv_n - m * m; v = v < 0.f ? 0.f : v;
      mean[bg] = m; rstd[bg] = rsqrtf(v + eps);
    }
  }
}

// NHWC in -> NHWC out, normalize + affine + silu
__global__ void gn_apply_nhwc(const float* __restrict__ x, const float* __restrict__ mean,
                              const float* __restrict__ rstd, const float* __restrict__ gamma,
                              const float* __restrict__ beta, float* __restrict__ out, long N) {
  const int b = blockIdx.y;
  const long total = N * (CC / 4);   // number of float4 for this batch
  const float4* xv = reinterpret_cast<const float4*>(x + (long)b * N * CC);
  float4* ov = reinterpret_cast<float4*>(out + (long)b * N * CC);
  const float4* gv = reinterpret_cast<const float4*>(gamma);
  const float4* bv = reinterpret_cast<const float4*>(beta);
  const long stride = (long)gridDim.x * blockDim.x;
  for (long i = (long)blockIdx.x * blockDim.x + threadIdx.x; i < total; i += stride) {
    int c4 = (int)(i & (CC / 4 - 1));
    int g = c4 / (CPG / 4);
    float m = mean[b * GG + g], r = rstd[b * GG + g];
    float4 gm = gv[c4], bt = bv[c4], v = xv[i], o;
    o.x = silu((v.x - m) * r * gm.x + bt.x);
    o.y = silu((v.y - m) * r * gm.y + bt.y);
    o.z = silu((v.z - m) * r * gm.z + bt.z);
    o.w = silu((v.w - m) * r * gm.w + bt.w);
    ov[i] = o;
  }
}

// NHWC in + NCHW residual -> NCHW out, normalize + affine + silu + add
// 32x32 tile geometry (restore_occupancy_shrink_tile_and_registers):
// shrinks shared memory from 18176 B (64x64 tile) to 4480 B (32x32 tile) and
// caps registers at 32 via __launch_bounds__(256, 8), raising theoretical
// occupancy from 62.5% to 100% so the kernel becomes DRAM-limited instead of
// latency-limited (mirrors the 32x32 geometry that nchw_to_nhwc_kernel
// already proves reaches ~1.90 TB/s).
#define TP2 32
#define TC2 32
__global__ void __launch_bounds__(256, 8) gn_apply_nhwc2nchw(
    const float* __restrict__ x, const float* __restrict__ mean,
    const float* __restrict__ rstd, const float* __restrict__ gamma,
    const float* __restrict__ beta, const float* __restrict__ res,
    float* __restrict__ out, long N) {
  __shared__ float tile[TP2][TC2 + 1];   // tile[pixel][channel]
  __shared__ float ssc[TC2], ssh[TC2];
  const long p0 = (long)blockIdx.x * TP2;
  const int c0 = blockIdx.y * TC2;
  const int b = blockIdx.z;

  const int tid = threadIdx.y * 32 + threadIdx.x;
  if (tid < TC2) {
    int c = c0 + tid;
    int g = c >> 3;   // CPG == 8
    float sc = rstd[b * GG + g] * gamma[c];
    ssc[tid] = sc;
    ssh[tid] = beta[c] - mean[b * GG + g] * sc;
  }
  __syncthreads();

  const long xbase = ((long)b * N + p0) * CC + c0;
  #pragma unroll
  for (int k = 0; k < 4; ++k) {
    int pl = threadIdx.y + 8 * k;
    long p = p0 + pl;
    float v = 0.f;
    if (p < N) {
      v = x[xbase + (long)pl * CC + threadIdx.x];
      v = silu(v * ssc[threadIdx.x] + ssh[threadIdx.x]);
    }
    tile[pl][threadIdx.x] = v;
  }
  __syncthreads();

  long p = p0 + threadIdx.x;
  if (p < N) {
    #pragma unroll
    for (int k = 0; k < 4; ++k) {
      int c = c0 + threadIdx.y + 8 * k;
      long idx = ((long)b * CC + c) * N + p;
      out[idx] = tile[threadIdx.x][threadIdx.y + 8 * k] + res[idx];
    }
  }
}

std::vector<torch::Tensor> gn_stats_nhwc_run(torch::Tensor x, double eps) {
  const int B = x.size(0);
  const long N = x.size(2) * (long)x.size(3);
  auto opts = torch::TensorOptions().dtype(torch::kFloat).device(x.device());
  int nchunk = (int)std::min<long>(256L, std::max<long>(1L, N / 64));
  auto psum = torch::empty({(long)B * nchunk * GG}, opts);
  auto psum2 = torch::empty({(long)B * nchunk * GG}, opts);
  auto mean = torch::empty({(long)B * GG}, opts);
  auto rstd = torch::empty({(long)B * GG}, opts);
  auto stream = at::cuda::getCurrentCUDAStream();
  gn_stats_nhwc<<<dim3(nchunk, B), CC, 0, stream>>>(x.data_ptr<float>(), psum.data_ptr<float>(),
      psum2.data_ptr<float>(), N, nchunk);
  C10_CUDA_KERNEL_LAUNCH_CHECK();
  gn_finalize_nhwc<<<B * GG, 256, 0, stream>>>(psum.data_ptr<float>(), psum2.data_ptr<float>(),
      mean.data_ptr<float>(), rstd.data_ptr<float>(), nchunk,
      (float)(1.0 / (double)(CPG * N)), (float)eps);
  C10_CUDA_KERNEL_LAUNCH_CHECK();
  return {mean, rstd};
}

// x: channels-last (B,256,H,W) -> out channels-last

/* ---------------- coalesced NCHW->NHWC staging transpose ---------------- */
// Grid: dim3((N+31)/32, C/32, B); block dim3(32,8) = 256 threads.
__global__ void nchw_to_nhwc_kernel(const float* __restrict__ x, float* __restrict__ y,
                                     long N, int C) {
  __shared__ float tile[32][33];
  const long p0 = (long)blockIdx.x * 32;
  const int c0 = blockIdx.y * 32;
  const int b = blockIdx.z;

  const long p = p0 + threadIdx.x;
  #pragma unroll
  for (int k = 0; k < 4; ++k) {
    int c = c0 + threadIdx.y + 8 * k;
    tile[threadIdx.y + 8 * k][threadIdx.x] = (p < N) ? x[((long)(b * C + c)) * N + p] : 0.f;
  }
  __syncthreads();

  #pragma unroll
  for (int k = 0; k < 4; ++k) {
    long pp = p0 + threadIdx.y + 8 * k;
    if (pp < N) {
      y[((long)b * N + pp) * C + c0 + threadIdx.x] = tile[threadIdx.x][threadIdx.y + 8 * k];
    }
  }
}

torch::Tensor to_nhwc(torch::Tensor x) {
  TORCH_CHECK(x.is_cuda() && x.scalar_type() == at::kFloat);
  TORCH_CHECK(x.is_contiguous(), "x must be NCHW-contiguous");
  const int B = x.size(0), C = x.size(1), H = x.size(2), W = x.size(3);
  TORCH_CHECK(C % 32 == 0, "C must be multiple of 32");
  const long N = (long)H * W;
  auto y = torch::empty({B, H, W, C}, x.options());
  auto stream = at::cuda::getCurrentCUDAStream();
  dim3 block(32, 8);
  dim3 grid((unsigned)((N + 31) / 32), C / 32, B);
  nchw_to_nhwc_kernel<<<grid, block, 0, stream>>>(x.data_ptr<float>(), y.data_ptr<float>(), N, C);
  C10_CUDA_KERNEL_LAUNCH_CHECK();
  return y.permute({0, 3, 1, 2});
}

/* ---------------- coalesced weight (O,C,3,3)->(O,3,3,C) staging permute ---------------- */
__global__ void weight_to_nhwc_kernel(const float* __restrict__ w, float* __restrict__ o, int C) {
  extern __shared__ float s[];
  const int oc = blockIdx.x;
  const long base = (long)oc * C * 9;
  for (int i = threadIdx.x; i < C * 9; i += blockDim.x) s[i] = w[base + i];
  __syncthreads();
  for (int j = threadIdx.x; j < C * 9; j += blockDim.x) {
    int rs = j / C, c = j - rs * C;
    o[base + j] = s[c * 9 + rs];
  }
}

torch::Tensor weight_to_nhwc(torch::Tensor w) {
  TORCH_CHECK(w.is_cuda() && w.scalar_type() == at::kFloat);
  TORCH_CHECK(w.is_contiguous(), "weight must be contiguous");
  const int O = w.size(0), C = w.size(1), KH = w.size(2), KW = w.size(3);
  TORCH_CHECK(KH == 3 && KW == 3, "expects 3x3 kernel");
  auto o = torch::empty({O, 3, 3, C}, w.options());

        super().__init__()

    def forward(self, x, conv1_weight, norm1_weight, norm1_bias,
                conv2_weight, norm2_weight, norm2_bias, eps):
        G = 32
        x = x if x.is_contiguous() else x.contiguous()
