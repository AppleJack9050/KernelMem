_CUDA_SRC = r"""
#include <torch/extension.h>
#include <ATen/cuda/CUDAContext.h>
#include <c10/cuda/CUDAException.h>
#include <cuda_runtime.h>
#include <algorithm>

#define CCH 256          // channels (const per problem definition)
#define NGRP 32          // num_groups (const)
#define CPG  8           // channels per group

__device__ __forceinline__ float silu_f(float t) {
    return t / (1.0f + __expf(-t));
}

// ---------------------------------------------------------------------------
// K0: NCHW -> NHWC tiled transpose (32 pixels x 32 channels per block)
// ---------------------------------------------------------------------------
__global__ void to_nhwc_kernel(const float* __restrict__ x,
                               float* __restrict__ out,
                               int HW, int C) {
    __shared__ float sh[32][33];
    const int n  = blockIdx.z;
    const int cb = blockIdx.y * 32;
    const int pb = blockIdx.x * 32;
    const int tx = threadIdx.x;   // 0..31
    const int ty = threadIdx.y;   // 0..7

    const float* xb = x + (long)n * C * HW;
    #pragma unroll
    for (int i = 0; i < 4; ++i) {
        int c = cb + ty + i * 8;
        int p = pb + tx;
        float v = 0.0f;
        if (p < HW) v = xb[(long)c * HW + p];
        sh[ty + i * 8][tx] = v;
    }
    __syncthreads();

    float* ob = out + (long)n * (long)HW * C;
    #pragma unroll
    for (int i = 0; i < 4; ++i) {
        int p = pb + ty + i * 8;
        int c = cb + tx;
        if (p < HW) ob[(long)p * C + c] = sh[tx][ty + i * 8];
    }
}

// ---------------------------------------------------------------------------
// K1: per-(n, group) partial sums over a tile of pixels. NHWC, float4 loads.
// ---------------------------------------------------------------------------
__global__ void gn_stats_kernel(const float* __restrict__ y,
                                float* __restrict__ psum,
                                float* __restrict__ psq,
                                int HW, int tile, int ntiles) {
    const int n = blockIdx.y;
    const int t = blockIdx.x;
    const int c4 = threadIdx.x & 63;     // float4 index within the 256 channels
    const int sp = threadIdx.x >> 6;     // 0..3 pixel phase

    const float4* y4 = (const float4*)(y + (long)n * (long)HW * CCH);

    int p0 = t * tile;
    int p1 = p0 + tile;
    if (p1 > HW) p1 = HW;

    float s = 0.0f, q = 0.0f;
    for (int p = p0 + sp; p < p1; p += 4) {
        float4 v = y4[(long)p * 64 + c4];
        s += v.x + v.y + v.z + v.w;
        q += v.x * v.x + v.y * v.y + v.z * v.z + v.w * v.w;
    }

    s += __shfl_xor_sync(0xffffffffu, s, 1);
    q += __shfl_xor_sync(0xffffffffu, q, 1);

    __shared__ float ss[4][32];
    __shared__ float sq[4][32];
    if ((threadIdx.x & 1) == 0) {
        int g = c4 >> 1;                 // 0..31
        ss[sp][g] = s;
        sq[sp][g] = q;
    }
    __syncthreads();

    if (threadIdx.x < NGRP) {
        int g = threadIdx.x;
        float ts = ss[0][g] + ss[1][g] + ss[2][g] + ss[3][g];
        float tq = sq[0][g] + sq[1][g] + sq[2][g] + sq[3][g];
        long idx = ((long)n * NGRP + g) * ntiles + t;
        psum[idx] = ts;
        psq[idx]  = tq;
    }
}

// ---------------------------------------------------------------------------
// K1b (NEW, plan item 3): per-(n, group) partial sums, NCHW layout.
//   grid = (ntiles, N*NGRP); each block reduces a contiguous pixel range over
//   the 8 channels of one group.  float4 along W when HW % 4 == 0.
// ---------------------------------------------------------------------------
__global__ void gn_stats_nchw_kernel(const float* __restrict__ x,
                                     float* __restrict__ psum,
                                     float* __restrict__ psq,
                                     int HW, int tile, int ntiles, int vec) {
    const int t  = blockIdx.x;
    const int ng = blockIdx.y;           // n*NGRP + g
    const int n  = ng / NGRP;
    const int g  = ng - n * NGRP;

    int p0 = t * tile;
    int p1 = p0 + tile;
    if (p1 > HW) p1 = HW;

    const float* base = x + ((long)n * CCH + (long)g * CPG) * (long)HW;

    float s = 0.0f, q = 0.0f;

    if (vec) {
        const int v0 = p0 >> 2;
        const int v1 = p1 >> 2;
        #pragma unroll
        for (int c = 0; c < CPG; ++c) {
            const float4* b4 = (const float4*)(base + (long)c * HW);
            for (int v = v0 + threadIdx.x; v < v1; v += blockDim.x) {
                float4 val = b4[v];
                s += val.x + val.y + val.z + val.w;
                q += val.x * val.x + val.y * val.y + val.z * val.z + val.w * val.w;
            }
        }
    } else {
        #pragma unroll
        for (int c = 0; c < CPG; ++c) {
            const float* b = base + (long)c * HW;
            for (int p = p0 + threadIdx.x; p < p1; p += blockDim.x) {
                float v = b[p];
                s += v;
                q += v * v;
            }
        }
    }

    #pragma unroll
    for (int off = 16; off > 0; off >>= 1) {
        s += __shfl_down_sync(0xffffffffu, s, off);
        q += __shfl_down_sync(0xffffffffu, q, off);
    }

    __shared__ float ws_[8], wq_[8];
    const int warp = threadIdx.x >> 5;
    const int lane = threadIdx.x & 31;
    if (lane == 0) { ws_[warp] = s; wq_[warp] = q; }
    __syncthreads();

    if (threadIdx.x == 0) {
        const int nw = blockDim.x >> 5;
        float ts = 0.0f, tq = 0.0f;
        for (int i = 0; i < nw; ++i) { ts += ws_[i]; tq += wq_[i]; }
        long idx = (long)ng * ntiles + t;
        psum[idx] = ts;
        psq[idx]  = tq;
    }
}

// ---------------------------------------------------------------------------
// K2: reduce partials -> mean/rstd -> per-(n,channel) scale/shift  (UNCHANGED)
// ---------------------------------------------------------------------------
__global__ void gn_reduce_kernel(const float* __restrict__ psum,
                                 const float* __restrict__ psq,
                                 const float* __restrict__ gamma,
                                 const float* __restrict__ beta,
                                 float* __restrict__ scale,
                                 float* __restrict__ shift,
                                 int ntiles, int HW, float eps) {
    const int ng = blockIdx.x;
    const int n  = ng / NGRP;
    const int g  = ng % NGRP;

    const float* pb_s = psum + (long)ng * ntiles;
    const float* pb_q = psq  + (long)ng * ntiles;

    float s = 0.0f, q = 0.0f;
    for (int i = threadIdx.x; i < ntiles; i += blockDim.x) {
        s += pb_s[i];
        q += pb_q[i];
    }
    #pragma unroll
    for (int off = 16; off > 0; off >>= 1) {
        s += __shfl_down_sync(0xffffffffu, s, off);
        q += __shfl_down_sync(0xffffffffu, q, off);
    }
    __shared__ float ws[8], wq[8];
    int warp = threadIdx.x >> 5;
    int lane = threadIdx.x & 31;
    if (lane == 0) { ws[warp] = s; wq[warp] = q; }
    __syncthreads();

    if (threadIdx.x < CPG) {
        int nwarps = blockDim.x >> 5;
        float ts = 0.0f, tq = 0.0f;
        for (int i = 0; i < nwarps; ++i) { ts += ws[i]; tq += wq[i]; }
        float cnt  = (float)HW * (float)CPG;
        float mean = ts / cnt;
        float var  = tq / cnt - mean * mean;
        if (var < 0.0f) var = 0.0f;
        float rstd = rsqrtf(var + eps);
        int c = g * CPG + (int)threadIdx.x;
        float gm = gamma[c];
        float bt = beta[c];
        scale[n * CCH + c] = rstd * gm;
        shift[n * CCH + c] = bt - mean * rstd * gm;
    }
}

// ---------------------------------------------------------------------------
// K3: GN affine + SiLU, NHWC in / NHWC out, float4
// ---------------------------------------------------------------------------
__global__ void gn_apply_silu_nhwc_kernel(const float* __restrict__ y,
                                          const float* __restrict__ scale,
                                          const float* __restrict__ shift,
                                          float* __restrict__ out,
                                          int HW) {
    const int n = blockIdx.y;
    long i = (long)blockIdx.x * blockDim.x + threadIdx.x;
    long total = (long)HW * 64;
    if (i >= total) return;

    const float4* y4 = (const float4*)(y   + (long)n * (long)HW * CCH);
    float4*       o4 = (float4*)      (out + (long)n * (long)HW * CCH);
    const float4* s4 = (const float4*)(scale + (long)n * CCH);
    const float4* b4 = (const float4*)(shift + (long)n * CCH);

    int c4 = (int)(i & 63);
    float4 v = y4[i];
    float4 s = s4[c4];
    float4 b = b4[c4];
    float4 r;
    r.x = silu_f(v.x * s.x + b.x);
    r.y = silu_f(v.y * s.y + b.y);
    r.z = silu_f(v.z * s.z + b.z);
    r.w = silu_f(v.w * s.w + b.w);
    o4[i] = r;
}

// ---------------------------------------------------------------------------
// K4: GN affine + SiLU + residual add + NHWC->NCHW transpose (one pass)
// ---------------------------------------------------------------------------
__global__ void gn_apply_silu_add_t_kernel(const float* __restrict__ y,     // NHWC
                                           const float* __restrict__ res,   // NCHW
                                           const float* __restrict__ scale,
                                           const float* __restrict__ shift,
                                           float* __restrict__ out,         // NCHW
                                           int HW) {
    __shared__ float sh[32][33];
    const int n  = blockIdx.z;
    const int cb = blockIdx.y * 32;
    const int pb = blockIdx.x * 32;
    const int tx = threadIdx.x;   // 0..31
    const int ty = threadIdx.y;   // 0..7

    const float* yb = y + (long)n * (long)HW * CCH;
    const int c_in = cb + tx;
    const float sc = scale[n * CCH + c_in];
    const float sf = shift[n * CCH + c_in];

    #pragma unroll
    for (int i = 0; i < 4; ++i) {
        int p = pb + ty + i * 8;
        float o = 0.0f;
        if (p < HW) {
            float v = yb[(long)p * CCH + c_in];
            o = silu_f(v * sc + sf);
        }
        sh[ty + i * 8][tx] = o;
    }
    __syncthreads();

    const float* rb = res + (long)n * CCH * (long)HW;
    float*       ob = out + (long)n * CCH * (long)HW;
    #pragma unroll
    for (int i = 0; i < 4; ++i) {
        int c = cb + ty + i * 8;
        int p = pb + tx;
        if (p < HW) {
            ob[(long)c * HW + p] = sh[tx][ty + i * 8] + rb[(long)c * HW + p];
        }
    }
}

static void compute_affine(const at::Tensor& y, const at::Tensor& gamma,
                           const at::Tensor& beta, int N, int HW, float eps,
                           at::Tensor& scale, at::Tensor& shift) {
    int bpi = (HW + 15) / 16;
    int cap = std::max(1, 8192 / std::max(1, N));
    if (bpi > cap) bpi = cap;
    int tile = (HW + bpi - 1) / bpi;
    tile = ((tile + 3) / 4) * 4;              // multiple of 4 pixels
    if (tile < 4) tile = 4;
    bpi = (HW + tile - 1) / tile;             // no empty tile

    auto opts = y.options();
    auto psum = at::empty({(long)N * NGRP * bpi}, opts);
    auto psq  = at::empty({(long)N * NGRP * bpi}, opts);

    auto stream = at::cuda::getCurrentCUDAStream();

    gn_stats_kernel<<<dim3(bpi, N), 256, 0, stream>>>(
        y.data_ptr<float>(), psum.data_ptr<float>(), psq.data_ptr<float>(),
        HW, tile, bpi);
    C10_CUDA_KERNEL_LAUNCH_CHECK();

    gn_reduce_kernel<<<N * NGRP, 256, 0, stream>>>(
        psum.data_ptr<float>(), psq.data_ptr<float>(),
        gamma.data_ptr<float>(), beta.data_ptr<float>(),
        scale.data_ptr<float>(), shift.data_ptr<float>(),
        bpi, HW, eps);
    C10_CUDA_KERNEL_LAUNCH_CHECK();
}

// NEW: NCHW statistics + reduce (gn_reduce_kernel reused unchanged)
static void compute_affine_nchw(const at::Tensor& y, const at::Tensor& gamma,
                                const at::Tensor& beta, int N, int HW, float eps,
                                at::Tensor& scale, at::Tensor& shift) {
    int bpi = (HW + 2047) / 2048;
    int cap = std::max(1, 4096 / std::max(1, N));
    if (bpi > cap) bpi = cap;
    if (bpi < 1) bpi = 1;
    int tile = (HW + bpi - 1) / bpi;
    tile = ((tile + 3) / 4) * 4;              // multiple of 4 pixels
    if (tile < 4) tile = 4;
    bpi = (HW + tile - 1) / tile;

    const int vec = (HW % 4 == 0) ? 1 : 0;

    auto opts = y.options();
    auto psum = at::empty({(long)N * NGRP * bpi}, opts);
    auto psq  = at::empty({(long)N * NGRP * bpi}, opts);

    auto stream = at::cuda::getCurrentCUDAStream();

    gn_stats_nchw_kernel<<<dim3(bpi, N * NGRP), 256, 0, stream>>>(
        y.data_ptr<float>(), psum.data_ptr<float>(), psq.data_ptr<float>(),
        HW, tile, bpi, vec);
    C10_CUDA_KERNEL_LAUNCH_CHECK();

    gn_reduce_kernel<<<N * NGRP, 256, 0, stream>>>(
        psum.data_ptr<float>(), psq.data_ptr<float>(),
        gamma.data_ptr<float>(), beta.data_ptr<float>(),
        scale.data_ptr<float>(), shift.data_ptr<float>(),
        bpi, HW, eps);
    C10_CUDA_KERNEL_LAUNCH_CHECK();
}

at::Tensor to_nhwc_cuda(at::Tensor x) {
    TORCH_CHECK(x.is_cuda() && x.scalar_type() == at::kFloat, "float32 cuda tensor expected");
    TORCH_CHECK(x.dim() == 4, "4d tensor expected");
    int N = (int)x.size(0), C = (int)x.size(1);
    int H = (int)x.size(2), W = (int)x.size(3);
    TORCH_CHECK(C % 32 == 0, "channels must be a multiple of 32");
    int HW = H * W;
    auto xc = x.is_contiguous() ? x : x.contiguous();
    auto out = at::empty_like(xc, xc.options(), at::MemoryFormat::ChannelsLast);
    auto stream = at::cuda::getCurrentCUDAStream();
    dim3 grid((HW + 31) / 32, C / 32, N);
    dim3 blk(32, 8);
    to_nhwc_kernel<<<grid, blk, 0, stream>>>(xc.data_ptr<float>(), out.data_ptr<float>(), HW, C);
    C10_CUDA_KERNEL_LAUNCH_CHECK();
    return out;
}

at::Tensor gn_silu_nhwc_cuda(at::Tensor y, at::Tensor gamma, at::Tensor beta, double eps) {
    TORCH_CHECK(y.is_cuda() && y.scalar_type() == at::kFloat, "float32 cuda tensor expected");
    TORCH_CHECK(y.dim() == 4 && y.size(1) == CCH, "expected (N,256,H,W)");
    TORCH_CHECK(y.is_contiguous(at::MemoryFormat::ChannelsLast), "expected channels_last");
    int N = (int)y.size(0);
    int HW = (int)(y.size(2) * y.size(3));

    auto scale = at::empty({(long)N * CCH}, y.options());
    auto shift = at::empty({(long)N * CCH}, y.options());
    compute_affine(y, gamma, beta, N, HW, (float)eps, scale, shift);

    auto out = at::empty_like(y);
    auto stream = at::cuda::getCurrentCUDAStream();
    long per_img = (long)HW * 64;
    int blocks = (int)((per_img + 255) / 256);
    dim3 grid(blocks, N);
    gn_apply_silu_nhwc_kernel<<<grid, 256, 0, stream>>>(
        y.data_ptr<float>(), scale.data_ptr<float>(), shift.data_ptr<float>(),
        out.data_ptr<float>(), HW);
    C10_CUDA_KERNEL_LAUNCH_CHECK();
    return out;
}

static void gn_silu_add_nchw_core(at::Tensor& y, at::Tensor& res, at::Tensor& gamma,
                                  at::Tensor& beta, double eps, at::Tensor& out) {
    TORCH_CHECK(y.is_cuda() && y.scalar_type() == at::kFloat, "float32 cuda tensor expected");
    TORCH_CHECK(y.dim() == 4 && y.size(1) == CCH, "expected (N,256,H,W)");
    TORCH_CHECK(y.is_contiguous(at::MemoryFormat::ChannelsLast), "expected channels_last");
    TORCH_CHECK(res.is_contiguous(), "residual must be contiguous NCHW");
    TORCH_CHECK(out.is_cuda() && out.scalar_type() == at::kFloat, "float32 cuda out expected");
    TORCH_CHECK(out.is_contiguous(), "out must be contiguous NCHW");
    int N = (int)y.size(0);
    int H = (int)y.size(2), W = (int)y.size(3);

    TORCH_CHECK(out.dim() == 4 && out.size(0) == N && out.size(1) == CCH &&

    TORCH_CHECK(res.size(0) == N && res.size(1) == CCH &&
                res.size(2) == H && res.size(3) == W, "residual shape mismatch");

    auto scale = at::empty({(long)N * CCH}, y.options());
    auto shift = at::empty({(long)N * CCH}, y.options());
    compute_affine(y, gamma, beta, N, HW, (float)eps, scale, shift);

    dim3 blk(32, 8);

static inline int grid_for(long n, int block) {
    long b = (n + block - 1) / block;
    if (b < 1) b = 1;
    if (b > 262144) b = 262144;
    return (int)b;
}

static void gn_silu_nchw_core(at::Tensor& y, at::Tensor& gamma, at::Tensor& beta,
                              double eps, at::Tensor& out) {
    TORCH_CHECK(y.is_cuda() && y.scalar_type() == at::kFloat, "float32 cuda tensor expected");
    TORCH_CHECK(y.dim() == 4 && y.size(1) == CCH, "expected (N,256,H,W)");
    TORCH_CHECK(y.is_contiguous(), "expected contiguous NCHW");
    TORCH_CHECK(out.is_contiguous(), "out must be contiguous NCHW");
    const int N  = (int)y.size(0);
    const int HW = (int)(y.size(2) * y.size(3));

    auto scale = at::empty({(long)N * CCH}, y.options());
    auto shift = at::empty({(long)N * CCH}, y.options());
    compute_affine_nchw(y, gamma, beta, N, HW, (float)eps, scale, shift);

    auto stream = at::cuda::getCurrentCUDAStream();
    const long total = (long)N * CCH * (long)HW;
    if (HW % 4 == 0) {
        const long total4 = total >> 2;
        int blocks = grid_for(total4, 256);
        gn_apply_silu_nchw_v4_kernel<<<blocks, 256, 0, stream>>>(
            (const float4*)y.data_ptr<float>(),
            scale.data_ptr<float>(), shift.data_ptr<float>(),
            (float4*)out.data_ptr<float>(), HW >> 2, total4);
    } else {
        int blocks = grid_for(total, 256);
        gn_apply_silu_nchw_kernel<<<blocks, 256, 0, stream>>>(
            y.data_ptr<float>(), scale.data_ptr<float>(), shift.data_ptr<float>(),
            out.data_ptr<float>(), HW, total);
    }
    C10_CUDA_KERNEL_LAUNCH_CHECK();
}

_CONV_CUDA_SRC = r"""
#include <torch/extension.h>
#include <ATen/cuda/CUDAContext.h>
#include <ATen/cudnn/Handle.h>
#include <cudnn.h>
#include <cuda_runtime.h>
#include <map>
#include <tuple>
#include <vector>
#include <algorithm>

#define CK(expr) TORCH_CHECK((expr) == CUDNN_STATUS_SUCCESS, "cudnn call failed: ", #expr)

struct ConvPlan {
    // NHWC implicit-GEMM (round-3 path, untouched)
    cudnnTensorDescriptor_t      xd = nullptr;
    cudnnTensorDescriptor_t      od = nullptr;
    cudnnFilterDescriptor_t      fd = nullptr;
    cudnnConvolutionDescriptor_t cd = nullptr;
    cudnnConvolutionFwdAlgo_t    algo = CUDNN_CONVOLUTION_FWD_ALGO_IMPLICIT_PRECOMP_GEMM;
    size_t                       ws = 0;

    // NCHW Winograd (plan item 1)
    cudnnTensorDescriptor_t      xd_n = nullptr;
    cudnnTensorDescriptor_t      od_n = nullptr;
    cudnnFilterDescriptor_t      fd_n = nullptr;
    cudnnConvolutionDescriptor_t cd_n = nullptr;
    bool                         use_winograd = false;
    cudnnConvolutionFwdAlgo_t    algo_w = CUDNN_CONVOLUTION_FWD_ALGO_WINOGRAD;
    size_t                       ws_w = 0;
};

using ck_t = std::tuple<int,int,int,int,int>;
static std::map<ck_t, ConvPlan> g_plans;

static ConvPlan& get_plan(int N, int C, int H, int W, int K) {
    ck_t key(N, C, H, W, K);
    auto it = g_plans.find(key);
    if (it != g_plans.end()) return it->second;

    ConvPlan p;
    CK(cudnnCreateTensorDescriptor(&p.xd));
    CK(cudnnSetTensor4dDescriptor(p.xd, CUDNN_TENSOR_NHWC, CUDNN_DATA_FLOAT, N, C, H, W));
    CK(cudnnCreateTensorDescriptor(&p.od));
    CK(cudnnSetTensor4dDescriptor(p.od, CUDNN_TENSOR_NHWC, CUDNN_DATA_FLOAT, N, K, H, W));
    CK(cudnnCreateFilterDescriptor(&p.fd));
    CK(cudnnSetFilter4dDescriptor(p.fd, CUDNN_DATA_FLOAT, CUDNN_TENSOR_NHWC, K, C, 3, 3));
    CK(cudnnCreateConvolutionDescriptor(&p.cd));
    CK(cudnnSetConvolution2dDescriptor(p.cd, 1, 1, 1, 1, 1, 1,
                                       CUDNN_CROSS_CORRELATION, CUDNN_DATA_FLOAT));
    CK(cudnnSetConvolutionMathType(p.cd, CUDNN_DEFAULT_MATH));
    CK(cudnnSetConvolutionGroupCount(p.cd, 1));

    cudnnHandle_t h = at::native::getCudnnHandle();

    const size_t WS_CAP = 64ull * 1024ull * 1024ull;
    bool found = false;
    int cnt = 0;
    std::vector<cudnnConvolutionFwdAlgoPerf_t> perf(CUDNN_CONVOLUTION_FWD_ALGO_COUNT);
    if (cudnnGetConvolutionForwardAlgorithm_v7(h, p.xd, p.fd, p.cd, p.od,
            CUDNN_CONVOLUTION_FWD_ALGO_COUNT, &cnt, perf.data()) == CUDNN_STATUS_SUCCESS) {
        for (int i = 0; i < cnt; ++i) {
            if (perf[i].status == CUDNN_STATUS_SUCCESS && perf[i].memory <= WS_CAP) {
                p.algo = perf[i].algo;
                p.ws   = perf[i].memory;
                found  = true;
                break;
            }
        }
    }
    if (!found) {
        p.algo = CUDNN_CONVOLUTION_FWD_ALGO_IMPLICIT_PRECOMP_GEMM;
        size_t sz = 0;
        if (cudnnGetConvolutionForwardWorkspaceSize(h, p.xd, p.fd, p.cd, p.od,
                                                    p.algo, &sz) != CUDNN_STATUS_SUCCESS) {
            sz = 0;
        }
        p.ws = sz;
    }

    // ------------------------------------------------------------------
    // NCHW / Winograd plan (plan items 1 & 8)
    // ------------------------------------------------------------------
    if (cudnnCreateTensorDescriptor(&p.xd_n) == CUDNN_STATUS_SUCCESS &&
        cudnnCreateTensorDescriptor(&p.od_n) == CUDNN_STATUS_SUCCESS &&
        cudnnCreateFilterDescriptor(&p.fd_n) == CUDNN_STATUS_SUCCESS &&
        cudnnCreateConvolutionDescriptor(&p.cd_n) == CUDNN_STATUS_SUCCESS) {

        bool ok = true;
        ok = ok && (cudnnSetTensor4dDescriptor(p.xd_n, CUDNN_TENSOR_NCHW, CUDNN_DATA_FLOAT,
                                               N, C, H, W) == CUDNN_STATUS_SUCCESS);
        ok = ok && (cudnnSetTensor4dDescriptor(p.od_n, CUDNN_TENSOR_NCHW, CUDNN_DATA_FLOAT,
                                               N, K, H, W) == CUDNN_STATUS_SUCCESS);
        ok = ok && (cudnnSetFilter4dDescriptor(p.fd_n, CUDNN_DATA_FLOAT, CUDNN_TENSOR_NCHW,
                                               K, C, 3, 3) == CUDNN_STATUS_SUCCESS);
        ok = ok && (cudnnSetConvolution2dDescriptor(p.cd_n, 1, 1, 1, 1, 1, 1,
                        CUDNN_CROSS_CORRELATION, CUDNN_DATA_FLOAT) == CUDNN_STATUS_SUCCESS);
        if (ok) {
            cudnnSetConvolutionMathType(p.cd_n, CUDNN_DEFAULT_MATH);
            cudnnSetConvolutionGroupCount(p.cd_n, 1);

            size_t freeB = 0, totalB = 0;
            if (cudaMemGetInfo(&freeB, &totalB) != cudaSuccess) { freeB = 0; }
            size_t cap = 512ull * 1024ull * 1024ull;
            if (freeB > 0) cap = std::min(cap, freeB / 4);   // <= 25% of free VRAM

            const cudnnConvolutionFwdAlgo_t cand[2] = {
                CUDNN_CONVOLUTION_FWD_ALGO_WINOGRAD,
                CUDNN_CONVOLUTION_FWD_ALGO_WINOGRAD_NONFUSED
            };
            for (int i = 0; i < 2; ++i) {
                size_t sz = 0;
                cudnnStatus_t st = cudnnGetConvolutionForwardWorkspaceSize(
                    h, p.xd_n, p.fd_n, p.cd_n, p.od_n, cand[i], &sz);
                if (st == CUDNN_STATUS_SUCCESS && sz <= cap) {
                    p.use_winograd = true;
                    p.algo_w = cand[i];
                    p.ws_w   = sz;
                    break;
                }
            }
        }
    }

    auto res = g_plans.emplace(key, p);
    return res.first->second;
}

std::vector<int64_t> conv3x3_plan(int64_t N, int64_t C, int64_t H, int64_t W, int64_t K) {
    ConvPlan& p = get_plan((int)N, (int)C, (int)H, (int)W, (int)K);
    return std::vector<int64_t>{(int64_t)p.algo, (int64_t)p.ws};
}

std::vector<int64_t> conv3x3_plan_nchw(int64_t N, int64_t C, int64_t H, int64_t W, int64_t K) {

    return std::vector<int64_t>{(int64_t)(p.use_winograd ? 1 : 0),
                                (int64_t)p.algo_w, (int64_t)p.ws_w};
}

void conv3x3_nhwc_cuda(at::Tensor x, at::Tensor w, at::Tensor out, at::Tensor ws) {

    TORCH_CHECK(w.is_cuda() && w.scalar_type() == at::kFloat, "fp32 cuda weight expected");
    TORCH_CHECK(out.is_cuda() && out.scalar_type() == at::kFloat, "fp32 cuda output expected");

#include <torch/extension.h>

#include <algorithm>

#define CCH 256
#define NGRP 32
#define CPG  8

__device__ __forceinline__ float silu_f(float t) {
    return t / (1.0f + __expf(-t));
}

// ------------------------- K0: NCHW -> NHWC transpose ----------------------
__global__ void to_nhwc_kernel(const float* __restrict__ x,
                               float* __restrict__ out,
                               int HW, int C) {
    __shared__ float sh[32][33];
    const int n  = blockIdx.z;
    const int cb = blockIdx.y * 32;
    const int pb = blockIdx.x * 32;
    const int tx = threadIdx.x;
    const int ty = threadIdx.y;

    const float* xb = x + (long)n * C * HW;
    #pragma unroll
    for (int i = 0; i < 4; ++i) {
        int c = cb + ty + i * 8;
        int p = pb + tx;
        float v = 0.0f;
        if (p < HW) v = xb[(long)c * HW + p];
        sh[ty + i * 8][tx] = v;
    }
    __syncthreads();

    float* ob = out + (long)n * (long)HW * C;
    #pragma unroll
    for (int i = 0; i < 4; ++i) {
        int p = pb + ty + i * 8;
        int c = cb + tx;
        if (p < HW) ob[(long)p * C + c] = sh[tx][ty + i * 8];
    }
}

// ------------------------- K1: NHWC group stats ----------------------------
__global__ void gn_stats_kernel(const float* __restrict__ y,
                                float* __restrict__ psum,
                                float* __restrict__ psq,
                                int HW, int tile, int ntiles) {
    const int n = blockIdx.y;
    const int t = blockIdx.x;
    const int c4 = threadIdx.x & 63;
    const int sp = threadIdx.x >> 6;

    const float4* y4 = (const float4*)(y + (long)n * (long)HW * CCH);

    int p0 = t * tile;
    int p1 = p0 + tile;
    if (p1 > HW) p1 = HW;

    float s = 0.0f, q = 0.0f;
    for (int p = p0 + sp; p < p1; p += 4) {
        float4 v = y4[(long)p * 64 + c4];
        s += v.x + v.y + v.z + v.w;
        q += v.x * v.x + v.y * v.y + v.z * v.z + v.w * v.w;
    }

    s += __shfl_xor_sync(0xffffffffu, s, 1);
    q += __shfl_xor_sync(0xffffffffu, q, 1);

    __shared__ float ss[4][32];
    __shared__ float sq[4][32];
    if ((threadIdx.x & 1) == 0) {
        int g = c4 >> 1;
        ss[sp][g] = s;
        sq[sp][g] = q;
    }
    __syncthreads();

    if (threadIdx.x < NGRP) {
        int g = threadIdx.x;
        float ts = ss[0][g] + ss[1][g] + ss[2][g] + ss[3][g];
        float tq = sq[0][g] + sq[1][g] + sq[2][g] + sq[3][g];
        long idx = ((long)n * NGRP + g) * ntiles + t;
        psum[idx] = ts;
        psq[idx]  = tq;
    }
}

// ---------------- K1b (NEW, plan item 3): NCHW group stats -----------------
__global__ void gn_stats_nchw_kernel(const float* __restrict__ x,
                                     float* __restrict__ psum,
                                     float* __restrict__ psq,
                                     int HW, int tile, int ntiles, int vec) {
    const int t  = blockIdx.x;
    const int ng = blockIdx.y;            // n*NGRP + g
    const int n  = ng / NGRP;
    const int g  = ng - n * NGRP;

    int p0 = t * tile;
    int p1 = p0 + tile;
    if (p1 > HW) p1 = HW;

    const float* base = x + ((long)n * CCH + (long)g * CPG) * (long)HW;

    float s = 0.0f, q = 0.0f;

    if (vec) {
        const int v0 = p0 >> 2;
        const int v1 = p1 >> 2;
        #pragma unroll
        for (int c = 0; c < CPG; ++c) {
            const float4* b4 = (const float4*)(base + (long)c * HW);
            for (int v = v0 + threadIdx.x; v < v1; v += blockDim.x) {
                float4 val = b4[v];
                s += val.x + val.y + val.z + val.w;
                q += val.x * val.x + val.y * val.y + val.z * val.z + val.w * val.w;
            }
        }
    } else {
        #pragma unroll
        for (int c = 0; c < CPG; ++c) {
            const float* b = base + (long)c * HW;
            for (int p = p0 + threadIdx.x; p < p1; p += blockDim.x) {
                float v = b[p];
                s += v;
                q += v * v;
            }
        }
    }

    #pragma unroll
    for (int off = 16; off > 0; off >>= 1) {
        s += __shfl_down_sync(0xffffffffu, s, off);
        q += __shfl_down_sync(0xffffffffu, q, off);
    }

    __shared__ float ws_[8], wq_[8];
    const int warp = threadIdx.x >> 5;
    const int lane = threadIdx.x & 31;
    if (lane == 0) { ws_[warp] = s; wq_[warp] = q; }
    __syncthreads();

    if (threadIdx.x == 0) {
        const int nw = blockDim.x >> 5;
        float ts = 0.0f, tq = 0.0f;
        for (int i = 0; i < nw; ++i) { ts += ws_[i]; tq += wq_[i]; }
        long idx = (long)ng * ntiles + t;
        psum[idx] = ts;
        psq[idx]  = tq;
    }
}

// ---------------- K2: reduce -> scale/shift (UNCHANGED, reused) -------------
__global__ void gn_reduce_kernel(const float* __restrict__ psum,
                                 const float* __restrict__ psq,
                                 const float* __restrict__ gamma,
                                 const float* __restrict__ beta,
                                 float* __restrict__ scale,
                                 float* __restrict__ shift,
                                 int ntiles, int HW, float eps) {
    const int ng = blockIdx.x;
    const int n  = ng / NGRP;
    const int g  = ng % NGRP;

    const float* pb_s = psum + (long)ng * ntiles;
    const float* pb_q = psq  + (long)ng * ntiles;

    float s = 0.0f, q = 0.0f;
    for (int i = threadIdx.x; i < ntiles; i += blockDim.x) {
        s += pb_s[i];
        q += pb_q[i];
    }
    #pragma unroll
    for (int off = 16; off > 0; off >>= 1) {
        s += __shfl_down_sync(0xffffffffu, s, off);
        q += __shfl_down_sync(0xffffffffu, q, off);
    }
    __shared__ float ws[8], wq[8];
    int warp = threadIdx.x >> 5;
    int lane = threadIdx.x & 31;
    if (lane == 0) { ws[warp] = s; wq[warp] = q; }
    __syncthreads();

    if (threadIdx.x < CPG) {
        int nwarps = blockDim.x >> 5;
        float ts = 0.0f, tq = 0.0f;
        for (int i = 0; i < nwarps; ++i) { ts += ws[i]; tq += wq[i]; }
        float cnt  = (float)HW * (float)CPG;
        float mean = ts / cnt;
        float var  = tq / cnt - mean * mean;
        if (var < 0.0f) var = 0.0f;
        float rstd = rsqrtf(var + eps);
        int c = g * CPG + (int)threadIdx.x;
        float gm = gamma[c];
        float bt = beta[c];
        scale[n * CCH + c] = rstd * gm;
        shift[n * CCH + c] = bt - mean * rstd * gm;
    }
}

// ---------------- K3: NHWC GN affine + SiLU --------------------------------
__global__ void gn_apply_silu_nhwc_kernel(const float* __restrict__ y,
                                          const float* __restrict__ scale,
                                          const float* __restrict__ shift,
                                          float* __restrict__ out,
                                          int HW) {
    const int n = blockIdx.y;
    long i = (long)blockIdx.x * blockDim.x + threadIdx.x;
    long total = (long)HW * 64;
    if (i >= total) return;

    const float4* y4 = (const float4*)(y   + (long)n * (long)HW * CCH);
    float4*       o4 = (float4*)      (out + (long)n * (long)HW * CCH);
    const float4* s4 = (const float4*)(scale + (long)n * CCH);
    const float4* b4 = (const float4*)(shift + (long)n * CCH);

    int c4 = (int)(i & 63);
    float4 v = y4[i];
    float4 s = s4[c4];
    float4 b = b4[c4];
    float4 r;
    r.x = silu_f(v.x * s.x + b.x);
    r.y = silu_f(v.y * s.y + b.y);
    r.z = silu_f(v.z * s.z + b.z);
    r.w = silu_f(v.w * s.w + b.w);
    o4[i] = r;
}

// ---------------- K4: NHWC GN+SiLU+add+transpose ---------------------------
__global__ void gn_apply_silu_add_t_kernel(const float* __restrict__ y,
                                           const float* __restrict__ res,
                                           const float* __restrict__ scale,
                                           const float* __restrict__ shift,
                                           float* __restrict__ out,
                                           int HW) {
    __shared__ float sh[32][33];
    const int n  = blockIdx.z;
    const int cb = blockIdx.y * 32;
    const int pb = blockIdx.x * 32;
    const int tx = threadIdx.x;
    const int ty = threadIdx.y;

    const float* yb = y + (long)n * (long)HW * CCH;
    const int c_in = cb + tx;
    const float sc = scale[n * CCH + c_in];
    const float sf = shift[n * CCH + c_in];

    #pragma unroll
    for (int i = 0; i < 4; ++i) {
        int p = pb + ty + i * 8;
        float o = 0.0f;
        if (p < HW) {
            float v = yb[(long)p * CCH + c_in];
            o = silu_f(v * sc + sf);
        }
        sh[ty + i * 8][tx] = o;
    }
    __syncthreads();

    const float* rb = res + (long)n * CCH * (long)HW;
    float*       ob = out + (long)n * CCH * (long)HW;
    #pragma unroll
    for (int i = 0; i < 4; ++i) {
        int c = cb + ty + i * 8;
        int p = pb + tx;
        if (p < HW) {
            ob[(long)c * HW + p] = sh[tx][ty + i * 8] + rb[(long)c * HW + p];
        }
    }
}

static void compute_affine(const at::Tensor& y, const at::Tensor& gamma,
                           const at::Tensor& beta, int N, int HW, float eps,
                           at::Tensor& scale, at::Tensor& shift) {
    int bpi = (HW + 15) / 16;
    int cap = std::max(1, 8192 / std::max(1, N));
    if (bpi > cap) bpi = cap;
    int tile = (HW + bpi - 1) / bpi;
    tile = ((tile + 3) / 4) * 4;
    if (tile < 4) tile = 4;
    bpi = (HW + tile - 1) / tile;

    auto opts = y.options();
    auto psum = at::empty({(long)N * NGRP * bpi}, opts);
    auto psq  = at::empty({(long)N * NGRP * bpi}, opts);

    auto stream = at::cuda::getCurrentCUDAStream();

    gn_stats_kernel<<<dim3(bpi, N), 256, 0, stream>>>(
        y.data_ptr<float>(), psum.data_ptr<float>(), psq.data_ptr<float>(),
        HW, tile, bpi);
    C10_CUDA_KERNEL_LAUNCH_CHECK();

    gn_reduce_kernel<<<N * NGRP, 256, 0, stream>>>(
        psum.data_ptr<float>(), psq.data_ptr<float>(),
        gamma.data_ptr<float>(), beta.data_ptr<float>(),
        scale.data_ptr<float>(), shift.data_ptr<float>(),
        bpi, HW, eps);
    C10_CUDA_KERNEL_LAUNCH_CHECK();
}

static void compute_affine_nchw(const at::Tensor& y, const at::Tensor& gamma,
                                const at::Tensor& beta, int N, int HW, float eps,
                                at::Tensor& scale, at::Tensor& shift) {
    int bpi = (HW + 2047) / 2048;
    int cap = std::max(1, 4096 / std::max(1, N));
    if (bpi > cap) bpi = cap;
    if (bpi < 1) bpi = 1;
    int tile = (HW + bpi - 1) / bpi;
    tile = ((tile + 3) / 4) * 4;
    if (tile < 4) tile = 4;
    bpi = (HW + tile - 1) / tile;

    const int vec = (HW % 4 == 0) ? 1 : 0;

    auto opts = y.options();
    auto psum = at::empty({(long)N * NGRP * bpi}, opts);
    auto psq  = at::empty({(long)N * NGRP * bpi}, opts);

    auto stream = at::cuda::getCurrentCUDAStream();

    gn_stats_nchw_kernel<<<dim3(bpi, N * NGRP), 256, 0, stream>>>(
        y.data_ptr<float>(), psum.data_ptr<float>(), psq.data_ptr<float>(),
        HW, tile, bpi, vec);
    C10_CUDA_KERNEL_LAUNCH_CHECK();

    gn_reduce_kernel<<<N * NGRP, 256, 0, stream>>>(
        psum.data_ptr<float>(), psq.data_ptr<float>(),
        gamma.data_ptr<float>(), beta.data_ptr<float>(),
        scale.data_ptr<float>(), shift.data_ptr<float>(),
        bpi, HW, eps);
    C10_CUDA_KERNEL_LAUNCH_CHECK();
}

static inline int grid_for(long n, int block) {
    long b = (n + block - 1) / block;
    if (b < 1) b = 1;
    if (b > 262144) b = 262144;
    return (int)b;
}

at::Tensor to_nhwc_cuda(at::Tensor x) {
    TORCH_CHECK(x.is_cuda() && x.scalar_type() == at::kFloat, "float32 cuda tensor expected");
    TORCH_CHECK(x.dim() == 4, "4d tensor expected");
    int N = (int)x.size(0), C = (int)x.size(1);
    int H = (int)x.size(2), W = (int)x.size(3);
    TORCH_CHECK(C % 32 == 0, "channels must be a multiple of 32");
    int HW = H * W;
    auto xc = x.is_contiguous() ? x : x.contiguous();
    auto out = at::empty_like(xc, xc.options(), at::MemoryFormat::ChannelsLast);

static void gn_silu_nchw_core(at::Tensor& y, at::Tensor& gamma, at::Tensor& beta,

    auto stream = at::cuda::getCurrentCUDAStream();
    const long total = (long)N * CCH * (long)HW;
    if (HW % 4 == 0) {

        int blocks = grid_for(total4, 256);
        gn_apply_silu_nchw_v4_kernel<<<blocks, 256, 0, stream>>>(
            (const float4*)y.data_ptr<float>(),

            y.data_ptr<float>(), scale.data_ptr<float>(), shift.data_ptr<float>(),
            out.data_ptr<float>(), HW, total);

_CONV_CUDA_SRC = r"""
#include <torch/extension.h>
#include <ATen/cuda/CUDAContext.h>
#include <ATen/cudnn/Handle.h>
#include <cudnn.h>
#include <cuda_runtime.h>
#include <map>
#include <tuple>
#include <vector>
#include <algorithm>

#define CK(expr) TORCH_CHECK((expr) == CUDNN_STATUS_SUCCESS, "cudnn call failed: ", #expr)

struct ConvPlan {
    cudnnTensorDescriptor_t      xd = nullptr;
    cudnnTensorDescriptor_t      od = nullptr;
    cudnnFilterDescriptor_t      fd = nullptr;
    cudnnConvolutionDescriptor_t cd = nullptr;
    cudnnConvolutionFwdAlgo_t    algo = CUDNN_CONVOLUTION_FWD_ALGO_IMPLICIT_PRECOMP_GEMM;
    size_t                       ws = 0;

    // NCHW Winograd (plan item 1)
    cudnnTensorDescriptor_t      xd_n = nullptr;
    cudnnTensorDescriptor_t      od_n = nullptr;
    cudnnFilterDescriptor_t      fd_n = nullptr;
    cudnnConvolutionDescriptor_t cd_n = nullptr;
    bool                         use_winograd = false;
    cudnnConvolutionFwdAlgo_t    algo_w = CUDNN_CONVOLUTION_FWD_ALGO_WINOGRAD;
    size_t                       ws_w = 0;
};

using ck_t = std::tuple<int,int,int,int,int>;
static std::map<ck_t, ConvPlan> g_plans;

static ConvPlan& get_plan(int N, int C, int H, int W, int K) {
    ck_t key(N, C, H, W, K);
    auto it = g_plans.find(key);
    if (it != g_plans.end()) return it->second;

    ConvPlan p;
    CK(cudnnCreateTensorDescriptor(&p.xd));
    CK(cudnnSetTensor4dDescriptor(p.xd, CUDNN_TENSOR_NHWC, CUDNN_DATA_FLOAT, N, C, H, W));
    CK(cudnnCreateTensorDescriptor(&p.od));
    CK(cudnnSetTensor4dDescriptor(p.od, CUDNN_TENSOR_NHWC, CUDNN_DATA_FLOAT, N, K, H, W));
    CK(cudnnCreateFilterDescriptor(&p.fd));
    CK(cudnnSetFilter4dDescriptor(p.fd, CUDNN_DATA_FLOAT, CUDNN_TENSOR_NHWC, K, C, 3, 3));
    CK(cudnnCreateConvolutionDescriptor(&p.cd));
    CK(cudnnSetConvolution2dDescriptor(p.cd, 1, 1, 1, 1, 1, 1,
                                       CUDNN_CROSS_CORRELATION, CUDNN_DATA_FLOAT));
    CK(cudnnSetConvolutionMathType(p.cd, CUDNN_DEFAULT_MATH));
    CK(cudnnSetConvolutionGroupCount(p.cd, 1));

    cudnnHandle_t h = at::native::getCudnnHandle();

    const size_t WS_CAP = 64ull * 1024ull * 1024ull;
    bool found = false;
    int cnt = 0;
    std::vector<cudnnConvolutionFwdAlgoPerf_t> perf(CUDNN_CONVOLUTION_FWD_ALGO_COUNT);
    if (cudnnGetConvolutionForwardAlgorithm_v7(h, p.xd, p.fd, p.cd, p.od,
            CUDNN_CONVOLUTION_FWD_ALGO_COUNT, &cnt, perf.data()) == CUDNN_STATUS_SUCCESS) {
        for (int i = 0; i < cnt; ++i) {
            if (perf[i].status == CUDNN_STATUS_SUCCESS && perf[i].memory <= WS_CAP) {
                p.algo = perf[i].algo;
                p.ws   = perf[i].memory;
                found  = true;
                break;
            }
        }
    }
    if (!found) {
        p.algo = CUDNN_CONVOLUTION_FWD_ALGO_IMPLICIT_PRECOMP_GEMM;
        size_t sz = 0;
        if (cudnnGetConvolutionForwardWorkspaceSize(h, p.xd, p.fd, p.cd, p.od,
                                                    p.algo, &sz) != CUDNN_STATUS_SUCCESS) {
            sz = 0;
        }
        p.ws = sz;
    }

    // ---------------- NCHW / Winograd plan (plan items 1 & 8) --------------
    if (cudnnCreateTensorDescriptor(&p.xd_n) == CUDNN_STATUS_SUCCESS &&
        cudnnCreateTensorDescriptor(&p.od_n) == CUDNN_STATUS_SUCCESS &&
        cudnnCreateFilterDescriptor(&p.fd_n) == CUDNN_STATUS_SUCCESS &&
        cudnnCreateConvolutionDescriptor(&p.cd_n) == CUDNN_STATUS_SUCCESS) {

        bool ok = true;
        ok = ok && (cudnnSetTensor4dDescriptor(p.xd_n, CUDNN_TENSOR_NCHW, CUDNN_DATA_FLOAT,
                                               N, C, H, W) == CUDNN_STATUS_SUCCESS);
        ok = ok && (cudnnSetTensor4dDescriptor(p.od_n, CUDNN_TENSOR_NCHW, CUDNN_DATA_FLOAT,
                                               N, K, H, W) == CUDNN_STATUS_SUCCESS);
        ok = ok && (cudnnSetFilter4dDescriptor(p.fd_n, CUDNN_DATA_FLOAT, CUDNN_TENSOR_NCHW,
                                               K, C, 3, 3) == CUDNN_STATUS_SUCCESS);
        ok = ok && (cudnnSetConvolution2dDescriptor(p.cd_n, 1, 1, 1, 1, 1, 1,
                        CUDNN_CROSS_CORRELATION, CUDNN_DATA_FLOAT) == CUDNN_STATUS_SUCCESS);
        if (ok) {
            cudnnSetConvolutionMathType(p.cd_n, CUDNN_DEFAULT_MATH);
            cudnnSetConvolutionGroupCount(p.cd_n, 1);

            size_t freeB = 0, totalB = 0;
            if (cudaMemGetInfo(&freeB, &totalB) != cudaSuccess) { freeB = 0; }
            size_t cap = 512ull * 1024ull * 1024ull;
            if (freeB > 0) cap = std::min(cap, freeB / 4);   // <= 25% of free VRAM

            const cudnnConvolutionFwdAlgo_t cand[2] = {
                CUDNN_CONVOLUTION_FWD_ALGO_WINOGRAD,
                CUDNN_CONVOLUTION_FWD_ALGO_WINOGRAD_NONFUSED
            };
            for (int i = 0; i < 2; ++i) {
                size_t sz = 0;
                cudnnStatus_t st = cudnnGetConvolutionForwardWorkspaceSize(
                    h, p.xd_n, p.fd_n, p.cd_n, p.od_n, cand[i], &sz);
                if (st == CUDNN_STATUS_SUCCESS && sz <= cap) {
                    p.use_winograd = true;
                    p.algo_w = cand[i];
                    p.ws_w   = sz;
                    break;
                }
            }
        }
    }

    auto res = g_plans.emplace(key, p);
    return res.first->second;
}

std::vector<int64_t> conv3x3_plan(int64_t N, int64_t C, int64_t H, int64_t W, int64_t K) {
    ConvPlan& p = get_plan((int)N, (int)C, (int)H, (int)W, (int)K);
    return std::vector<int64_t>{(int64_t)p.algo, (int64_t)p.ws};
}

std::vector<int64_t> conv3x3_plan_nchw(int64_t N, int64_t C, int64_t H, int64_t W, int64_t K) {

    return std::vector<int64_t>{(int64_t)(p.use_winograd ? 1 : 0),
                                (int64_t)p.algo_w, (int64_t)p.ws_w};
}

void conv3x3_nhwc_cuda(at::Tensor x, at::Tensor w, at::Tensor out, at::Tensor ws) {
    TORCH_CHECK(x.is_cuda() && x.scalar_type() == at::kFloat, "fp32 cuda input expected");
    TORCH_CHECK(w.is_cuda() && w.scalar_type() == at::kFloat, "fp32 cuda weight expected");
    TORCH_CHECK(out.is_cuda() && out.scalar_type() == at::kFloat, "fp32 cuda output expected");
    TORCH_CHECK(x.dim() == 4 && w.dim() == 4 && out.dim() == 4, "4d tensors expected");
    TORCH_CHECK(x.is_contiguous(at::MemoryFormat::ChannelsLast), "x must be channels_last");
    TORCH_CHECK(w.is_contiguous(at::MemoryFormat::ChannelsLast), "w must be channels_last");
    TORCH_CHECK(out.is_contiguous(at::MemoryFormat::ChannelsLast), "out must be channels_last");
    TORCH_CHECK(w.size(2) == 3 && w.size(3) == 3, "3x3 filter expected");

    const int N = (int)x.size(0), C = (int)x.size(1);
    const int H = (int)x.size(2), W_ = (int)x.size(3);
    const int K = (int)w.size(0);
    TORCH_CHECK(w.size(1) == C, "weight channel mismatch");
    TORCH_CHECK(out.size(0) == N && out.size(1) == K && out.size(2) == H && out.size(3) == W_,
                "output shape mismatch");

    ConvPlan& p = get_plan(N, C, H, W_, K);

    const size_t ws_bytes = (size_t)ws.numel() * (size_t)ws.element_size();
    TORCH_CHECK(ws_bytes >= p.ws, "workspace too small");

    cudnnHandle_t h = at::native::getCudnnHandle();

    const float one = 1.0f, zero = 0.0f;
    cudnnStatus_t st = cudnnConvolutionForward(
        h, &one,
        p.xd, x.data_ptr(),
        p.fd, w.data_ptr(),
        p.cd, p.algo,
        (ws_bytes > 0 ? ws.data_ptr() : nullptr), ws_bytes,
        &zero,
        p.od, out.data_ptr());
    TORCH_CHECK(st == CUDNN_STATUS_SUCCESS, "cudnnConvolutionForward failed: ", (int)st);
}

def _is_capturing():
    try:
        return torch.cuda.is_current_stream_capturing()
    except Exception:
        return False


class ModelNew(nn.Module):
    """Fused VAE res-block + CUDA-graph capture + 2-way batch-split streams
       + NCHW Winograd conv path selected by a per-shape warmup gate."""

    _MAX_GRAPHS = 8
    _MIN_N_FOR_SPLIT = 4

    def __init__(self):
        super().__init__()
        self.ext = _ext
        try:
            torch.backends.cudnn.benchmark = True
        except Exception:
            pass

        self._graphs = OrderedDict()

        # NHWC implicit-GEMM state (round-3 path, preserved)
        self._conv_ext = _CONV_EXT
        self._v7_ok = _CONV_EXT is not None
        self._v7_validated = False
        self._conv_state = {}

        # NCHW Winograd state (new)
        self._wino_ok = _CONV_EXT is not None
        self._wino_validated = False
        self._conv_state_nchw = {}
        self._impl_choice = {}

        self._streams = None
        try:
            if torch.cuda.is_available():
                self._streams = [torch.cuda.Stream(), torch.cuda.Stream()]
        except Exception:
            self._streams = None

    # ---------------- NHWC conv state / issue point (unchanged) -------------
    def _make_conv_state(self, x, w, idx, slot):
        N, C, H, W_ = int(x.size(0)), int(x.size(1)), int(x.size(2)), int(x.size(3))
        K = int(w.size(0))
        algo, ws_bytes = self._conv_ext.conv3x3_plan(N, C, H, W_, K)
        out = torch.empty((N, K, H, W_), dtype=x.dtype, device=x.device,
                          memory_format=torch.channels_last)
        ws = torch.empty(max(int(ws_bytes), 1), dtype=torch.uint8, device=x.device)

        if not self._v7_validated:
            ref = F.conv2d(x, w, None, 1, 1)
            self._conv_ext.conv3x3_nhwc_cuda(x, w, out, ws)
            if not torch.allclose(out, ref.contiguous(memory_format=torch.channels_last),
                                  atol=1e-3, rtol=1e-3):
                raise RuntimeError("cudnn v7 conv numeric mismatch")
            self._v7_validated = True

        self._conv_state[(slot, idx, N, C, H, W_, K)] = (out, ws)
        return out, ws

    def _conv(self, x, w, idx, slot=0):
        if not self._v7_ok:
            return F.conv2d(x, w, None, 1, 1)

        key = (slot, idx, int(x.size(0)), int(x.size(1)), int(x.size(2)),
               int(x.size(3)), int(w.size(0)))
        try:
            st = self._conv_state.get(key)
            if st is None:
                if _is_capturing():
                    return F.conv2d(x, w, None, 1, 1)
                st = self._make_conv_state(x, w, idx, slot)
            out, ws = st
            self._conv_ext.conv3x3_nhwc_cuda(x, w, out, ws)
            return out
        except Exception:
            self._v7_ok = False
            self._conv_state.clear()
            return F.conv2d(x, w, None, 1, 1)

    # ------------- NCHW Winograd state / issue point (plan 2,6,7,8) ---------
    def _make_conv_state_nchw(self, x, w, idx, slot):
        if self._conv_ext is None:
            raise RuntimeError("no conv extension")
        N, C, H, W_ = int(x.size(0)), int(x.size(1)), int(x.size(2)), int(x.size(3))
        K = int(w.size(0))
        use_w, algo, ws_bytes = self._conv_ext.conv3x3_plan_nchw(N, C, H, W_, K)
        if int(use_w) == 0:
            raise RuntimeError("winograd unavailable / workspace over cap")

        out = torch.empty((N, K, H, W_), dtype=x.dtype, device=x.device)
        ws = torch.empty(max(int(ws_bytes), 1), dtype=torch.uint8, device=x.device)

        if not self._wino_validated:
            ref = F.conv2d(x, w, None, 1, 1)
            self._conv_ext.conv3x3_nchw_cuda(x, w, out, ws)
            if not torch.allclose(out, ref.contiguous(), atol=1e-3, rtol=1e-3):
                raise RuntimeError("winograd conv numeric mismatch")
            self._wino_validated = True

        self._conv_state_nchw[(slot, idx, N, C, H, W_, K)] = (out, ws)
        return out, ws

    def _conv_nchw(self, x, w, idx, slot=0):
        if not self._wino_ok or self._conv_ext is None:
            raise RuntimeError("winograd path disabled")
        key = (slot, idx, int(x.size(0)), int(x.size(1)), int(x.size(2)),
               int(x.size(3)), int(w.size(0)))
        st = self._conv_state_nchw.get(key)
        if st is None:
            if _is_capturing():
                return F.conv2d(x, w, None, 1, 1)
            st = self._make_conv_state_nchw(x, w, idx, slot)
        out, ws = st
        self._conv_ext.conv3x3_nchw_cuda(x, w, out, ws)
        return out

    # ---------------- NHWC single-stream body (unchanged) -------------------
    def _forward_impl(self, x, conv1_weight, norm1_weight, norm1_bias,
                      conv2_weight, norm2_weight, norm2_bias, eps_f):
        x_c = x if x.is_contiguous() else x.contiguous()

        x_nhwc = self.ext.to_nhwc_cuda(x_c)

        w1 = conv1_weight if conv1_weight.is_contiguous(memory_format=torch.channels_last) \
            else conv1_weight.contiguous(memory_format=torch.channels_last)
        w2 = conv2_weight if conv2_weight.is_contiguous(memory_format=torch.channels_last) \
            else conv2_weight.contiguous(memory_format=torch.channels_last)

        g1 = norm1_weight if norm1_weight.is_contiguous() else norm1_weight.contiguous()
        b1 = norm1_bias if norm1_bias.is_contiguous() else norm1_bias.contiguous()
        g2 = norm2_weight if norm2_weight.is_contiguous() else norm2_weight.contiguous()
        b2 = norm2_bias if norm2_bias.is_contiguous() else norm2_bias.contiguous()

        y1 = self._conv(x_nhwc, w1, 0, 0)
        if not y1.is_contiguous(memory_format=torch.channels_last):
            y1 = y1.contiguous(memory_format=torch.channels_last)

        z = self.ext.gn_silu_nhwc_cuda(y1, g1, b1, eps_f)

        y2 = self._conv(z, w2, 1, 0)
        if not y2.is_contiguous(memory_format=torch.channels_last):
            y2 = y2.contiguous(memory_format=torch.channels_last)

        out = self.ext.gn_silu_add_nchw_cuda(y2, x_c, g2, b2, eps_f)
        return out

    # ---------------- NHWC 2-way batch split (unchanged) --------------------
    def _forward_impl_multi(self, x, conv1_weight, norm1_weight, norm1_bias,
                            conv2_weight, norm2_weight, norm2_bias, eps_f):
        x_c = x if x.is_contiguous() else x.contiguous()
        N = int(x_c.size(0)); C = int(x_c.size(1))
        H = int(x_c.size(2)); W = int(x_c.size(3))

        c0 = N // 2
        bounds = ((0, c0), (c0, N))

        w1 = conv1_weight if conv1_weight.is_contiguous(memory_format=torch.channels_last) \
            else conv1_weight.contiguous(memory_format=torch.channels_last)
        w2 = conv2_weight if conv2_weight.is_contiguous(memory_format=torch.channels_last) \
            else conv2_weight.contiguous(memory_format=torch.channels_last)

        g1 = norm1_weight if norm1_weight.is_contiguous() else norm1_weight.contiguous()
        b1 = norm1_bias if norm1_bias.is_contiguous() else norm1_bias.contiguous()
        g2 = norm2_weight if norm2_weight.is_contiguous() else norm2_weight.contiguous()
        b2 = norm2_bias if norm2_bias.is_contiguous() else norm2_bias.contiguous()

        out = torch.empty((N, C, H, W), dtype=x_c.dtype, device=x_c.device)

        cur = torch.cuda.current_stream()
        capturing = _is_capturing()

        ev0 = torch.cuda.Event()
        ev0.record(cur)

        end_events = []
        for k, (n0, n1) in enumerate(bounds):
            s = self._streams[k]
            s.wait_event(ev0)
            with torch.cuda.stream(s):
                xk = x_c[n0:n1]
                outk = out[n0:n1]
                if not capturing:
                    try:
                        out.record_stream(s)
                    except Exception:
                        pass

                xk_nhwc = self.ext.to_nhwc_cuda(xk)

                y1 = self._conv(xk_nhwc, w1, 0, k)
                if not y1.is_contiguous(memory_format=torch.channels_last):
                    y1 = y1.contiguous(memory_format=torch.channels_last)

                z = self.ext.gn_silu_nhwc_cuda(y1, g1, b1, eps_f)

                y2 = self._conv(z, w2, 1, k)
                if not y2.is_contiguous(memory_format=torch.channels_last):
                    y2 = y2.contiguous(memory_format=torch.channels_last)

                self.ext.gn_silu_add_nchw_out_cuda(y2, xk, g2, b2, eps_f, outk)

                ev = torch.cuda.Event()
                ev.record(s)
                end_events.append(ev)

        for ev in end_events:
            cur.wait_event(ev)

        return out

    # ---------------- NCHW Winograd body (plan item 4) ----------------------
    def _forward_impl_nchw(self, x, conv1_weight, norm1_weight, norm1_bias,
                           conv2_weight, norm2_weight, norm2_bias, eps_f):
        x_c = x if x.is_contiguous() else x.contiguous()

        w1 = conv1_weight if conv1_weight.is_contiguous() else conv1_weight.contiguous()
        w2 = conv2_weight if conv2_weight.is_contiguous() else conv2_weight.contiguous()

        g1 = norm1_weight if norm1_weight.is_contiguous() else norm1_weight.contiguous()
        b1 = norm1_bias if norm1_bias.is_contiguous() else norm1_bias.contiguous()
        g2 = norm2_weight if norm2_weight.is_contiguous() else norm2_weight.contiguous()
        b2 = norm2_bias if norm2_bias.is_contiguous() else norm2_bias.contiguous()

        y1 = self._conv_nchw(x_c, w1, 0, 0)
        if not y1.is_contiguous():
            y1 = y1.contiguous()

        z = self.ext.gn_silu_nchw_cuda(y1, g1, b1, eps_f)

        y2 = self._conv_nchw(z, w2, 1, 0)
        if not y2.is_contiguous():
            y2 = y2.contiguous()

        out = self.ext.gn_silu_add_nchw_nn_cuda(y2, x_c, g2, b2, eps_f)
        return out

    # ---------------- NCHW Winograd 2-way batch split (plan items 4 & 7) ----
    def _forward_impl_nchw_multi(self, x, conv1_weight, norm1_weight, norm1_bias,
                                 conv2_weight, norm2_weight, norm2_bias, eps_f):
        x_c = x if x.is_contiguous() else x.contiguous()
        N = int(x_c.size(0)); C = int(x_c.size(1))
        H = int(x_c.size(2)); W = int(x_c.size(3))

        c0 = N // 2
        bounds = ((0, c0), (c0, N))

        w1 = conv1_weight if conv1_weight.is_contiguous() else conv1_weight.contiguous()
        w2 = conv2_weight if conv2_weight.is_contiguous() else conv2_weight.contiguous()

        g1 = norm1_weight if norm1_weight.is_contiguous() else norm1_weight.contiguous()
        b1 = norm1_bias if norm1_bias.is_contiguous() else norm1_bias.contiguous()
        g2 = norm2_weight if norm2_weight.is_contiguous() else norm2_weight.contiguous()
        b2 = norm2_bias if norm2_bias.is_contiguous() else norm2_bias.contiguous()

        out = torch.empty((N, C, H, W), dtype=x_c.dtype, device=x_c.device)

        cur = torch.cuda.current_stream()
        capturing = _is_capturing()

        ev0 = torch.cuda.Event()
        ev0.record(cur)

        end_events = []
        for k, (n0, n1) in enumerate(bounds):
            s = self._streams[k]
            s.wait_event(ev0)
            with torch.cuda.stream(s):
                xk = x_c[n0:n1]
                outk = out[n0:n1]
                if not capturing:
                    try:
                        out.record_stream(s)
                    except Exception:
                        pass

                y1 = self._conv_nchw(xk, w1, 0, k)
                if not y1.is_contiguous():
                    y1 = y1.contiguous()

                z = self.ext.gn_silu_nchw_cuda(y1, g1, b1, eps_f)

                y2 = self._conv_nchw(z, w2, 1, k)
                if not y2.is_contiguous():
                    y2 = y2.contiguous()

                self.ext.gn_silu_add_nchw_nn_out_cuda(y2, xk, g2, b2, eps_f, outk)

                ev = torch.cuda.Event()
                ev.record(s)
                end_events.append(ev)

        for ev in end_events:
            cur.wait_event(ev)

        return out

    def _use_multi(self, x):
        return (self._streams is not None) and (int(x.size(0)) >= self._MIN_N_FOR_SPLIT)

    # ---------------- per-shape warmup gate (plan item 5) -------------------
    def _time_fn(self, fn, args, iters=5):
        for _ in range(2):
            fn(*args)
        torch.cuda.synchronize()
        st = torch.cuda.Event(enable_timing=True)
        en = torch.cuda.Event(enable_timing=True)
        st.record()
        for _ in range(iters):
            fn(*args)
        en.record()
        torch.cuda.synchronize()
        return st.elapsed_time(en) / float(iters)

    def _select_impl(self, args, use_multi):
        x = args[0]
        shape_key = (tuple(x.shape), bool(use_multi))
        ch = self._impl_choice.get(shape_key)
        if ch is not None:
            return ch

        if not self._wino_ok:
            self._impl_choice[shape_key] = 'nhwc'
            return 'nhwc'

        nchw_fn = self._forward_impl_nchw_multi if use_multi else self._forward_impl_nchw
        nhwc_fn = self._forward_impl_multi if use_multi else self._forward_impl

        try:
            t_w = self._time_fn(nchw_fn, args)
        except Exception:
            # plan item 6: NOT_SUPPORTED / mismatch -> permanent fallback
            self._wino_ok = False
            self._conv_state_nchw.clear()
            try:
                torch.cuda.synchronize()
            except Exception:
                pass
            self._impl_choice[shape_key] = 'nhwc'
            return 'nhwc'

        try:
            t_n = self._time_fn(nhwc_fn, args)
        except Exception:
            t_n = float('inf')

        ch = 'nchw' if t_w < t_n else 'nhwc'
        self._impl_choice[shape_key] = ch
        return ch

    # ---------------- warmup + capture (unchanged) -------------------------
    def _warmup_and_capture(self, args, fn):
        s = torch.cuda.Stream()
        s.wait_stream(torch.cuda.current_stream())
        with torch.cuda.stream(s):
            tmp = None
            for _ in range(3):
                tmp = fn(*args)
            del tmp
        torch.cuda.current_stream().wait_stream(s)

        g = torch.cuda.CUDAGraph()
        with torch.cuda.graph(g):
            out = fn(*args)
        return g, out

    # ---------------------------- Dispatcher -------------------------------
    def forward(self, x, conv1_weight, norm1_weight, norm1_bias,
                conv2_weight, norm2_weight, norm2_bias, eps):
        eps_f = float(eps)
        assert x.dtype == torch.float32, "this kernel supports float32 only"

        args = (x, conv1_weight, norm1_weight, norm1_bias,
                conv2_weight, norm2_weight, norm2_bias, eps_f)

        if (not x.is_cuda) or (x.dtype != torch.float32) or _is_capturing():
            return self._forward_impl(*args)

        key = (tuple(x.shape),
               x.data_ptr(),
               conv1_weight.data_ptr(),
               conv2_weight.data_ptr(),
               norm1_weight.data_ptr(),
               norm1_bias.data_ptr(),
               norm2_weight.data_ptr(),
               norm2_bias.data_ptr(),
               eps_f)

        if key in self._graphs:
            entry = self._graphs[key]
            if entry is None:
                return self._forward_impl(*args)
            self._graphs.move_to_end(key)
            g, out = entry
            g.replay()
            return out

        while len(self._graphs) >= self._MAX_GRAPHS:
            old_key, old_entry = self._graphs.popitem(last=False)
            del old_key, old_entry

        use_multi = self._use_multi(x)
        choice = self._select_impl(args, use_multi)

        candidates = []
        if choice == 'nchw':
            candidates.append(self._forward_impl_nchw_multi if use_multi
                              else self._forward_impl_nchw)
        if use_multi:
            candidates.append(self._forward_impl_multi)
        candidates.append(self._forward_impl)

        for fn in candidates:
            try:
                g, out = self._warmup_and_capture(args, fn)
                g.replay()
                self._graphs[key] = (g, out)
                return out
            except Exception:
                try:
                    torch.cuda.synchronize()
                except Exception:
                    pass
                continue

        if self._v7_ok:
            self._v7_ok = False
            self._conv_state.clear()
            try:
                g, out = self._warmup_and_capture(args, self._forward_impl)
                g.replay()
                self._graphs[key] = (g, out)
                return out
            except Exception:
                pass

        self._graphs[key] = None
        return self._forward_impl(*args)
