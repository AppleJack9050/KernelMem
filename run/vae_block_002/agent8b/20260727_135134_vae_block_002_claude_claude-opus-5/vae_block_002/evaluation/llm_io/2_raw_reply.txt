Plan item 2: Both convs run as cuDNN FP16 NHWC tensor-core implicit GEMM.

Plan item 4: Pass-1 apply kernel half NHWC→half NHWC with per-channel scale/bias in shared memory, float4 vectorized.

Plan item 5: Pass-2 kernel with shared half tile (Phase A NHWC read/SiLU→smem, Phase B channel-major residual add → fp32 NCHW output).

Implemented in code: `ModelNew.forward` signature unchanged, only `contiguous()`/dtype casts, outputs via fresh `torch::empty`.

from torch.utils.cpp_extension import load_inline

_CUDA_SRC = r"""
#include <torch/extension.h>
#include <cuda_runtime.h>
#include <cuda_fp16.h>
#include <ATen/cuda/CUDAContext.h>
#include <c10/cuda/CUDAException.h>
#include <algorithm>

#define CDIV(a, b) (((a) + (b) - 1) / (b))

__device__ __forceinline__ float warp_sum(float v) {
#pragma unroll
    for (int o = 16; o > 0; o >>= 1) v += __shfl_xor_sync(0xffffffffu, v, o);
    return v;
}

// ===========================================================================
//  FP32 NCHW PATH (fallback, unchanged from base kernel)
// ===========================================================================
template <bool VEC>
__global__ void gn_stats_partial_kernel(const float* __restrict__ y,
                                        float* __restrict__ partial,
                                        long long group_size,
                                        int splits) {
    const long long g = blockIdx.y;
    const int s = blockIdx.x;
    const float* __restrict__ base = y + g * group_size;

    float sum = 0.f, sq = 0.f;

    if (VEC) {
        const long long nvec = group_size >> 2;
        const long long start = ((long long)s * nvec) / splits;
        const long long end = ((long long)(s + 1) * nvec) / splits;
        const float4* __restrict__ b4 = reinterpret_cast<const float4*>(base);
        for (long long i = start + threadIdx.x; i < end; i += blockDim.x) {
            float4 v = b4[i];
            sum += v.x + v.y + v.z + v.w;
            sq += v.x * v.x + v.y * v.y + v.z * v.z + v.w * v.w;
        }
    } else {
        const long long start = ((long long)s * group_size) / splits;
        const long long end = ((long long)(s + 1) * group_size) / splits;
        for (long long i = start + threadIdx.x; i < end; i += blockDim.x) {
            float v = base[i];
            sum += v;
            sq += v * v;
        }
    }

    __shared__ float ws[32];
    __shared__ float wq[32];
    const int lane = threadIdx.x & 31;
    const int wid = threadIdx.x >> 5;

    sum = warp_sum(sum);
    sq = warp_sum(sq);
    if (lane == 0) { ws[wid] = sum; wq[wid] = sq; }
    __syncthreads();

    if (wid == 0) {
        const int nw = blockDim.x >> 5;
        float a = (lane < nw) ? ws[lane] : 0.f;
        float b = (lane < nw) ? wq[lane] : 0.f;
        a = warp_sum(a);
        b = warp_sum(b);
        if (lane == 0) {
            partial[(g * splits + s) * 2 + 0] = a;
            partial[(g * splits + s) * 2 + 1] = b;
        }
    }
}

__global__ void gn_finalize_kernel(const float* __restrict__ partial,
                                   float* __restrict__ ms,
                                   int splits, float inv_n, float eps) {
    const int g = blockIdx.x;
    float s = 0.f, q = 0.f;
    for (int i = threadIdx.x; i < splits; i += 32) {
        s += partial[(long long)(g * splits + i) * 2 + 0];
        q += partial[(long long)(g * splits + i) * 2 + 1];
    }
    s = warp_sum(s);
    q = warp_sum(q);
    if (threadIdx.x == 0) {
        float m = s * inv_n;
        float var = q * inv_n - m * m;
        var = fmaxf(var, 0.f);
        ms[2 * g + 0] = m;
        ms[2 * g + 1] = rsqrtf(var + eps);
    }
}

template <bool VEC, bool RESID>
__global__ void gn_silu_apply_kernel(const float* __restrict__ y,
                                     const float* __restrict__ res,
                                     const float* __restrict__ gamma,
                                     const float* __restrict__ beta,
                                     const float* __restrict__ ms,
                                     float* __restrict__ out,
                                     int C, int cpg, int G, long long HW) {
    const int nc = blockIdx.y;
    const int c = nc % C;
    const int n = nc / C;
    const int gid = n * G + (c / cpg);

    const float mean = ms[2 * gid + 0];
    const float rstd = ms[2 * gid + 1];
    const float gam = gamma[c];
    const float bet = beta[c];
    const float scale = gam * rstd;
    const float bias = bet - mean * scale;

    const long long off = (long long)nc * HW;

    if (VEC) {
        const long long nvec = HW >> 2;
        const long long idx = (long long)blockIdx.x * blockDim.x + threadIdx.x;
        if (idx < nvec) {
            const float4* __restrict__ y4 = reinterpret_cast<const float4*>(y + off);
            float4* __restrict__ o4 = reinterpret_cast<float4*>(out + off);
            float4 v = y4[idx];
            float a0 = v.x * scale + bias;
            float a1 = v.y * scale + bias;
            float a2 = v.z * scale + bias;
            float a3 = v.w * scale + bias;
            a0 = a0 / (1.f + expf(-a0));
            a1 = a1 / (1.f + expf(-a1));
            a2 = a2 / (1.f + expf(-a2));
            a3 = a3 / (1.f + expf(-a3));
            if (RESID) {
                const float4* __restrict__ r4 = reinterpret_cast<const float4*>(res + off);
                float4 r = r4[idx];
                a0 += r.x; a1 += r.y; a2 += r.z; a3 += r.w;
            }
            float4 o;
            o.x = a0; o.y = a1; o.z = a2; o.w = a3;
            o4[idx] = o;
        }
    } else {
        const long long idx = (long long)blockIdx.x * blockDim.x + threadIdx.x;
        if (idx < HW) {
            float a = y[off + idx] * scale + bias;
            a = a / (1.f + expf(-a));
            if (RESID) a += res[off + idx];
            out[off + idx] = a;
        }
    }
}

static torch::Tensor gn_silu_impl(const torch::Tensor& y,
                                  const torch::Tensor& gamma,
                                  const torch::Tensor& beta,
                                  const torch::Tensor* res,
                                  int64_t num_groups,
                                  double eps) {
    TORCH_CHECK(y.is_cuda(), "input must be CUDA");
    TORCH_CHECK(y.scalar_type() == at::kFloat, "input must be float32");
    TORCH_CHECK(y.dim() == 4, "input must be 4D (N,C,H,W)");
    TORCH_CHECK(y.is_contiguous(), "input must be contiguous");

    const int64_t N = y.size(0);
    const int64_t C = y.size(1);
    const int64_t HW = y.size(2) * y.size(3);
    TORCH_CHECK(C % num_groups == 0, "C must be divisible by num_groups");
    const int64_t cpg = C / num_groups;
    const int64_t group_size = cpg * HW;
    const int64_t NG = N * num_groups;

    auto out = torch::empty_like(y);
    auto opts = y.options();
    auto ms = torch::empty({NG * 2}, opts);

    int splits = (int)std::min<int64_t>(32, std::max<int64_t>(1, 2048 / std::max<int64_t>(NG, 1)));
    while (splits > 1 && group_size / splits < 4096) splits >>= 1;
    if (splits < 1) splits = 1;

    auto partial = torch::empty({NG * (int64_t)splits * 2}, opts);
    auto stream = at::cuda::getCurrentCUDAStream();

    const bool vec_stats = (group_size % 4 == 0);
    dim3 sgrid((unsigned)splits, (unsigned)NG);
    if (vec_stats) {
        gn_stats_partial_kernel<true><<<sgrid, 256, 0, stream>>>(
            y.data_ptr<float>(), partial.data_ptr<float>(), (long long)group_size, splits);
    } else {
        gn_stats_partial_kernel<false><<<sgrid, 256, 0, stream>>>(
            y.data_ptr<float>(), partial.data_ptr<float>(), (long long)group_size, splits);
    }
    C10_CUDA_KERNEL_LAUNCH_CHECK();

    gn_finalize_kernel<<<(unsigned)NG, 32, 0, stream>>>(
        partial.data_ptr<float>(), ms.data_ptr<float>(), splits,
        (float)(1.0 / (double)group_size), (float)eps);
    C10_CUDA_KERNEL_LAUNCH_CHECK();

    const bool vec_apply = (HW % 4 == 0);
    const int block = 256;
    const int64_t units = vec_apply ? (HW / 4) : HW;
    dim3 agrid((unsigned)CDIV(units, (int64_t)block), (unsigned)(N * C));
    const float* rptr = (res != nullptr) ? res->data_ptr<float>() : y.data_ptr<float>();

    if (vec_apply) {
        if (res != nullptr)
            gn_silu_apply_kernel<true, true><<<agrid, block, 0, stream>>>(
                y.data_ptr<float>(), rptr, gamma.data_ptr<float>(), beta.data_ptr<float>(),
                ms.data_ptr<float>(), out.data_ptr<float>(),
                (int)C, (int)cpg, (int)num_groups, (long long)HW);
        else
            gn_silu_apply_kernel<true, false><<<agrid, block, 0, stream>>>(
                y.data_ptr<float>(), rptr, gamma.data_ptr<float>(), beta.data_ptr<float>(),
                ms.data_ptr<float>(), out.data_ptr<float>(),
                (int)C, (int)cpg, (int)num_groups, (long long)HW);
    } else {
        if (res != nullptr)
            gn_silu_apply_kernel<false, true><<<agrid, block, 0, stream>>>(
                y.data_ptr<float>(), rptr, gamma.data_ptr<float>(), beta.data_ptr<float>(),
                ms.data_ptr<float>(), out.data_ptr<float>(),
                (int)C, (int)cpg, (int)num_groups, (long long)HW);
        else
            gn_silu_apply_kernel<false, false><<<agrid, block, 0, stream>>>(
                y.data_ptr<float>(), rptr, gamma.data_ptr<float>(), beta.data_ptr<float>(),
                ms.data_ptr<float>(), out.data_ptr<float>(),
                (int)C, (int)cpg, (int)num_groups, (long long)HW);
    }
    C10_CUDA_KERNEL_LAUNCH_CHECK();
    return out;
}

// Stage 1 (half NHWC): each thread owns one float4 == 8 halves == one group of
// one pixel.  grid = (splits, N), block = 256.
__global__ void gn_stats_nhwc_h_kernel(const __half* __restrict__ y,
                                       float* __restrict__ partial,
                                       int C, int cpg, int G, long long HW,
                                       int splits, int VPP) {
    const int n = blockIdx.y;
    const int s = blockIdx.x;
    const long long p_start = ((long long)s * HW) / splits;
    const long long p_end   = ((long long)(s + 1) * HW) / splits;

    const int tid  = threadIdx.x;
    const int v    = tid % VPP;          // channel-vector index (channels v*8..v*8+7)
    const int prow = tid / VPP;          // pixel row handled by this thread
    const int rows = blockDim.x / VPP;

    float sum = 0.f, sq = 0.f;
    const float4* __restrict__ y4 = reinterpret_cast<const float4*>(y);
    const long long base_v = (long long)n * HW * (long long)VPP;

    for (long long p = p_start + prow; p < p_end; p += rows) {
        float4 raw = y4[base_v + p * (long long)VPP + v];
        const __half2* h = reinterpret_cast<const __half2*>(&raw);
#pragma unroll
        for (int k = 0; k < 4; ++k) {
            float2 f = __half22float2(h[k]);
            sum += f.x + f.y;
            sq  += f.x * f.x + f.y * f.y;
        }
    }

    extern __shared__ float sm[];       // 2 * blockDim.x floats
    sm[tid] = sum;
    sm[blockDim.x + tid] = sq;
    __syncthreads();

    if (tid < VPP) {
        float a = 0.f, b = 0.f;
        for (int r = 0; r < rows; ++r) {
            a += sm[r * VPP + tid];
            b += sm[blockDim.x + r * VPP + tid];
        }
        sm[tid] = a;
        sm[blockDim.x + tid] = b;
    }
    __syncthreads();

    const int vpg = cpg >> 3;            // vectors per group (cpg % 8 == 0)
    if (tid < G) {
        float a = 0.f, b = 0.f;
        for (int k = 0; k < vpg; ++k) {
            a += sm[tid * vpg + k];
            b += sm[blockDim.x + tid * vpg + k];
        }
        const long long gid = (long long)n * G + tid;
        partial[(gid * splits + s) * 2 + 0] = a;
        partial[(gid * splits + s) * 2 + 1] = b;
    }
}

// Stage 3a (half NHWC -> half NHWC): affine + SiLU, per-channel scale/bias in smem.
__global__ void gn_silu_apply_nhwc_h_kernel(const __half* __restrict__ y,
                                            const float* __restrict__ gamma,
                                            const float* __restrict__ beta,
                                            const float* __restrict__ ms,
                                            __half* __restrict__ out,
                                            int C, int cpg, int G,
                                            long long HW, int VPP) {
    extern __shared__ float sc[];        // [C] scale, [C] bias
    float* sb = sc + C;
    const int n = blockIdx.y;

    for (int c = threadIdx.x; c < C; c += blockDim.x) {
        const int gid = n * G + c / cpg;
        const float mean = ms[2 * gid + 0];
        const float rstd = ms[2 * gid + 1];
        const float s = gamma[c] * rstd;
        sc[c] = s;
        sb[c] = beta[c] - mean * s;
    }
    __syncthreads();

    const long long total_v = HW * (long long)VPP;
    const long long idx = (long long)blockIdx.x * blockDim.x + threadIdx.x;
    if (idx >= total_v) return;

    const int cb = (int)(idx % (long long)VPP) * 8;
    const float4* __restrict__ y4 = reinterpret_cast<const float4*>(y) + (long long)n * total_v;
    float4* __restrict__ o4 = reinterpret_cast<float4*>(out) + (long long)n * total_v;

    float4 raw = y4[idx];
    const __half2* h = reinterpret_cast<const __half2*>(&raw);
    float4 packed;
    __half2* oh = reinterpret_cast<__half2*>(&packed);
#pragma unroll
    for (int k = 0; k < 4; ++k) {
        float2 f = __half22float2(h[k]);
        float a0 = fmaf(f.x, sc[cb + 2 * k], sb[cb + 2 * k]);
        float a1 = fmaf(f.y, sc[cb + 2 * k + 1], sb[cb + 2 * k + 1]);
        a0 = a0 / (1.f + expf(-a0));
        a1 = a1 / (1.f + expf(-a1));
        oh[k] = __floats2half2_rn(a0, a1);
    }
    o4[idx] = packed;
}

// Stage 3b (half NHWC + fp32 NCHW residual -> fp32 NCHW): shared-memory tile
// transpose.  grid = (ceil(HW/P), N), block = 256.
__global__ void gn_silu_add_nhwc_h_kernel(const __half* __restrict__ y,
                                          const float* __restrict__ resid,
                                          const float* __restrict__ gamma,
                                          const float* __restrict__ beta,
                                          const float* __restrict__ ms,
                                          float* __restrict__ out,
                                          int C, int cpg, int G,
                                          long long HW, int VPP, int P, int S) {
    extern __shared__ float smf[];
    float* sc = smf;
    float* sb = smf + C;
    __half* tile = reinterpret_cast<__half*>(smf + 2 * C);

    const int n = blockIdx.y;
    const long long p0 = (long long)blockIdx.x * P;
    const int Pc = (int)((HW - p0) < (long long)P ? (HW - p0) : (long long)P);

    for (int c = threadIdx.x; c < C; c += blockDim.x) {
        const int gid = n * G + c / cpg;
        const float mean = ms[2 * gid + 0];
        const float rstd = ms[2 * gid + 1];
        const float s = gamma[c] * rstd;
        sc[c] = s;
        sb[c] = beta[c] - mean * s;
    }
    __syncthreads();

    // ---- Phase A: coalesced NHWC read, normalize + SiLU, store into smem tile
    const float4* __restrict__ y4 =
        reinterpret_cast<const float4*>(y) + ((long long)n * HW + p0) * (long long)VPP;
    const int totv = Pc * VPP;
    for (int i = threadIdx.x; i < totv; i += blockDim.x) {
        const int p  = i / VPP;
        const int cb = (i % VPP) * 8;
        float4 raw = y4[i];
        const __half2* h = reinterpret_cast<const __half2*>(&raw);
        float4 packed;
        __half2* oh = reinterpret_cast<__half2*>(&packed);
#pragma unroll
        for (int k = 0; k < 4; ++k) {
            float2 f = __half22float2(h[k]);
            float a0 = fmaf(f.x, sc[cb + 2 * k], sb[cb + 2 * k]);
            float a1 = fmaf(f.y, sc[cb + 2 * k + 1], sb[cb + 2 * k + 1]);
            a0 = a0 / (1.f + expf(-a0));
            a1 = a1 / (1.f + expf(-a1));
            oh[k] = __floats2half2_rn(a0, a1);
        }
        *reinterpret_cast<float4*>(tile + (long long)p * S + cb) = packed;
    }
    __syncthreads();

    // ---- Phase B: channel-major, coalesced fp32 NCHW residual add + store
    const int lane = threadIdx.x & 31;
    const int wid  = threadIdx.x >> 5;
    const int nw   = blockDim.x >> 5;
    for (int c = wid; c < C; c += nw) {
        const float* __restrict__ rp = resid + ((long long)n * C + c) * HW + p0;
        float* __restrict__ op = out + ((long long)n * C + c) * HW + p0;
        for (int p = lane; p < Pc; p += 32) {
            float v = __half2float(tile[(long long)p * S + c]);
            op[p] = v + rp[p];
        }
    }
}

// --------------------------- fp16 host launchers ---------------------------
static void gn_stats_h(const torch::Tensor& y, torch::Tensor& ms,
                       int64_t N, int64_t C, int64_t HW, int64_t num_groups,
                       int64_t cpg, double eps, cudaStream_t stream) {
    const int VPP = (int)(C / 8);
    const int64_t NG = N * num_groups;

    int splits = (int)std::max<int64_t>(1, 1024 / std::max<int64_t>(N, 1));
    if (splits > 1024) splits = 1024;
    while (splits > 1 && HW / splits < 128) splits >>= 1;
    if (splits < 1) splits = 1;

    auto partial = torch::empty({NG * (int64_t)splits * 2},
                                y.options().dtype(at::kFloat));

    const int block = 256;
    dim3 sgrid((unsigned)splits, (unsigned)N);
    gn_stats_nhwc_h_kernel<<<sgrid, block, 2 * block * sizeof(float), stream>>>(
        reinterpret_cast<const __half*>(y.data_ptr<at::Half>()),
        partial.data_ptr<float>(), (int)C, (int)cpg, (int)num_groups,
        (long long)HW, splits, VPP);
    C10_CUDA_KERNEL_LAUNCH_CHECK();

    gn_finalize_kernel<<<(unsigned)NG, 32, 0, stream>>>(
        partial.data_ptr<float>(), ms.data_ptr<float>(), splits,
        (float)(1.0 / (double)(cpg * HW)), (float)eps);
    C10_CUDA_KERNEL_LAUNCH_CHECK();
}

torch::Tensor gn_silu_h(torch::Tensor y, torch::Tensor gamma, torch::Tensor beta,
                        int64_t num_groups, double eps) {
    TORCH_CHECK(y.is_cuda() && y.scalar_type() == at::kHalf, "y must be cuda half");
    TORCH_CHECK(y.dim() == 4, "y must be 4D");
    TORCH_CHECK(y.is_contiguous(at::MemoryFormat::ChannelsLast), "y must be channels_last");

    const int64_t N = y.size(0), C = y.size(1);
    const int64_t HW = y.size(2) * y.size(3);
    TORCH_CHECK(C % num_groups == 0, "C % G");
    const int64_t cpg = C / num_groups;
    TORCH_CHECK(C % 8 == 0 && cpg % 8 == 0, "need C%8==0 and cpg%8==0");

    auto stream = at::cuda::getCurrentCUDAStream();
    auto fopts = y.options().dtype(at::kFloat);
    auto ms = torch::empty({N * num_groups * 2}, fopts);
    gn_stats_h(y, ms, N, C, HW, num_groups, cpg, eps, stream);

    auto out = torch::empty(y.sizes(),
                            y.options().memory_format(at::MemoryFormat::ChannelsLast));

    const int VPP = (int)(C / 8);
    const int block = 256;
    const int64_t total_v = HW * (int64_t)VPP;
    dim3 agrid((unsigned)CDIV(total_v, (int64_t)block), (unsigned)N);
    gn_silu_apply_nhwc_h_kernel<<<agrid, block, 2 * C * sizeof(float), stream>>>(
        reinterpret_cast<const __half*>(y.data_ptr<at::Half>()),
        gamma.data_ptr<float>(), beta.data_ptr<float>(), ms.data_ptr<float>(),
        reinterpret_cast<__half*>(out.data_ptr<at::Half>()),
        (int)C, (int)cpg, (int)num_groups, (long long)HW, VPP);
    C10_CUDA_KERNEL_LAUNCH_CHECK();
    return out;
}

    TORCH_CHECK(y.is_contiguous(at::MemoryFormat::ChannelsLast), "y must be channels_last");
    TORCH_CHECK(residual.is_cuda() && residual.scalar_type() == at::kFloat,
                "residual must be cuda float32");
    TORCH_CHECK(residual.is_contiguous(), "residual must be NCHW contiguous");
    TORCH_CHECK(residual.numel() == y.numel(), "residual numel mismatch");

    const int64_t N = y.size(0), C = y.size(1);
    const int64_t HW = y.size(2) * y.size(3);
    TORCH_CHECK(C % num_groups == 0, "C % G");
    const int64_t cpg = C / num_groups;
    TORCH_CHECK(C % 8 == 0 && cpg % 8 == 0, "need C%8==0 and cpg%8==0");

    auto stream = at::cuda::getCurrentCUDAStream();
    auto fopts = y.options().dtype(at::kFloat);
    auto ms = torch::empty({N * num_groups * 2}, fopts);
    gn_stats_h(y, ms, N, C, HW, num_groups, cpg, eps, stream);

    auto out = torch::empty(y.sizes(),
                            y.options().dtype(at::kFloat)
                                       .memory_format(at::MemoryFormat::Contiguous));

    const int VPP = (int)(C / 8);
    const int S = (int)C + 8;              // padded half stride (16B aligned)
    int P = 64;
    while (P > 4 && (size_t)(2 * C * sizeof(float) + (size_t)P * S * 2) > 46000u) P >>= 1;

    const int block = 256;
    const size_t smem = 2 * C * sizeof(float) + (size_t)P * S * 2;
    dim3 grid((unsigned)CDIV(HW, (int64_t)P), (unsigned)N);

        reinterpret_cast<const __half*>(y.data_ptr<at::Half>()),
        residual.data_ptr<float>(), gamma.data_ptr<float>(), beta.data_ptr<float>(),
        ms.data_ptr<float>(), out.data_ptr<float>(),
        (int)C, (int)cpg, (int)num_groups, (long long)HW, VPP, P, S);
    C10_CUDA_KERNEL_LAUNCH_CHECK();
    return out;
}

class ModelNew(nn.Module):
    """
    FP16 tensor-core downcast of the conv pair (channels_last / NHWC) plus
    half-precision NHWC GroupNorm+SiLU(+residual) custom kernels, with a
    first-call numerics gate + timing autotune falling back to the fp32 path.
    """

    _ATOL_GATE = 2.5e-3

    def __init__(self):
        super().__init__()
        self.ext = _ext
        self.num_groups = 32
        torch.backends.cudnn.benchmark = True
        self._mode = None            # None -> autotune on first call
        self._wcache = {}

    # ---------------- helpers ----------------
    def _half_w(self, w):
        key = (w.data_ptr(), w.shape, w._version)
        h = self._wcache.get(key)
        if h is None:
            h = w.to(dtype=torch.float16, memory_format=torch.channels_last)
            if len(self._wcache) > 8:
                self._wcache.clear()
            self._wcache[key] = h
        return h

    def _fp16_ok(self, x):
        C = x.size(1)
        G = self.num_groups
        if C % G != 0:
            return False
        cpg = C // G
        vpp = C // 8
        return (C % 8 == 0) and (cpg % 8 == 0) and vpp <= 256 and (256 % vpp == 0) and G <= 256

    # ---------------- fp32 (fallback / reference) path ----------------
    def _forward_fp32(self, x_c, w1, g1, b1, w2, g2, b2, eps_f):
        out = F.conv2d(x_c, w1, bias=None, stride=1, padding=1)
        out = out if out.is_contiguous() else out.contiguous()
        out = self.ext.gn_silu(out, g1, b1, self.num_groups, eps_f)
        out = F.conv2d(out, w2, bias=None, stride=1, padding=1)
        out = out if out.is_contiguous() else out.contiguous()
        return self.ext.gn_silu_add(out, x_c, g2, b2, self.num_groups, eps_f)

    # ---------------- fp16 tensor-core path ----------------
    def _forward_fp16(self, x_c, w1, g1, b1, w2, g2, b2, eps_f):
        xh = x_c.to(dtype=torch.float16, memory_format=torch.channels_last)
        w1h = self._half_w(w1)
        w2h = self._half_w(w2)

        y1 = F.conv2d(xh, w1h, bias=None, stride=1, padding=1)
        if not y1.is_contiguous(memory_format=torch.channels_last):
            y1 = y1.contiguous(memory_format=torch.channels_last)
        h1 = self.ext.gn_silu_h(y1, g1, b1, self.num_groups, eps_f)

        y2 = F.conv2d(h1, w2h, bias=None, stride=1, padding=1)
        if not y2.is_contiguous(memory_format=torch.channels_last):
            y2 = y2.contiguous(memory_format=torch.channels_last)
        return self.ext.gn_silu_add_h(y2, x_c, g2, b2, self.num_groups, eps_f)

    # ---------------- numerics gate + timing autotune ----------------
    def _time(self, fn, args, iters=3):
        for _ in range(2):
            fn(*args)
        torch.cuda.synchronize()
        s = torch.cuda.Event(enable_timing=True)
        e = torch.cuda.Event(enable_timing=True)
        s.record()
        for _ in range(iters):
            fn(*args)
        e.record()
        torch.cuda.synchronize()
        return s.elapsed_time(e) / iters

    def _autotune(self, args):
        try:
            o32 = self._forward_fp32(*args)
            o16 = self._forward_fp16(*args)
            torch.cuda.synchronize()
            diff = (o16.float() - o32).abs().max().item()
            if diff > self._ATOL_GATE:
                self._mode = "fp32"
                return o32
            t32 = self._time(self._forward_fp32, args)
            t16 = self._time(self._forward_fp16, args)
            self._mode = "fp16" if t16 * 1.02 < t32 else "fp32"
            return o16 if self._mode == "fp16" else o32
        except Exception:
            self._mode = "fp32"
            return self._forward_fp32(*args)

    # ---------------- public API ----------------
    def forward(self, x, conv1_weight, norm1_weight, norm1_bias,
                conv2_weight, norm2_weight, norm2_bias, eps):
        if isinstance(eps, torch.Tensor):
            eps_f = float(eps.item())
        else:
            eps_f = float(eps)

        x_c = x if x.is_contiguous() else x.contiguous()
        w1 = conv1_weight if conv1_weight.is_contiguous() else conv1_weight.contiguous()
        w2 = conv2_weight if conv2_weight.is_contiguous() else conv2_weight.contiguous()
        g1 = norm1_weight if norm1_weight.is_contiguous() else norm1_weight.contiguous()
        b1 = norm1_bias if norm1_bias.is_contiguous() else norm1_bias.contiguous()
        g2 = norm2_weight if norm2_weight.is_contiguous() else norm2_weight.contiguous()
        b2 = norm2_bias if norm2_bias.is_contiguous() else norm2_bias.contiguous()

        args = (x_c, w1, g1, b1, w2, g2, b2, eps_f)

        if self._mode is None:
            if x_c.dtype != torch.float32 or not self._fp16_ok(x_c):
                self._mode = "fp32"
                return self._forward_fp32(*args)
            return self._autotune(args)

        if self._mode == "fp16":
            return self._forward_fp16(*args)
        return self._forward_fp32(*args)
