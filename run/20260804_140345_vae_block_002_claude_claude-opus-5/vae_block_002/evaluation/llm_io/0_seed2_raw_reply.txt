# =============================================================================
# ModelNew — SOL 002_vae_conv3x3_groupnorm_silu_residual_fused
# GRANULARITY: (D) FULL FORWARD REWRITE.
#  1) Granularity: (D) fully rewrite forward.
#  2) Replaced ops: both F.conv2d(3x3,pad1,256->256), both F.group_norm(32),
#     both F.silu, and the residual add.  No cuDNN / at::conv2d anywhere.
#  3) Fusion map:
#     * transpose_w_kernel : packs W(256,2304) -> Wp(2304,256) once per call so
#       the implicit-GEMM A-tile loads are coalesced (2.36MB, negligible).
#     * conv_kernel : shared-mem tiled implicit GEMM (BM128 x BN128 x BK8,
#       8x8 register tile) with im2col addressing generated on the fly (the
#       im2col matrix is never materialised), FUSED with the GroupNorm
#       first/second-moment reduction in its epilogue -- moments are produced
#       while accumulators are still in registers, so GroupNorm never re-reads
#       the conv output.  Deterministic: each (tile,group) slot is written by
#       exactly one block; no global atomics.
#     * stats_kernel : partial moments -> per-(n,c) folded affine (scale,bias)
#       (mean/var/eps/weight/bias collapsed into one multiply-add), fp64 accum.
#     * act_kernel : SiLU(affine) in one pass -> input of conv2 (norm+act fused).
#     * final_kernel : SiLU(affine) + residual add fused, single pass -> output.
#  4) Left in PyTorch: only torch::empty allocations and shape/dtype checks -
#     no GPU math remains in ATen.
# Precision: fp32 storage AND fp32 FMA arithmetic throughout the convolutions
# (no TF32/half downcast); group statistics accumulate in fp64.
# =============================================================================

_CUDA_SRC = r"""
#include <torch/extension.h>
#include <ATen/cuda/CUDAContext.h>
#include <c10/cuda/CUDAException.h>
#include <cuda_runtime.h>

#define CCH      256
#define KDIM     2304
#define NGROUPS  32
#define CPG      8
#define BM       128
#define BN       128
#define BK       8

__global__ void transpose_w_kernel(const float* __restrict__ in,
                                   float* __restrict__ out)
{
    __shared__ float tile[32][33];
    const int x = blockIdx.x * 32 + threadIdx.x;   /* k */
    const int y = blockIdx.y * 32 + threadIdx.y;   /* m */
#pragma unroll
    for (int j = 0; j < 32; j += 8)
        tile[threadIdx.y + j][threadIdx.x] = in[(size_t)(y + j) * KDIM + x];
    __syncthreads();
    const int x2 = blockIdx.y * 32 + threadIdx.x;  /* m */
    const int y2 = blockIdx.x * 32 + threadIdx.y;  /* k */
#pragma unroll
    for (int j = 0; j < 32; j += 8)
        out[(size_t)(y2 + j) * CCH + x2] = tile[threadIdx.x][threadIdx.y + j];
}

__global__ __launch_bounds__(256, 2)
void conv_kernel(const float* __restrict__ X,
                 const float* __restrict__ Wp,
                 float* __restrict__ Out,
                 float* __restrict__ part,
                 int H, int W, int P, int tilesPerImage)
{
    const int tid    = threadIdx.x;
    const int nTile  = blockIdx.x;
    const int mBlock = blockIdx.y;
    const int n      = nTile / tilesPerImage;
    const int tile   = nTile - n * tilesPerImage;
    const int pbase  = tile * BN;
    const int mb     = mBlock * BM;

    __shared__ float As[BK][BM];
    __shared__ float Bs[BK][BN];
    __shared__ float ssum[16];
    __shared__ float ssq[16];

    const int tx = tid & 15;
    const int ty = tid >> 4;
    const int aK = tid >> 5;
    const int aM = (tid & 31) * 4;
    const int bK = tid >> 5;
    const int bJ = tid & 31;

    int  ph[4], pw[4];
    bool pv[4];
#pragma unroll
    for (int r = 0; r < 4; ++r) {
        int p = pbase + bJ + 32 * r;
        pv[r] = (p < P);
        int pp = pv[r] ? p : 0;
        ph[r] = pp / W;
        pw[r] = pp - ph[r] * W;
    }

    const size_t xbase = (size_t)n * CCH * (size_t)P;

    float acc[8][8];
#pragma unroll
    for (int i = 0; i < 8; ++i)
#pragma unroll
        for (int j = 0; j < 8; ++j) acc[i][j] = 0.f;

    float4 aReg;
    float  bReg[4];

    /* ---- prologue: prefetch first k-tile ---- */
    {
        aReg = *(const float4*)(Wp + (size_t)(0 + aK) * CCH + mb + aM);
        int k  = 0 + bK;
        int ic = k / 9;
        int t  = k - ic * 9;
        int kh = t / 3;
        int kw = t - kh * 3;
        const float* xb = X + xbase + (size_t)ic * (size_t)P;
#pragma unroll
        for (int r = 0; r < 4; ++r) {
            int ih = ph[r] + kh - 1;
            int iw = pw[r] + kw - 1;
            bool ok = pv[r] && ih >= 0 && ih < H && iw >= 0 && iw < W;
            bReg[r] = ok ? xb[(size_t)ih * (size_t)W + iw] : 0.f;
        }
    }

    for (int k0 = 0; k0 < KDIM; k0 += BK) {
        *(float4*)(&As[aK][aM]) = aReg;
#pragma unroll
        for (int r = 0; r < 4; ++r) Bs[bK][bJ + 32 * r] = bReg[r];
        __syncthreads();

        if (k0 + BK < KDIM) {
            int kn = k0 + BK;
            aReg = *(const float4*)(Wp + (size_t)(kn + aK) * CCH + mb + aM);
            int k  = kn + bK;
            int ic = k / 9;
            int t  = k - ic * 9;
            int kh = t / 3;
            int kw = t - kh * 3;
            const float* xb = X + xbase + (size_t)ic * (size_t)P;
#pragma unroll
            for (int r = 0; r < 4; ++r) {
                int ih = ph[r] + kh - 1;
                int iw = pw[r] + kw - 1;
                bool ok = pv[r] && ih >= 0 && ih < H && iw >= 0 && iw < W;
                bReg[r] = ok ? xb[(size_t)ih * (size_t)W + iw] : 0.f;
            }
        }

#pragma unroll
        for (int kk = 0; kk < BK; ++kk) {
            float a[8], b[8];
            *(float4*)(a)     = *(const float4*)(&As[kk][ty * 4]);
            *(float4*)(a + 4) = *(const float4*)(&As[kk][64 + ty * 4]);
            *(float4*)(b)     = *(const float4*)(&Bs[kk][tx * 4]);
            *(float4*)(b + 4) = *(const float4*)(&Bs[kk][64 + tx * 4]);
#pragma unroll
            for (int i = 0; i < 8; ++i)
#pragma unroll
                for (int j = 0; j < 8; ++j)
                    acc[i][j] = fmaf(a[i], b[j], acc[i][j]);
        }
        __syncthreads();
    }

    /* ---------------- epilogue: store + fused GroupNorm moments ---------- */
    if (tid < 16) { ssum[tid] = 0.f; ssq[tid] = 0.f; }
    __syncthreads();

    const bool vecOK = ((P & 3) == 0);
    float s0 = 0.f, q0 = 0.f, s1 = 0.f, q1 = 0.f;

#pragma unroll
    for (int i = 0; i < 8; ++i) {
        int rl = (i < 4) ? (ty * 4 + i) : (64 + ty * 4 + (i - 4));
        int ch = mb + rl;
        float* orow = Out + xbase + (size_t)ch * (size_t)P;
#pragma unroll
        for (int jj = 0; jj < 2; ++jj) {
            int cbase = (jj == 0) ? (tx * 4) : (64 + tx * 4);
            int p = pbase + cbase;
            float v0 = acc[i][jj * 4 + 0];
            float v1 = acc[i][jj * 4 + 1];
            float v2 = acc[i][jj * 4 + 2];
            float v3 = acc[i][jj * 4 + 3];
            if (vecOK && (p + 3) < P) {
                float4 vv = make_float4(v0, v1, v2, v3);
                *(float4*)(orow + p) = vv;
                float s = v0 + v1 + v2 + v3;
                float q = v0 * v0 + v1 * v1 + v2 * v2 + v3 * v3;
                if (i < 4) { s0 += s; q0 += q; } else { s1 += s; q1 += q; }
            } else {
                float vs[4] = {v0, v1, v2, v3};
#pragma unroll
                for (int t2 = 0; t2 < 4; ++t2) {
                    int pp = p + t2;
                    if (pp < P) {
                        float v = vs[t2];
                        orow[pp] = v;
                        if (i < 4) { s0 += v; q0 += v * v; }
                        else       { s1 += v; q1 += v * v; }
                    }
                }
            }
        }
    }

    atomicAdd(&ssum[ty >> 1], s0);
    atomicAdd(&ssq[ty >> 1],  q0);
    atomicAdd(&ssum[8 + (ty >> 1)], s1);
    atomicAdd(&ssq[8 + (ty >> 1)],  q1);
    __syncthreads();

    if (tid < 16) {
        int g = mBlock * 16 + tid;
        size_t off = ((size_t)nTile * NGROUPS + g) * 2;
        part[off]     = ssum[tid];
        part[off + 1] = ssq[tid];
    }
}

__global__ void stats_kernel(const float* __restrict__ part,
                             const float* __restrict__ gw,
                             const float* __restrict__ gb,
                             float* __restrict__ scale,
                             float* __restrict__ bias,
                             int nTiles, long cnt, float eps)
{
    const int blk = blockIdx.x;
    const int n   = blk / NGROUPS;
    const int g   = blk - n * NGROUPS;

    __shared__ double sh_s[256];
    __shared__ double sh_q[256];

    double s = 0.0, q = 0.0;
    for (int t = threadIdx.x; t < nTiles; t += blockDim.x) {
        size_t off = ((size_t)(n * nTiles + t) * NGROUPS + g) * 2;
        s += (double)part[off];
        q += (double)part[off + 1];
    }
    sh_s[threadIdx.x] = s;
    sh_q[threadIdx.x] = q;
    __syncthreads();
    for (int stride = 128; stride > 0; stride >>= 1) {
        if ((int)threadIdx.x < stride) {
            sh_s[threadIdx.x] += sh_s[threadIdx.x + stride];
            sh_q[threadIdx.x] += sh_q[threadIdx.x + stride];
        }
        __syncthreads();
    }
    if ((int)threadIdx.x < CPG) {
        double mean = sh_s[0] / (double)cnt;
        double var  = sh_q[0] / (double)cnt - mean * mean;
        if (var < 0.0) var = 0.0;
        float rstd = (float)(1.0 / sqrt(var + (double)eps));
        int c = g * CPG + (int)threadIdx.x;
        float w = gw[c], b = gb[c];
        scale[n * CCH + c] = w * rstd;
        bias[n * CCH + c]  = b - (float)mean * w * rstd;
    }
}

__global__ void act_kernel(const float* __restrict__ r,
                           const float* __restrict__ scale,
                           const float* __restrict__ bias,
                           float* __restrict__ o, int P)
{
    const int row = blockIdx.x;
    const float s = scale[row];
    const float b = bias[row];
    const float* rp = r + (size_t)row * (size_t)P;
    float* op = o + (size_t)row * (size_t)P;
    for (int i = blockIdx.y * blockDim.x + threadIdx.x; i < P;
         i += blockDim.x * gridDim.y) {
        float y = fmaf(rp[i], s, b);
        op[i] = y / (1.f + __expf(-y));
    }
}

__global__ void final_kernel(const float* __restrict__ r,
                             const float* __restrict__ scale,
                             const float* __restrict__ bias,
                             const float* __restrict__ xres,
                             float* __restrict__ o, int P)
{
    const int row = blockIdx.x;
    const float s = scale[row];
    const float b = bias[row];
    const float* rp = r + (size_t)row * (size_t)P;
    const float* xp = xres + (size_t)row * (size_t)P;
    float* op = o + (size_t)row * (size_t)P;
    for (int i = blockIdx.y * blockDim.x + threadIdx.x; i < P;
         i += blockDim.x * gridDim.y) {
        float y = fmaf(rp[i], s, b);
        op[i] = y / (1.f + __expf(-y)) + xp[i];
    }
}

/* ------------------------------------------------------------------ host */
static void pack_weights(const torch::Tensor& w, torch::Tensor& wp,
                         cudaStream_t stream)
{
    dim3 grid(KDIM / 32, CCH / 32);
    dim3 blk(32, 8);
    transpose_w_kernel<<<grid, blk, 0, stream>>>(w.data_ptr<float>(),
                                                 wp.data_ptr<float>());
    C10_CUDA_KERNEL_LAUNCH_CHECK();
}

static void run_conv(const torch::Tensor& in, const torch::Tensor& wp,
                     torch::Tensor& out, torch::Tensor& part,
                     int B, int H, int W, int P, int tiles,
                     cudaStream_t stream)
{

                                          out.data_ptr<float>(),
                                          part.data_ptr<float>(),
                                          H, W, P, tiles);
    C10_CUDA_KERNEL_LAUNCH_CHECK();
}
