import torch
import torch.nn as nn
import triton
import triton.language as tl


# Define the custom Triton kernel for element-wise addition
@triton.jit
def elementwise_add_kernel(a_ptr, b_ptr, out_ptr, n_elements, BLOCK_SIZE: tl.constexpr):
    pid = tl.program_id(axis=0)
    offsets = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offsets < n_elements
    a = tl.load(a_ptr + offsets, mask=mask)
    b = tl.load(b_ptr + offsets, mask=mask)
    tl.store(out_ptr + offsets, a + b, mask=mask)


def elementwise_add(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    # Ensure contiguity only when needed (contiguous() may allocate and copy)
    a = a if a.is_contiguous() else a.contiguous()
    b = b if b.is_contiguous() else b.contiguous()
    out = torch.empty_like(a)  # allocated on the same device as the inputs
    n_elements = out.numel()
    grid = lambda meta: (triton.cdiv(n_elements, meta["BLOCK_SIZE"]),)
    elementwise_add_kernel[grid](a, b, out, n_elements, BLOCK_SIZE=1024)
    return out
